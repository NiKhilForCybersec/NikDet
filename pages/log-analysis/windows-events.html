Windows Event Log Analysis

Complete reference for Windows Security, System, and Application event log analysis. Master the critical event IDs, understand what they mean, and learn how to use them for threat detection.

Overview

Windows Event Logs are the foundation of endpoint visibility on Windows systems. Security events capture authentication, authorization, and audit information critical for detecting attacks. Understanding which events to monitor and how to analyze them is essential for effective threat detection.

Detection Queries

Brute Force Detection

// KQL - Failed logon brute force
SecurityEvent
| where EventID == 4625
| summarize FailedAttempts=count(),
    TargetAccounts=dcount(TargetUserName),
    Accounts=make_set(TargetUserName, 10)
    by IpAddress, bin(TimeGenerated, 5m)
| where FailedAttempts > 10
| order by FailedAttempts desc

// KQL - Password spray pattern (many users, few attempts each)
SecurityEvent
| where EventID == 4625
| summarize 
    FailedAttempts=count(),
    UniqueUsers=dcount(TargetUserName),
    Users=make_set(TargetUserName, 50)
    by IpAddress, bin(TimeGenerated, 15m)
| where UniqueUsers > 5 and FailedAttempts / UniqueUsers < 3
| order by UniqueUsers desc

// SPL - Brute force detection
index=windows sourcetype="WinEventLog:Security" EventCode=4625
| bin _time span=5m
| stats count as FailedAttempts, dc(TargetUserName) as UniqueUsers,
    values(TargetUserName) as Accounts by IpAddress, _time
| where FailedAttempts > 10
| sort -FailedAttempts

Lateral Movement Detection

// KQL - Network logons from workstation to workstation
SecurityEvent
| where EventID == 4624
| where LogonType == 3 // Network logon
| where IpAddress !startswith "127."
| extend SourceHost = toupper(WorkstationName)
| extend TargetHost = toupper(Computer)
| where SourceHost != TargetHost
| where SourceHost !endswith "SERVER" and TargetHost !endswith "SERVER"
| project TimeGenerated, SourceHost, TargetHost, 
    Account=TargetUserName, IpAddress
| summarize Connections=count() by SourceHost, TargetHost, Account
| where Connections > 1

// KQL - RDP logons from unusual sources
SecurityEvent
| where EventID == 4624
| where LogonType == 10 // RDP
| summarize 
    RDPSessions=count(),
    Sources=make_set(IpAddress)
    by TargetUserName, Computer
| where RDPSessions > 1

// KQL - Admin share access
SecurityEvent
| where EventID == 5140
| where ShareName in ("\\\\*\\C$", "\\\\*\\ADMIN$", "\\\\*\\IPC$")
| project TimeGenerated, Computer, Account=SubjectUserName,
    IpAddress, ShareName
| order by TimeGenerated desc

Privilege Escalation Detection

// KQL - User added to privileged groups
SecurityEvent
| where EventID in (4728, 4732, 4756)
| extend GroupName = TargetUserName
| where GroupName in ("Domain Admins", "Enterprise Admins",
    "Schema Admins", "Administrators", "Account Operators",
    "Backup Operators", "Server Operators")
| project TimeGenerated, Computer, GroupName,
    AddedUser=MemberName, AddedBy=SubjectUserName

// KQL - New admin account created
SecurityEvent
| where EventID == 4720
| project AccountCreated=TargetUserName, CreatedBy=SubjectUserName,
    TimeGenerated, Computer
| join kind=inner (
    SecurityEvent
    | where EventID == 4732
    | where TargetUserName == "Administrators"
    | project AddedUser=MemberName, TimeAdded=TimeGenerated
) on $left.AccountCreated == $right.AddedUser
| where TimeAdded - TimeGenerated < 1h
| project TimeGenerated, AccountCreated, CreatedBy, TimeAdded

// SPL - Privileged group modifications
index=windows sourcetype="WinEventLog:Security" 
(EventCode=4728 OR EventCode=4732 OR EventCode=4756)
TargetUserName IN ("Domain Admins", "Administrators", 
    "Enterprise Admins", "Schema Admins")
| table _time, Computer, EventCode, TargetUserName, 
    MemberName, SubjectUserName

Persistence Detection

// KQL - Suspicious scheduled task creation
SecurityEvent
| where EventID == 4698
| extend TaskContent = tostring(EventData)
| where TaskContent has_any ("powershell", "cmd", "wscript",
    "\\Temp\\", "\\AppData\\", "http://", "https://")
| project TimeGenerated, Computer, Account=SubjectUserName,
    TaskContent

// KQL - Service installation (System log)
Event
| where Source == "Service Control Manager"
| where EventID == 7045
| extend ServiceName = extract("Service Name:\\s+(.*?)\\s+Service", 1, RenderedDescription)
| extend ImagePath = extract("Service File Name:\\s+(.*?)\\s+Service Type", 1, RenderedDescription)
| where ImagePath has_any ("\\Temp\\", "\\AppData\\", "cmd",
    "powershell", "\\Users\\")
| project TimeGenerated, Computer, ServiceName, ImagePath

// SPL - Scheduled task persistence
index=windows sourcetype="WinEventLog:Security" EventCode=4698
| rex field=TaskContent "(?[^<]+)"
| rex field=TaskContent "(?[^<]+)"
| search Command="*powershell*" OR Command="*cmd*" 
    OR Arguments="*http*"
| table _time, host, SubjectUserName, TaskName, Command, Arguments

Credential Access Detection

// KQL - NTLM authentication anomalies
SecurityEvent
| where EventID == 4776
| where Status != "0x0" // Non-success
| summarize 
    FailedAttempts=count(),
    Workstations=make_set(Workstation)
    by TargetUserName
| where FailedAttempts > 5
| order by FailedAttempts desc

// KQL - Pass-the-Hash indicators
// Look for NTLM auth (4624 Type 3) where account doesn't normally 
// authenticate from that workstation
SecurityEvent
| where EventID == 4624
| where LogonType == 3
| where AuthenticationPackageName == "NTLM"
| summarize 
    FirstSeen=min(TimeGenerated),
    Logons=count()
    by TargetUserName, WorkstationName, Computer
| where Logons < 3 // New/rare combination

// KQL - Kerberoasting (many service ticket requests)
SecurityEvent
| where EventID == 4769
| where ServiceName !endswith "$" // Not machine accounts
| where ServiceName !in ("krbtgt", "kadmin")
| where TicketEncryptionType in ("0x17", "0x18") // RC4 encryption
| summarize 
    Requests=count(),
    Services=dcount(ServiceName),
    ServiceList=make_set(ServiceName, 20)
    by IpAddress, TargetUserName
| where Services > 5
| order by Services desc

Sysmon Detection Queries

// KQL - Process injection (CreateRemoteThread)
Event
| where Source == "Microsoft-Windows-Sysmon"
| where EventID == 8
| extend SourceImage = extract("SourceImage:\\s+(.*?)\\s+", 1, RenderedDescription)
| extend TargetImage = extract("TargetImage:\\s+(.*?)\\s+", 1, RenderedDescription)
| where TargetImage !has SourceImage // Not self-injection
| project TimeGenerated, Computer, SourceImage, TargetImage

// KQL - LSASS access (Sysmon 10)
Event
| where Source == "Microsoft-Windows-Sysmon"
| where EventID == 10
| where RenderedDescription has "lsass.exe"
| extend SourceImage = extract("SourceImage:\\s+(.*?)\\s+", 1, RenderedDescription)
| where SourceImage !in ("C:\\Windows\\System32\\lsass.exe",
    "C:\\Windows\\System32\\csrss.exe")
| project TimeGenerated, Computer, SourceImage

// SPL - Sysmon process creation with encoded commands
index=windows sourcetype="WinEventLog:Microsoft-Windows-Sysmon/Operational"
EventCode=1
(CommandLine="*-enc*" OR CommandLine="*-e *" OR CommandLine="*encodedcommand*")
| table _time, host, User, Image, ParentImage, CommandLine

Baseline Guidance

Before hunting, establish what's normal in your environment:

  • Normal admin activity hours - When do IT admins work?
  • Expected service accounts - Which accounts should have Type 3/4/5 logons?
  • Authorized remote access sources - Which IPs/subnets for RDP?
  • Expected scheduled tasks - What tasks exist normally?
  • Normal PowerShell usage - What scripts run regularly?
