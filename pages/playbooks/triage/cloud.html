Azure/Entra ID Alert Triage

Risky Sign-In Detection

// Get sign-in details with risk context
SigninLogs
| where TimeGenerated > ago(24h)
| where UserPrincipalName == "alerteduser@company.com"
| project TimeGenerated, UserPrincipalName, IPAddress, Location, 
    AppDisplayName, DeviceDetail, RiskLevelDuringSignIn,
    RiskEventTypes_V2, ConditionalAccessStatus, MfaDetail
| order by TimeGenerated desc

// Check for post-login suspicious activity
AuditLogs
| where TimeGenerated > ago(24h)
| where InitiatedBy.user.userPrincipalName == "alerteduser@company.com"
| project TimeGenerated, OperationName, TargetResources, Result
| order by TimeGenerated desc

// Check for mail rules created (common post-compromise)
OfficeActivity
| where TimeGenerated > ago(24h)
| where UserId == "alerteduser@company.com"
| where Operation in ("New-InboxRule", "Set-InboxRule", "UpdateInboxRules")
| project TimeGenerated, Operation, Parameters

AWS Alert Triage

CloudTrail Investigation

// User activity timeline
AWSCloudTrail
| where TimeGenerated > ago(24h)
| where UserIdentityArn contains "alerteduser"
| project TimeGenerated, EventName, EventSource, SourceIpAddress,
    UserAgent, RequestParameters, ResponseElements, ErrorCode
| order by TimeGenerated desc

// Credential-related events
AWSCloudTrail
| where TimeGenerated > ago(24h)
| where EventName in ("CreateAccessKey", "CreateLoginProfile", 
    "UpdateLoginProfile", "AttachUserPolicy", "PutUserPolicy",
    "CreateUser", "AddUserToGroup")
| project TimeGenerated, UserIdentityArn, EventName, 
    RequestParameters, SourceIpAddress

// Resource exposure events
AWSCloudTrail
| where TimeGenerated > ago(24h)
| where EventName in ("PutBucketPolicy", "PutBucketAcl",
    "AuthorizeSecurityGroupIngress", "CreateSecurityGroup",
    "ModifySnapshotAttribute", "ModifyImageAttribute")
| project TimeGenerated, UserIdentityArn, EventName, RequestParameters

M365/Office 365 Alert Triage

Key Investigation Queries

// File access and sharing activity
OfficeActivity
| where TimeGenerated > ago(24h)
| where UserId == "alerteduser@company.com"
| where Operation in ("FileDownloaded", "FileAccessed", 
    "AnonymousLinkCreated", "SharingInvitationCreated", "FileSyncDownloadedFull")
| summarize Count=count() by Operation, bin(TimeGenerated, 1h)

// Email forwarding rules
OfficeActivity
| where TimeGenerated > ago(24h)
| where Operation in ("New-InboxRule", "Set-InboxRule", "Set-Mailbox")
| where Parameters has_any ("ForwardTo", "ForwardAsAttachmentTo", "RedirectTo")
| project TimeGenerated, UserId, Operation, Parameters

// Admin activity
OfficeActivity
| where TimeGenerated > ago(24h)
| where RecordType in (1, 8, 14, 15) // Exchange Admin, Azure AD, SharePoint Admin, DLP
| project TimeGenerated, UserId, Operation, OfficeObjectId, ClientIP

Cloud-Specific False Positive Sources

  • VPN/Proxy services: Corporate VPN exit nodes appear as impossible travel
  • Mobile carriers: IP geolocation is unreliable for mobile
  • Cloud shell/console: Azure/AWS console IPs may be flagged
  • Automation accounts: Service principals running on schedules
  • Third-party integrations: Approved apps may look suspicious
  • Migration activities: Large file operations during planned migrations
