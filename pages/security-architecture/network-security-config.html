Network Monitoring

Network Detection Queries

// KQL - Outbound connections to rare destinations
CommonSecurityLog
| where TimeGenerated > ago(24h)
| where DeviceAction == "Allow"
| where DestinationIP !startswith "10." and DestinationIP !startswith "192.168."
| summarize ConnectionCount = count(), Bytes = sum(SentBytes) by DestinationIP, DestinationPort
| where ConnectionCount < 5
| order by Bytes desc

// KQL - Potential lateral movement
CommonSecurityLog
| where DestinationPort in (445, 135, 3389, 22, 5985)
| where SourceIP startswith "10." and DestinationIP startswith "10."
| summarize TargetCount = dcount(DestinationIP) by SourceIP, DestinationPort
| where TargetCount > 10

// KQL - DNS query anomalies
DnsEvents
| where TimeGenerated > ago(1h)
| extend SubdomainLength = strlen(tostring(split(Name, ".")[0]))
| where SubdomainLength > 40
| summarize Count = count() by Name, ClientIP
| where Count > 10

// KQL - Large outbound transfers
CommonSecurityLog
| where SentBytes > 104857600  // 100MB
| where DestinationIP !startswith "10."
| project TimeGenerated, SourceIP, DestinationIP, SentBytes, ApplicationProtocol
