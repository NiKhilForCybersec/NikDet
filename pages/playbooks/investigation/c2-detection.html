INV-007

Command & Control Detection Investigation

Comprehensive investigation procedure for suspected C2 communication. Covers beaconing analysis, DNS tunneling, encrypted channels, protocol abuse, and framework identification including Cobalt Strike, Sliver, and other attack tools.

SeverityCritical
MITRE TacticTA0011
Time1-3 hours
T1071.001 T1071.004 T1571 T1573 T1572 T1090 T1102
Active Threat Indicator

C2 communication indicates an active compromise with an attacker controlling your systems. They can issue commands, exfiltrate data, deploy ransomware, or pivot to other systems at any time. Time is critical - every minute the C2 channel remains open increases risk.

Step 2: Network Traffic Analysis (30 min)

2.1 Connection Pattern Analysis

// KQL - All connections to suspicious destination
let suspiciousIP = "DESTINATION_IP";
let suspiciousDomain = "suspicious.domain.com";
DeviceNetworkEvents
| where TimeGenerated > ago(7d)
| where RemoteIP == suspiciousIP 
    or RemoteUrl has suspiciousDomain
| project TimeGenerated, DeviceName, LocalIP, RemoteIP, RemotePort,
    RemoteUrl, InitiatingProcessFileName, 
    InitiatingProcessCommandLine, InitiatingProcessSHA256,
    SentBytes, ReceivedBytes
| order by TimeGenerated asc

// KQL - Beaconing pattern detection (jitter analysis)
let targetIP = "SUSPICIOUS_IP";
DeviceNetworkEvents
| where TimeGenerated > ago(24h)
| where RemoteIP == targetIP
| project TimeGenerated, DeviceName
| sort by TimeGenerated asc
| extend TimeDiff = datetime_diff('second', TimeGenerated, prev(TimeGenerated))
| where isnotnull(TimeDiff)
| summarize 
    Connections=count(),
    AvgInterval=avg(TimeDiff),
    StdDevInterval=stdev(TimeDiff),
    MinInterval=min(TimeDiff),
    MaxInterval=max(TimeDiff),
    MedianInterval=percentile(TimeDiff, 50)
| extend JitterRatio = StdDevInterval / AvgInterval
| extend Assessment = case(
    JitterRatio < 0.1, "Very Low Jitter - Likely C2 Beaconing",
    JitterRatio < 0.3, "Low Jitter - Possible C2",
    JitterRatio < 0.5, "Moderate Jitter - Investigate",
    "High Jitter - Less Likely C2"
)

// KQL - Data transfer ratio analysis
DeviceNetworkEvents
| where TimeGenerated > ago(24h)
| where RemoteIP == "SUSPICIOUS_IP"
| summarize 
    TotalBytesSent=sum(SentBytes),
    TotalBytesReceived=sum(ReceivedBytes),
    Connections=count()
| extend SentMB = TotalBytesSent / 1048576.0
| extend ReceivedMB = TotalBytesReceived / 1048576.0
| extend UploadRatio = todouble(TotalBytesSent) / todouble(TotalBytesReceived)
| extend Assessment = case(
    UploadRatio > 10, "Very High Upload - Likely Exfiltration",
    UploadRatio > 5, "High Upload - Possible Exfiltration",
    UploadRatio < 0.1, "High Download - Possible Staging",
    "Balanced - Typical C2 Command Traffic"
)

// KQL - Time-of-day analysis
DeviceNetworkEvents
| where RemoteIP == "SUSPICIOUS_IP"
| extend Hour = hourofday(TimeGenerated)
| extend DayOfWeek = dayofweek(TimeGenerated)
| summarize Connections=count() by Hour, DayOfWeek
| order by DayOfWeek asc, Hour asc
// C2 often runs 24/7, legitimate traffic follows business hours

// SPL - Connection timeline with statistics
index=firewall dest_ip="SUSPICIOUS_IP"
| timechart span=1m count as connections
| streamstats window=10 avg(connections) as avg_conn stdev(connections) as stdev_conn
| eval jitter_ratio = stdev_conn / avg_conn

Step 3: Endpoint Investigation (30 min)

3.1 Identify the Malicious Process

// KQL - Process making C2 connections
DeviceNetworkEvents
| where RemoteIP == "C2_IP"
| project TimeGenerated, DeviceName, 
    ProcessName = InitiatingProcessFileName,
    ProcessPath = InitiatingProcessFolderPath,
    CommandLine = InitiatingProcessCommandLine,
    ProcessHash = InitiatingProcessSHA256,
    ParentProcess = InitiatingProcessParentFileName,
    User = InitiatingProcessAccountName
| distinct ProcessName, ProcessPath, ProcessHash, ParentProcess

// KQL - Full process tree analysis
let c2Process = "suspicious.exe";
let affectedHost = "HOSTNAME";
DeviceProcessEvents
| where DeviceName =~ affectedHost
| where FileName =~ c2Process or InitiatingProcessFileName =~ c2Process
| project TimeGenerated, 
    Process = FileName,
    CommandLine = ProcessCommandLine,
    Parent = InitiatingProcessFileName,
    ParentCmdLine = InitiatingProcessCommandLine,
    User = AccountName,
    PID = ProcessId,
    ParentPID = InitiatingProcessId
| order by TimeGenerated asc

// KQL - Hash prevalence analysis
let malwareHash = "SHA256_HASH";
DeviceFileEvents
| where SHA256 == malwareHash
| summarize 
    AffectedDevices = dcount(DeviceName),
    DeviceList = make_set(DeviceName, 20),
    FirstSeen = min(TimeGenerated),
    LastSeen = max(TimeGenerated)
| extend Prevalence = case(
    AffectedDevices > 100, "Common - Likely FP or Widespread",
    AffectedDevices > 10, "Moderate - Investigate Spread",
    AffectedDevices > 1, "Limited - Targeted Attack",
    "Single System - Initial Compromise"
)

// KQL - Process injection detection
DeviceEvents
| where DeviceName =~ "AFFECTED_HOST"
| where ActionType in ("CreateRemoteThreadApiCall", 
    "QueueUserApcRemoteApiCall", "SetThreadContextRemoteApiCall",
    "WriteProcessMemoryApiCall", "NtMapViewOfSectionRemoteApiCall")
| project TimeGenerated, ActionType, 
    SourceProcess = InitiatingProcessFileName,
    TargetProcess = FileName,
    TargetPID = ProcessId

3.2 Persistence Mechanism Check

// KQL - Registry persistence on affected host
let affectedHost = "HOSTNAME";
DeviceRegistryEvents
| where DeviceName =~ affectedHost
| where RegistryKey has_any (
    "\\CurrentVersion\\Run",
    "\\CurrentVersion\\RunOnce",
    "\\CurrentVersion\\RunServices",
    "\\Winlogon\\Shell",
    "\\Winlogon\\Userinit",
    "\\Explorer\\Shell Folders",
    "\\Image File Execution Options"
)
| project TimeGenerated, RegistryKey, RegistryValueName, 
    RegistryValueData, InitiatingProcessFileName

// KQL - Scheduled tasks
DeviceProcessEvents
| where DeviceName =~ "AFFECTED_HOST"
| where FileName =~ "schtasks.exe" and ProcessCommandLine has "/create"
| project TimeGenerated, ProcessCommandLine, AccountName

// KQL - Service installations
DeviceRegistryEvents
| where DeviceName =~ "AFFECTED_HOST"
| where RegistryKey has "\\Services\\" 
    and RegistryValueName == "ImagePath"
| project TimeGenerated, ServiceName = extract("Services\\\\([^\\\\]+)", 1, RegistryKey),
    ImagePath = RegistryValueData

// KQL - WMI event subscriptions
DeviceEvents
| where DeviceName =~ "AFFECTED_HOST"
| where ActionType == "WmiBindEventFilter"
| project TimeGenerated, FilterName = FileName, Consumer = ProcessCommandLine

4.2 Cobalt Strike Specific Detection

// KQL - Cobalt Strike named pipe detection
DeviceEvents
| where ActionType == "NamedPipeEvent"
| where PipeName matches regex @"\\\\\.\\pipe\\(msagent_|postex_|status_|msse-|MSSE-)"
| project TimeGenerated, DeviceName, PipeName, 
    InitiatingProcessFileName

// KQL - Cobalt Strike spawn patterns
DeviceProcessEvents
| where InitiatingProcessFileName =~ "rundll32.exe"
| where InitiatingProcessCommandLine == "" or 
    InitiatingProcessCommandLine matches regex @"^rundll32(\.exe)?$"
| where FileName in~ ("cmd.exe", "powershell.exe", "whoami.exe",
    "net.exe", "ipconfig.exe", "systeminfo.exe")
| project TimeGenerated, DeviceName, FileName, ProcessCommandLine

// KQL - Beacon sleep/jitter patterns
// Look for connections at regular intervals with slight variation
DeviceNetworkEvents
| where RemoteIP == "SUSPECT_IP"
| project TimeGenerated
| sort by TimeGenerated asc
| extend Interval = datetime_diff('second', TimeGenerated, prev(TimeGenerated))
| where Interval between (55 .. 65) // ~60 second beacon with jitter
| count

// SPL - Cobalt Strike watermark detection
index=proxy cs_host="*"
| rex field=cs_user_agent "(?MSIE \d+\.\d+)"
| stats count by watermark, dest_ip

Step 5: DNS-Based C2 Analysis

// KQL - DNS tunneling detection
DnsEvents
| where TimeGenerated > ago(24h)
| extend SubdomainLength = strlen(tostring(split(Name, ".")[0]))
| extend TotalQueryLength = strlen(Name)
| where SubdomainLength > 40 or TotalQueryLength > 100
| summarize 
    QueryCount = count(),
    AvgSubdomainLength = avg(SubdomainLength),
    UniqueSubdomains = dcount(Name),
    SampleQueries = make_set(Name, 5)
    by ClientIP, tostring(split(Name, ".")[-2])
| where QueryCount > 50 or AvgSubdomainLength > 30
| order by QueryCount desc

// KQL - High-frequency DNS to single domain
DnsEvents
| where TimeGenerated > ago(24h)
| extend BaseDomain = strcat(tostring(split(Name, ".")[-2]), ".", 
    tostring(split(Name, ".")[-1]))
| summarize 
    Queries = count(),
    UniqueSubdomains = dcount(Name)
    by ClientIP, BaseDomain
| where Queries > 1000 or UniqueSubdomains > 100
| order by Queries desc

// KQL - TXT record abuse (common for C2)
DnsEvents
| where QueryType == "TXT"
| summarize Queries = count() by ClientIP, Name
| where Queries > 10
| order by Queries desc

// KQL - DNS query entropy analysis
DnsEvents
| extend Subdomain = tostring(split(Name, ".")[0])
| extend CharCount = strlen(Subdomain)
| extend UniqueChars = array_length(make_set(split(Subdomain, "")))
| extend EntropyProxy = todouble(UniqueChars) / todouble(CharCount)
| where CharCount > 20 and EntropyProxy > 0.7
| project TimeGenerated, ClientIP, Name, EntropyProxy

Step 6: Scope Assessment

// KQL - All systems contacting C2 infrastructure
let c2Indicators = dynamic(["IP1", "IP2", "domain1.com", "domain2.com"]);
DeviceNetworkEvents
| where TimeGenerated > ago(30d)
| where RemoteIP in (c2Indicators) or RemoteUrl has_any (c2Indicators)
| summarize 
    FirstContact = min(TimeGenerated),
    LastContact = max(TimeGenerated),
    Connections = count(),
    DataSent = sum(SentBytes),
    DataReceived = sum(ReceivedBytes)
    by DeviceName
| extend DurationDays = datetime_diff('day', LastContact, FirstContact)
| order by FirstContact asc

// KQL - Systems with same malware hash
let malwareHash = "SHA256";
DeviceFileEvents
| where SHA256 == malwareHash
| summarize FirstSeen = min(TimeGenerated), LastSeen = max(TimeGenerated) 
    by DeviceName
| order by FirstSeen asc

// KQL - Lateral movement from patient zero
let patientZero = "FIRST_COMPROMISED_HOST";
let compromiseTime = datetime(2025-01-01T00:00:00Z);
DeviceNetworkEvents
| where TimeGenerated > compromiseTime
| where DeviceName =~ patientZero
| where RemoteIP startswith "10." or RemoteIP startswith "192.168."
| where RemotePort in (445, 135, 139, 3389, 22, 5985, 5986)
| summarize Connections = count() by RemoteIP, RemotePort
| order by Connections desc

Escalation Criteria

Escalate Immediately If:
  • Multiple systems infected (widespread compromise)
  • Server or domain controller compromised
  • Known APT-associated infrastructure detected
  • Evidence of data exfiltration
  • C2 actively receiving and executing commands
  • Ransomware precursor activity observed
  • Cobalt Strike or similar commercial framework detected
  • Persistence mechanisms established
