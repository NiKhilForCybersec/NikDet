Essential Security Searches

Authentication Analysis

// Brute force detection - failed logins followed by success
index=wineventlog EventCode=4625 OR EventCode=4624
| transaction Account_Name maxspan=10m
| where eventcount > 5
| search EventCode=4624
| table _time, Account_Name, src_ip, eventcount

// Password spray detection - same password across accounts
index=wineventlog EventCode=4625 
| bucket _time span=5m
| stats dc(Account_Name) as unique_accounts values(Account_Name) as accounts by _time, Source_Network_Address
| where unique_accounts > 10
| sort -unique_accounts

// After-hours privileged access
index=wineventlog EventCode=4624 Logon_Type=10
| eval hour=strftime(_time, "%H")
| where hour < 6 OR hour > 22
| lookup privileged_users user as Account_Name OUTPUT is_privileged
| where is_privileged="true"
| table _time, Account_Name, Source_Network_Address, ComputerName

Endpoint Threat Detection

// Suspicious process execution from Office apps
index=sysmon EventCode=1
| where match(ParentImage, "(?i)(winword|excel|powerpnt|outlook)\.exe")
| where match(Image, "(?i)(cmd|powershell|wscript|cscript|mshta|regsvr32)\.exe")
| table _time, ComputerName, User, ParentImage, Image, CommandLine

// LSASS access detection (credential dumping)
index=sysmon EventCode=10 TargetImage="*\\lsass.exe"
| where NOT match(SourceImage, "(?i)(MsMpEng|csrss|services|svchost)\.exe$")
| table _time, ComputerName, SourceImage, GrantedAccess, CallTrace

// Encoded PowerShell execution
index=sysmon EventCode=1 Image="*powershell.exe"
| where match(CommandLine, "(?i)(-enc|-encodedcommand|-e\s+[A-Za-z0-9+/=]{20,})")
| table _time, ComputerName, User, ParentImage, CommandLine
| eval decoded=base64decode(mvindex(split(CommandLine, " "), -1))

Network Threat Detection

// Beaconing detection - regular interval connections
index=firewall action=allowed dest_port=443 OR dest_port=80
| bucket _time span=1m
| stats count by src_ip, dest_ip, _time
| stats stdev(count) as stdev avg(count) as avg count as intervals by src_ip, dest_ip
| where stdev < 1 AND intervals > 60
| sort -intervals

// DNS tunneling detection - high volume queries
index=dns
| rex field=query "(?[^\.]+)\.(?.+)"
| stats count dc(subdomain) as unique_subdomains avg(len(subdomain)) as avg_len by domain
| where unique_subdomains > 100 OR avg_len > 30
| sort -unique_subdomains

// Data exfiltration - large outbound transfers
index=firewall direction=outbound
| stats sum(bytes_out) as total_bytes by src_ip, dest_ip
| where total_bytes > 100000000
| eval GB=round(total_bytes/1024/1024/1024, 2)
| sort -total_bytes

ES Correlation Search Example

// Multiple authentication failures from single source
| tstats summariesonly=t count from datamodel=Authentication 
  where Authentication.action="failure" 
  by _time, Authentication.src, Authentication.user span=5m
| rename Authentication.* as *
| where count > 10
| sendalert notable_event 
  param.rule_name="Brute Force Attack"
  param.security_domain="access"
  param.severity="high"

Splunk Security Best Practices

  • Always specify index and time range - Never search all indexes without limits
  • Use data models for dashboards - tstats is dramatically faster
  • Limit real-time searches - They consume significant resources
  • Create summary indexes - For frequent aggregate queries
  • Use event sampling - For initial query development
  • Schedule searches during off-peak - Reduce impact on ad-hoc searches
  • Monitor search performance - Use Job Inspector to optimize
