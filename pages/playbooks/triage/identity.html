Privilege Escalation Triage

Investigation Queries

// Azure AD role assignments
AuditLogs
| where TimeGenerated > ago(24h)
| where OperationName has_any ("Add member to role", "Add eligible member to role")
| extend TargetUser = tostring(TargetResources[0].userPrincipalName)
| extend RoleAssigned = tostring(TargetResources[0].modifiedProperties[1].newValue)
| extend InitiatedBy = tostring(InitiatedBy.user.userPrincipalName)
| project TimeGenerated, InitiatedBy, TargetUser, RoleAssigned, Result

// Windows domain admin additions
SecurityEvent
| where TimeGenerated > ago(24h)
| where EventID == 4728 or EventID == 4732 or EventID == 4756
| where TargetUserName in ("Domain Admins", "Enterprise Admins", "Schema Admins", "Administrators")
| project TimeGenerated, SubjectUserName, MemberName, TargetUserName, Computer

// AWS IAM privilege escalation
AWSCloudTrail
| where TimeGenerated > ago(24h)
| where EventName in ("AttachUserPolicy", "PutUserPolicy", "AttachGroupPolicy", 
    "PutGroupPolicy", "AttachRolePolicy", "PutRolePolicy", "CreatePolicyVersion")
| where RequestParameters has_any ("Admin", "*:*", "iam:*")
| project TimeGenerated, UserIdentityArn, EventName, RequestParameters

Account Creation/Modification Triage

// New account creation
AuditLogs
| where TimeGenerated > ago(24h)
| where OperationName == "Add user"
| extend NewUser = tostring(TargetResources[0].userPrincipalName)
| extend CreatedBy = tostring(InitiatedBy.user.userPrincipalName)
| project TimeGenerated, CreatedBy, NewUser, CorrelationId

// Account modifications
AuditLogs
| where TimeGenerated > ago(24h)
| where OperationName in ("Update user", "Reset password", "Update authentication method")
| extend TargetUser = tostring(TargetResources[0].userPrincipalName)
| extend ModifiedBy = tostring(InitiatedBy.user.userPrincipalName)
| extend Changes = tostring(TargetResources[0].modifiedProperties)
| project TimeGenerated, ModifiedBy, TargetUser, OperationName, Changes

// Windows local account creation
SecurityEvent
| where TimeGenerated > ago(24h)
| where EventID == 4720
| project TimeGenerated, SubjectUserName, TargetUserName, Computer

Credential Change Triage

// Password resets (user-initiated vs admin)
AuditLogs
| where TimeGenerated > ago(24h)
| where OperationName has "password"
| extend TargetUser = tostring(TargetResources[0].userPrincipalName)
| extend InitiatedBy = coalesce(
    tostring(InitiatedBy.user.userPrincipalName),
    tostring(InitiatedBy.app.displayName))
| extend SelfService = TargetUser == InitiatedBy
| project TimeGenerated, InitiatedBy, TargetUser, OperationName, SelfService

// MFA method changes
AuditLogs
| where TimeGenerated > ago(24h)
| where OperationName has_any ("authentication method", "StrongAuthentication")
| extend TargetUser = tostring(TargetResources[0].userPrincipalName)
| project TimeGenerated, TargetUser, OperationName, 
    tostring(TargetResources[0].modifiedProperties)

// Password change after suspicious login
SigninLogs
| where TimeGenerated > ago(24h)
| where RiskLevelDuringSignIn in ("high", "medium")
| project SignInTime=TimeGenerated, UserPrincipalName, RiskLevelDuringSignIn
| join kind=inner (
    AuditLogs
    | where OperationName has "password"
    | extend TargetUser = tostring(TargetResources[0].userPrincipalName)
    | project PasswordChangeTime=TimeGenerated, TargetUser
) on $left.UserPrincipalName == $right.TargetUser
| where PasswordChangeTime between (SignInTime .. (SignInTime + 1h))

Common False Positives

  • HR onboarding: Bulk account creation during hiring waves
  • Scheduled password rotation: Service accounts with automated rotation
  • PAM/PIM activation: Just-in-time privilege activation
  • Group policy updates: Planned security configuration changes
  • Directory sync: Azure AD Connect synchronization activities
  • Self-service password reset: Users resetting their own passwords
