Investigation Steps

Step 1: Identify Mining Activity

KQL - Detect Known Miners (MDE)

// Detect known cryptocurrency miners
DeviceProcessEvents
| where TimeGenerated > ago(7d)
| where FileName in~ ("xmrig.exe", "xmrig", "xmr-stak.exe", "minerd.exe", 
    "cpuminer.exe", "cgminer.exe", "bfgminer.exe", "nbminer.exe", 
    "phoenixminer.exe", "t-rex.exe", "gminer.exe", "lolminer.exe")
    or ProcessCommandLine has_any ("stratum+tcp", "stratum+ssl", "mining.subscribe",
        "--donate-level", "-o pool.", "pool.minexmr", "xmrpool", "moneroocean")
| project TimeGenerated, DeviceName, FileName, ProcessCommandLine, 
    AccountName, InitiatingProcessFileName
| order by TimeGenerated desc

KQL - High CPU Processes

// Find processes with sustained high CPU (potential miners)
DeviceProcessEvents
| where TimeGenerated > ago(24h)
| join kind=inner (
    DevicePerformanceEvents
    | where TimeGenerated > ago(24h)
    | where CounterName == "% Processor Time"
    | where CounterValue > 80
    | summarize HighCPUCount = count() by DeviceName
    | where HighCPUCount > 10  // Sustained high CPU
) on DeviceName
| summarize count() by DeviceName, FileName, ProcessCommandLine
| order by count_ desc

KQL - Mining Pool Connections

// Detect connections to mining pools
DeviceNetworkEvents
| where TimeGenerated > ago(7d)
| where RemotePort in (3333, 4444, 5555, 7777, 8888, 9999, 14444, 45700)
    or RemoteUrl has_any ("pool", "mining", "xmr", "monero", "nicehash", 
        "ethermine", "f2pool", "antpool", "poolin")
| summarize 
    ConnectionCount = count(),
    FirstSeen = min(TimeGenerated),
    LastSeen = max(TimeGenerated),
    Processes = make_set(InitiatingProcessFileName)
    by DeviceName, RemoteIP, RemotePort, RemoteUrl
| order by ConnectionCount desc

SPL - Cryptominer Detection (Splunk)

index=endpoint sourcetype=sysmon EventCode=1
| where process_name IN ("xmrig.exe", "xmrig", "minerd.exe", "cpuminer.exe")
    OR match(command_line, "(?i)(stratum|mining\.subscribe|pool\.|donate-level)")
| stats count by host, process_name, command_line, parent_process_name
| sort -count

KQL - Trace Infection Source

// Find what created/downloaded the miner
let MinerDevices = DeviceProcessEvents
| where TimeGenerated > ago(7d)
| where FileName in~ ("xmrig.exe", "xmrig") or ProcessCommandLine has "stratum"
| distinct DeviceName, FileName;
DeviceFileEvents
| where TimeGenerated > ago(7d)
| where DeviceName in ((MinerDevices | project DeviceName))
| where FileName has_any ("xmrig", "miner", "cpu", "gpu")
| project TimeGenerated, DeviceName, ActionType, FileName, FolderPath, 
    InitiatingProcessFileName, InitiatingProcessCommandLine
| order by TimeGenerated asc

Step 3: Cloud Infrastructure Check

KQL - Azure Unauthorized VM Creation

// Detect unauthorized VM creation (potential cryptojacking)
AzureActivity
| where TimeGenerated > ago(7d)
| where OperationNameValue == "Microsoft.Compute/virtualMachines/write"
| where ActivityStatusValue == "Success"
| project TimeGenerated, Caller, CallerIpAddress, 
    Resource, ResourceGroup, SubscriptionId
| join kind=leftouter (
    AzureActivity
    | where OperationNameValue == "Microsoft.Compute/virtualMachines/delete"
    | project DeleteTime = TimeGenerated, Resource
) on Resource
| where isnull(DeleteTime) or DeleteTime > TimeGenerated + 1h
| order by TimeGenerated desc

AWS CloudTrail - Unauthorized EC2

// CloudTrail query for unauthorized instances
fields @timestamp, userIdentity.arn, sourceIPAddress, 
    requestParameters.instanceType, responseElements.instancesSet.items.0.instanceId
| filter eventName = "RunInstances"
| filter requestParameters.instanceType like /p3|p4|g4|g5|c5|c6/  // GPU/compute optimized
| sort @timestamp desc
| limit 100

Step 4: Scope the Compromise

KQL - Find All Affected Systems

// Identify all systems with mining indicators
union 
(
    DeviceProcessEvents
    | where TimeGenerated > ago(7d)
    | where ProcessCommandLine has_any ("stratum", "pool.", "mining")
    | distinct DeviceName
    | extend Source = "Process"
),
(
    DeviceNetworkEvents
    | where TimeGenerated > ago(7d)
    | where RemotePort in (3333, 4444, 5555, 14444)
    | distinct DeviceName
    | extend Source = "Network"
),
(
    DeviceFileEvents
    | where TimeGenerated > ago(7d)
    | where FileName has_any ("xmrig", "miner")
    | distinct DeviceName
    | extend Source = "File"
)
| summarize Sources = make_set(Source) by DeviceName

PowerShell - Kill Mining Process

# Find and kill mining processes
Get-Process | Where-Object {
    $_.ProcessName -match "xmrig|miner|cgminer|cpuminer" -or
    $_.Path -match "temp|appdata" -and $_.CPU -gt 50
} | Stop-Process -Force

# Remove common persistence
Get-ScheduledTask | Where-Object {
    $_.Actions.Execute -match "xmrig|miner|powershell.*hidden.*http"
} | Unregister-ScheduledTask -Confirm:$false

# Check and remove services
Get-Service | Where-Object {
    $_.DisplayName -match "miner|update service" -and 
    $_.BinaryPathName -match "temp|appdata"
} | Stop-Service -Force
sc.exe delete $_.Name

Interview-Ready Answer

"How do you detect and investigate cryptomining activity?"

My Answer:

"Cryptomining might seem lower severity than ransomware, but it indicates unauthorized access and resource abuse - the same access could be used for worse.

Detection Indicators:
I look for several patterns: High CPU usage - sustained 80-100% CPU without legitimate cause. Network connections to mining pools (ports 3333, 4444, 8333 or connections to known mining pool domains). Process names - miners often have recognizable names or are running from temp directories. PowerShell or scheduled tasks - persistence mechanisms for miners.

Investigation Focus:
The miner itself is almost secondary - I want to know how it got there. Was this a compromised server with weak credentials? An exploited vulnerability? A malicious email? A supply chain compromise? The miner is a symptom; the root cause is the real issue. An attacker who can install a miner can install ransomware.

Response:
I terminate the mining process and remove persistence mechanisms (scheduled tasks, services, startup entries). I block mining pool connections at the firewall. But more importantly, I investigate and remediate the initial access vector. I also check for other compromised systems - cryptomining often spreads across networks to maximize profit.

Business Impact:
While explaining to leadership, I emphasize: Yes, it's 'just' mining, but it's consuming resources we pay for, it indicates a security gap that could be exploited for worse, and in cloud environments, it can generate significant unexpected costs. It's not a low-priority issue."

💡 Pro Tips for Interview:
  • Emphasize that cryptomining indicates compromise - same access could be ransomware
  • Know common mining ports: 3333, 4444, 8333
  • Focus on finding initial access vector, not just removing miner
  • Mention cloud cost implications
