KQL Query Library

Ready-to-use KQL queries for Microsoft Sentinel covering authentication, endpoint, email, cloud, and threat hunting scenarios.

How to Use

Copy these queries and customize the placeholder values (marked in comments) for your environment. All queries are designed for Microsoft Sentinel.

Authentication & Identity

Failed Sign-ins by User

Sentinel
SigninLogs
| where TimeGenerated > ago(7d)
| where ResultType != "0"
| summarize 
    FailedAttempts = count(),
    UniqueIPs = dcount(IPAddress),
    LastAttempt = max(TimeGenerated)
    by UserPrincipalName
| where FailedAttempts > 10
| order by FailedAttempts desc

Impossible Travel Detection

Sentinel
SigninLogs
| where TimeGenerated > ago(24h)
| where ResultType == "0"
| extend Country = tostring(LocationDetails.countryOrRegion)
| order by UserPrincipalName, TimeGenerated asc
| extend PrevCountry = prev(Country), PrevUser = prev(UserPrincipalName), PrevTime = prev(TimeGenerated)
| where UserPrincipalName == PrevUser
| where Country != PrevCountry
| extend MinutesBetween = datetime_diff('minute', TimeGenerated, PrevTime)
| where MinutesBetween < 120
| project TimeGenerated, UserPrincipalName, FromCountry = PrevCountry, ToCountry = Country, MinutesBetween, IPAddress

Sign-ins from Anonymous/Tor IPs

Sentinel
SigninLogs
| where TimeGenerated > ago(24h)
| where RiskLevelDuringSignIn in ("high", "medium")
| where ResultType == "0"
| extend RiskDetail = tostring(RiskDetail)
| where RiskDetail has_any ("anonymizedIPAddress", "maliciousIPAddress")
| project TimeGenerated, UserPrincipalName, IPAddress, AppDisplayName, RiskLevelDuringSignIn, RiskDetail

Password Spray Detection

Sentinel
SigninLogs
| where TimeGenerated > ago(1h)
| where ResultType == "50126"  // Invalid password
| summarize 
    TargetedUsers = dcount(UserPrincipalName),
    UserList = make_set(UserPrincipalName, 20),
    TotalAttempts = count()
    by IPAddress
| where TargetedUsers > 10  // Many users from same IP
| extend AttemptsPerUser = TotalAttempts / TargetedUsers
| where AttemptsPerUser < 3  // Few attempts per user (spray pattern)

MFA Fatigue Attack Detection

Sentinel
SigninLogs
| where TimeGenerated > ago(1h)
| where ResultType in ("50074", "50076")  // MFA required
| summarize MfaPrompts = count() by UserPrincipalName, bin(TimeGenerated, 10m)
| where MfaPrompts > 5
| join kind=inner (
    SigninLogs
    | where TimeGenerated > ago(1h)
    | where ResultType == "0"
) on UserPrincipalName
| project TimeGenerated, UserPrincipalName, MfaPrompts, IPAddress

Endpoint Security

Suspicious PowerShell Execution

Sentinel
DeviceProcessEvents
| where TimeGenerated > ago(24h)
| where FileName in~ ("powershell.exe", "pwsh.exe")
| where ProcessCommandLine has_any (
    "-enc", "-encoded", "-e ",
    "downloadstring", "downloadfile", "webclient",
    "invoke-expression", "iex ", "bypass",
    "-nop", "-w hidden", "frombase64"
)
| project TimeGenerated, DeviceName, AccountName, ProcessCommandLine, InitiatingProcessFileName

LSASS Access Detection

Sentinel
DeviceProcessEvents
| where TimeGenerated > ago(24h)
| where ProcessCommandLine has_any ("sekurlsa", "mimikatz", "lsass")
    or (FileName =~ "procdump.exe" and ProcessCommandLine has "lsass")
    or (FileName =~ "rundll32.exe" and ProcessCommandLine has "comsvcs" and ProcessCommandLine has "MiniDump")
| project TimeGenerated, DeviceName, AccountName, FileName, ProcessCommandLine, InitiatingProcessFileName

Persistence via Registry Run Keys

Sentinel
DeviceRegistryEvents
| where TimeGenerated > ago(24h)
| where ActionType == "RegistryValueSet"
| where RegistryKey has_any ("\\CurrentVersion\\Run", "\\CurrentVersion\\RunOnce")
| project TimeGenerated, DeviceName, RegistryKey, RegistryValueName, RegistryValueData, InitiatingProcessFileName, InitiatingProcessAccountName

Scheduled Task Creation

Sentinel
DeviceProcessEvents
| where TimeGenerated > ago(24h)
| where FileName =~ "schtasks.exe"
| where ProcessCommandLine has "/create"
| extend TaskName = extract(@"/tn\s+['\"]?([^'\"\/]+)", 1, ProcessCommandLine)
| extend TaskAction = extract(@"/tr\s+['\"]?([^'\"]+)", 1, ProcessCommandLine)
| project TimeGenerated, DeviceName, AccountName, TaskName, TaskAction, ProcessCommandLine

Lateral Movement via Remote Services

Sentinel
DeviceNetworkEvents
| where TimeGenerated > ago(24h)
| where RemotePort in (135, 445, 3389, 5985, 5986)  // RPC, SMB, RDP, WinRM
| where ActionType == "ConnectionSuccess"
| where LocalIP != RemoteIP
| summarize 
    ConnectionCount = count(),
    TargetPorts = make_set(RemotePort),
    TargetIPs = make_set(RemoteIP)
    by DeviceName, InitiatingProcessFileName
| where ConnectionCount > 10

Email Security

Emails with Suspicious Attachments

Sentinel
EmailAttachmentInfo
| where TimeGenerated > ago(24h)
| where FileName has_any (".exe", ".js", ".vbs", ".ps1", ".bat", ".cmd", ".hta", ".iso", ".img")
| join kind=inner (
    EmailEvents
    | where TimeGenerated > ago(24h)
) on NetworkMessageId
| project TimeGenerated, SenderFromAddress, RecipientEmailAddress, Subject, FileName, FileType, SHA256

URL Clicks Leading to External Sites

Sentinel
UrlClickEvents
| where TimeGenerated > ago(24h)
| where IsClickedThrough == true
| extend UrlDomain = tostring(parse_url(Url).Host)
| where UrlDomain !endswith "microsoft.com" and UrlDomain !endswith "office.com"
| summarize ClickCount = count(), Users = make_set(AccountUpn) by Url, UrlDomain
| order by ClickCount desc

Inbox Rule Creation (Persistence)

Sentinel
OfficeActivity
| where TimeGenerated > ago(7d)
| where Operation in ("New-InboxRule", "Set-InboxRule")
| extend RuleParams = tostring(Parameters)
| where RuleParams has_any ("DeleteMessage", "ForwardTo", "RedirectTo", "MoveToFolder")
| project TimeGenerated, UserId, Operation, RuleParams, ClientIP

Cloud & Azure AD

OAuth App Consent (Consent Phishing)

Sentinel
AuditLogs
| where TimeGenerated > ago(7d)
| where OperationName == "Consent to application"
| extend AppName = tostring(TargetResources[0].displayName)
| extend User = tostring(InitiatedBy.user.userPrincipalName)
| extend Permissions = tostring(TargetResources[0].modifiedProperties)
| project TimeGenerated, User, AppName, Permissions

Privileged Role Assignment

Sentinel
AuditLogs
| where TimeGenerated > ago(7d)
| where OperationName has "Add member to role"
| extend TargetUser = tostring(TargetResources[0].userPrincipalName)
| extend RoleName = tostring(TargetResources[0].modifiedProperties[1].newValue)
| extend InitiatedBy = tostring(InitiatedBy.user.userPrincipalName)
| where RoleName has_any ("Global Administrator", "Privileged", "Security")
| project TimeGenerated, InitiatedBy, TargetUser, RoleName

Guest User Added

Sentinel
AuditLogs
| where TimeGenerated > ago(7d)
| where OperationName == "Invite external user"
| extend InvitedUser = tostring(TargetResources[0].userPrincipalName)
| extend InvitedBy = tostring(InitiatedBy.user.userPrincipalName)
| project TimeGenerated, InvitedBy, InvitedUser

Threat Hunting

Beaconing Detection (C2)

Sentinel
DeviceNetworkEvents
| where TimeGenerated > ago(24h)
| where ActionType == "ConnectionSuccess"
| where RemoteIPType == "Public"
| summarize 
    ConnectionCount = count(),
    Timestamps = make_list(TimeGenerated)
    by DeviceName, RemoteIP, InitiatingProcessFileName
| where ConnectionCount > 20
| mv-apply Timestamps to typeof(datetime) on (
    order by Timestamps asc
    | extend NextTime = next(Timestamps)
    | extend Interval = datetime_diff('second', NextTime, Timestamps)
    | summarize AvgInterval = avg(Interval), StdDev = stdev(Interval)
)
| where StdDev < 60  // Low variance = regular beaconing
| where AvgInterval between (30 .. 3600)

Data Staging Detection

Sentinel
DeviceFileEvents
| where TimeGenerated > ago(24h)
| where ActionType == "FileCreated"
| where FileName endswith ".zip" or FileName endswith ".rar" or FileName endswith ".7z"
| where FolderPath has_any ("\\Temp\\", "\\AppData\\", "\\ProgramData\\")
| extend FileSizeMB = FileSize / 1024 / 1024
| where FileSizeMB > 50
| project TimeGenerated, DeviceName, AccountName, FileName, FolderPath, FileSizeMB

First-Time User on System

Sentinel
let HistoricalUsers = 
    DeviceLogonEvents
    | where TimeGenerated between (ago(30d) .. ago(1d))
    | summarize KnownUsers = make_set(AccountName) by DeviceName;
DeviceLogonEvents
| where TimeGenerated > ago(1d)
| where LogonType in ("Interactive", "RemoteInteractive")
| join kind=leftanti (HistoricalUsers) on DeviceName, $left.AccountName == $right.KnownUsers
| project TimeGenerated, DeviceName, AccountName, LogonType, RemoteIP
