INV-010

Insider Threat Investigation

Comprehensive investigation playbook for insider threat incidents. Covers data theft, sabotage, policy violations, and departing employee scenarios with HR and Legal coordination requirements.

MITRE ATT&CK Mapping

Insider threats map to multiple MITRE techniques: T1005 (Data from Local System), T1039 (Data from Network Shared Drive), T1567 (Exfiltration Over Web Service), T1048 (Exfiltration Over Alternative Protocol), T1052 (Exfiltration Over Physical Medium), T1485 (Data Destruction - sabotage), T1489 (Service Stop - sabotage).

SeverityHigh
Time4-8 hours
TeamSOC + HR + Legal
Legal and HR Coordination Required

Insider threat investigations have significant legal and employment law implications. ALWAYS coordinate with HR and Legal before taking any action that could affect employment. Maintain strict confidentiality, follow evidence chain of custody, and document all activities meticulously.

Step 2: Establish Behavioral Baseline (30 min)

2.1 Normal Activity Patterns

// KQL - Historical working hours pattern
let targetUser = "user@company.com";
SigninLogs
| where TimeGenerated between (ago(90d) .. ago(7d)) // Historical baseline
| where UserPrincipalName =~ targetUser
| where ResultType == 0
| extend Hour = hourofday(TimeGenerated)
| extend DayOfWeek = dayofweek(TimeGenerated)
| summarize LoginCount = count() by Hour, DayOfWeek
| order by DayOfWeek asc, Hour asc

// KQL - Typical applications accessed
SigninLogs
| where TimeGenerated between (ago(90d) .. ago(7d))
| where UserPrincipalName =~ targetUser
| where ResultType == 0
| summarize 
    AccessCount = count(),
    LastAccess = max(TimeGenerated)
    by AppDisplayName
| order by AccessCount desc

// KQL - Normal file access volume
DeviceFileEvents
| where TimeGenerated between (ago(90d) .. ago(7d))
| where InitiatingProcessAccountName =~ "username"
| where ActionType in ("FileRead", "FileCopied")
| summarize DailyFiles = count() by bin(TimeGenerated, 1d)
| summarize 
    AvgDailyFiles = round(avg(DailyFiles), 0),
    MaxDailyFiles = max(DailyFiles),
    StdDevFiles = round(stdev(DailyFiles), 0)

// KQL - Historical data transfer volume
DeviceNetworkEvents
| where TimeGenerated between (ago(90d) .. ago(7d))
| where InitiatingProcessAccountName =~ "username"
| where RemoteIPType == "Public"
| summarize 
    DailySent = sum(SentBytes),
    DailyReceived = sum(ReceivedBytes)
    by bin(TimeGenerated, 1d)
| summarize 
    AvgDailySentMB = round(avg(DailySent) / 1048576.0, 2),
    MaxDailySentMB = round(max(DailySent) / 1048576.0, 2)

Step 3: Anomaly Detection (45 min)

3.1 Access Anomalies

// KQL - Off-hours activity (compared to baseline)
let targetUser = "user@company.com";
SigninLogs
| where TimeGenerated > ago(30d)
| where UserPrincipalName =~ targetUser
| where ResultType == 0
| extend Hour = hourofday(TimeGenerated)
| extend DayOfWeek = dayofweek(TimeGenerated)
| where Hour < 6 or Hour > 22 or DayOfWeek in (0d, 6d)
| project TimeGenerated, Hour, DayOfWeek, AppDisplayName, 
    IPAddress, DeviceDetail

// KQL - First-time system access
let targetUser = "user@company.com";
let recentApps = SigninLogs
| where TimeGenerated > ago(7d)
| where UserPrincipalName =~ targetUser
| where ResultType == 0
| distinct AppDisplayName;
let historicalApps = SigninLogs
| where TimeGenerated between (ago(90d) .. ago(7d))
| where UserPrincipalName =~ targetUser
| where ResultType == 0
| distinct AppDisplayName;
recentApps
| join kind=leftanti historicalApps on AppDisplayName
// Shows apps accessed recently that were never accessed before

// KQL - Spike in file access volume
let targetUser = "username";
let baseline = DeviceFileEvents
| where TimeGenerated between (ago(90d) .. ago(7d))
| where InitiatingProcessAccountName =~ targetUser
| summarize DailyFiles = count() by bin(TimeGenerated, 1d)
| summarize AvgDaily = avg(DailyFiles), StdDev = stdev(DailyFiles);
DeviceFileEvents
| where TimeGenerated > ago(7d)
| where InitiatingProcessAccountName =~ targetUser
| summarize RecentDaily = count() by bin(TimeGenerated, 1d)
| extend AvgBaseline = toscalar(baseline | project AvgDaily)
| extend StdDevBaseline = toscalar(baseline | project StdDev)
| extend ZScore = (RecentDaily - AvgBaseline) / StdDevBaseline
| where ZScore > 3 // More than 3 standard deviations above normal

// KQL - Access to systems outside normal scope
let targetUser = "user@company.com";
// Get user's normal systems from historical data
let normalSystems = SigninLogs
| where TimeGenerated between (ago(90d) .. ago(14d))
| where UserPrincipalName =~ targetUser
| where ResultType == 0
| distinct AppDisplayName, ResourceDisplayName;
// Find recent access to new systems
SigninLogs
| where TimeGenerated > ago(14d)
| where UserPrincipalName =~ targetUser
| where ResultType == 0
| join kind=leftanti normalSystems on AppDisplayName
| project TimeGenerated, AppDisplayName, IPAddress

3.2 Data Access Anomalies

// KQL - Access to sensitive directories
let targetUser = "username";
let sensitivePaths = dynamic([
    "HR", "Finance", "Legal", "Executive", "Confidential",
    "Restricted", "M&A", "Contracts", "Salary", "Payroll",
    "Customer", "Trade Secret", "Source Code"
]);
DeviceFileEvents
| where TimeGenerated > ago(30d)
| where InitiatingProcessAccountName =~ targetUser
| where FolderPath has_any (sensitivePaths)
| summarize 
    AccessCount = count(),
    FilesAccessed = dcount(FileName),
    TotalSize = sum(FileSize)
    by FolderPath
| order by AccessCount desc

// KQL - Mass file downloads/copies
let targetUser = "username";
DeviceFileEvents
| where TimeGenerated > ago(14d)
| where InitiatingProcessAccountName =~ targetUser
| where ActionType in ("FileRead", "FileCopied")
| summarize 
    FileCount = count(),
    TotalSize = sum(FileSize),
    UniqueFiles = dcount(FileName)
    by bin(TimeGenerated, 1h), DeviceName
| where FileCount > 100 or TotalSize > 104857600 // 100 files or 100 MB
| extend SizeMB = round(TotalSize / 1048576.0, 2)
| order by TimeGenerated desc

// KQL - Database query patterns
DeviceProcessEvents
| where TimeGenerated > ago(30d)
| where AccountName =~ "targetuser"
| where FileName in~ ("sqlcmd.exe", "sqlplus.exe", "mysql.exe",
    "psql.exe", "mongo.exe", "bcp.exe")
| project TimeGenerated, DeviceName, FileName, ProcessCommandLine

// KQL - Search queries indicating data hunting
DeviceEvents
| where TimeGenerated > ago(30d)
| where AccountName =~ "targetuser"
| where ActionType == "SearchQuery"
| where AdditionalFields has_any ("password", "confidential",
    "salary", "customer list", "source code", "trade secret")
| project TimeGenerated, SearchQuery = AdditionalFields

Step 4: Exfiltration Analysis (30 min)

// KQL - Emails to personal accounts
let targetUser = "target.user@company.com";
let personalDomains = dynamic(["gmail.com", "yahoo.com", 
    "hotmail.com", "outlook.com", "icloud.com", "protonmail.com"]);
EmailEvents
| where TimeGenerated > ago(90d)
| where SenderFromAddress =~ targetUser
| extend RecipientDomain = tostring(split(RecipientEmailAddress, "@")[1])
| where RecipientDomain in (personalDomains)
| summarize 
    EmailCount = count(),
    WithAttachments = countif(AttachmentCount > 0),
    Recipients = make_set(RecipientEmailAddress, 5)
| extend RiskIndicator = iff(WithAttachments > 10, "HIGH", 
    iff(WithAttachments > 3, "MEDIUM", "LOW"))

// KQL - Cloud storage uploads
DeviceNetworkEvents
| where TimeGenerated > ago(30d)
| where InitiatingProcessAccountName =~ "targetuser"
| where RemoteUrl has_any ("dropbox.com", "drive.google.com",
    "onedrive.live.com", "box.com", "mega.nz", "wetransfer.com")
| where SentBytes > 1048576 // 1 MB
| summarize 
    TotalSentMB = round(sum(SentBytes) / 1048576.0, 2),
    Connections = count()
    by RemoteUrl, bin(TimeGenerated, 1d)
| order by TotalSentMB desc

// KQL - USB device usage
DeviceEvents
| where TimeGenerated > ago(30d)
| where AccountName =~ "targetuser"
| where ActionType == "PnpDeviceConnected"
| where DeviceDescription has_any ("USB", "Removable", "Mass Storage")
| project TimeGenerated, DeviceName, DeviceDescription, DeviceId

// KQL - Printing activity (often overlooked)
DeviceEvents
| where TimeGenerated > ago(30d)
| where AccountName =~ "targetuser"
| where ActionType == "PrintJobStart"
| summarize PrintJobs = count() by bin(TimeGenerated, 1d)
| where PrintJobs > 20 // Unusual print volume

// KQL - Archive file creation
DeviceProcessEvents
| where TimeGenerated > ago(30d)
| where AccountName =~ "targetuser"
| where FileName in~ ("7z.exe", "winrar.exe", "zip.exe", "tar.exe")
| project TimeGenerated, DeviceName, ProcessCommandLine

// KQL - Screenshots and screen recording
DeviceProcessEvents
| where TimeGenerated > ago(30d)
| where AccountName =~ "targetuser"
| where FileName in~ ("snippingtool.exe", "screenshotx.exe", 
    "obs64.exe", "camtasia.exe", "sharex.exe")
| project TimeGenerated, DeviceName, FileName, ProcessCommandLine

Step 5: Communication Analysis (20 min)

// KQL - Emails to competitors (if known)
let targetUser = "target.user@company.com";
let competitorDomains = dynamic(["competitor1.com", "competitor2.com",
    "competitor3.com"]);
EmailEvents
| where TimeGenerated > ago(180d)
| where SenderFromAddress =~ targetUser
| extend RecipientDomain = tostring(split(RecipientEmailAddress, "@")[1])
| where RecipientDomain in (competitorDomains)
| project TimeGenerated, RecipientEmailAddress, Subject, AttachmentCount

// KQL - External email patterns
EmailEvents
| where TimeGenerated > ago(90d)
| where SenderFromAddress =~ "target.user@company.com"
| where RecipientEmailAddress !endswith "@company.com"
| extend RecipientDomain = tostring(split(RecipientEmailAddress, "@")[1])
| summarize EmailCount = count() by RecipientDomain
| order by EmailCount desc
// Review for unusual or unexpected external contacts

// KQL - Teams/Slack messages to external
CloudAppEvents
| where TimeGenerated > ago(30d)
| where AccountDisplayName =~ "Target User"
| where Application in ("Microsoft Teams", "Slack")
| where ActionType has "Message"
| where RawEventData has "external" or RawEventData has "guest"
| project TimeGenerated, ActionType, RawEventData

Containment Options

Do NOT take containment actions without explicit HR and Legal approval. Premature action can compromise legal proceedings, violate employment law, or alert the subject.

Escalation Criteria

Escalate Immediately If:
  • Confirmed data exfiltration in progress
  • Access to trade secrets or critical IP confirmed
  • Evidence of collusion with competitors
  • Sabotage of systems or data detected
  • Multiple insiders involved
  • Evidence of fraud or financial crimes
  • Classified or regulated data involved
  • Subject appears aware of investigation

Questions to Ask (Not the User - The Data)

  • Timeline: When did the suspicious behavior start? What changed?
  • Pattern: Is this a one-time event or ongoing pattern?
  • Volume: How much data is involved? Is it unusual?
  • Destination: Where is the data going? Can we trace it?
  • Context: Is there a legitimate business reason this could be innocent?
  • History: Has this user had previous DLP/policy alerts?
  • Employment status: Resigned? On PIP? Recent discipline? Denied promotion?
