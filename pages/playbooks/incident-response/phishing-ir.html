Phishing Incident Response

Complete response playbook for phishing campaigns, from detection through remediation and prevention.

Phase 1: Detection & Scoping

Identify Campaign Scope

// Find all recipients of phishing email
EmailEvents
| where TimeGenerated > ago(7d)
| where SenderFromAddress == "phisher@malicious-domain.com"
    or Subject == "Urgent: Verify Your Account"
    or NetworkMessageId == "<specific-message-id>"
| summarize Recipients=make_set(RecipientEmailAddress), 
    Count=count() by SenderFromAddress, Subject
| extend RecipientCount = array_length(Recipients)

// Track who clicked
UrlClickEvents
| where TimeGenerated > ago(7d)
| where Url contains "malicious-domain.com"
| project TimeGenerated, AccountUpn, Url, ActionType, IsClickedThrough, NetworkMessageId

// Find similar campaigns (sender infrastructure)
EmailEvents
| where TimeGenerated > ago(30d)
| where SenderFromDomain == "malicious-domain.com"
    or SenderMailFromDomain == "malicious-domain.com"
    or SenderIPv4 == "185.123.45.67"
| summarize Campaigns=dcount(Subject), 
    Recipients=dcount(RecipientEmailAddress),
    Subjects=make_set(Subject)
    by SenderFromAddress

Email Purge Process

// Create compliance search
New-ComplianceSearch -Name "Phishing-IR-2025-001" `
    -ExchangeLocation All `
    -ContentMatchQuery '(From:phisher@malicious.com) AND (Subject:"Urgent: Verify Your Account")'

// Start search
Start-ComplianceSearch -Identity "Phishing-IR-2025-001"

// Check results
Get-ComplianceSearch -Identity "Phishing-IR-2025-001" | FL Status, Items

// Soft delete (recoverable)
New-ComplianceSearchAction -SearchName "Phishing-IR-2025-001" -Purge -PurgeType SoftDelete

// Monitor purge progress
Get-ComplianceSearchAction -Identity "Phishing-IR-2025-001_Purge" | FL Status, Results

Phase 3: Compromised Account Response

// Check for malicious inbox rules
Get-InboxRule -Mailbox victim@company.com | Where-Object {
    $_.ForwardTo -or $_.ForwardAsAttachmentTo -or $_.RedirectTo -or 
    $_.DeleteMessage -or $_.MoveToFolder -eq "RSS Feeds"
} | Format-List Name, ForwardTo, DeleteMessage, MoveToFolder

// Remove malicious rules
Get-InboxRule -Mailbox victim@company.com | Where-Object {
    $_.ForwardTo -like "*@external*"
} | Remove-InboxRule -Confirm:$false

// Check mailbox forwarding
Get-Mailbox -Identity victim@company.com | FL ForwardingSMTPAddress, DeliverToMailboxAndForward

// Remove forwarding
Set-Mailbox -Identity victim@company.com -ForwardingSMTPAddress $null

// Check for OAuth app consents
Get-MgUserOauth2PermissionGrant -UserId "victim@company.com" | 
    Select-Object ClientId, Scope, ConsentType

// Review sign-in activity post-compromise
SigninLogs
| where TimeGenerated > ago(24h)
| where UserPrincipalName == "victim@company.com"
| project TimeGenerated, IPAddress, Location, AppDisplayName, 
    ResultType, ResultDescription, DeviceDetail

Phase 4: Investigation

Post-Compromise Activity Check

// Check for sent emails (attacker using mailbox)
EmailEvents
| where TimeGenerated > ago(7d)
| where SenderFromAddress == "victim@company.com"
| where RecipientEmailAddress !endswith "@company.com"
| project TimeGenerated, RecipientEmailAddress, Subject

// Check for file access
OfficeActivity
| where TimeGenerated > ago(7d)
| where UserId == "victim@company.com"
| where Operation in ("FileDownloaded", "FileAccessed", "FileSyncDownloadedFull")
| summarize Count=count(), Files=make_set(SourceFileName) by Operation

// Check for admin actions
AuditLogs
| where TimeGenerated > ago(7d)
| where InitiatedBy.user.userPrincipalName == "victim@company.com"
| where Category == "RoleManagement" or Category == "UserManagement"
| project TimeGenerated, OperationName, TargetResources

Phase 6: Lessons Learned

  • Why did email bypass filters? (new domain, look-alike, compromised sender)
  • How many users clicked? What training needed?
  • Can detection be improved? (Sigma rule, custom detection)
  • Was response time adequate?
  • Should similar domains be preemptively blocked?
