C2 Communication Triage

Investigation Queries

// Check connection patterns from source host
CommonSecurityLog
| where TimeGenerated > ago(24h)
| where SourceIP == "10.1.2.50"
| where DestinationIP !startswith "10." and DestinationIP !startswith "192.168."
| summarize ConnectionCount=count(), 
    BytesSent=sum(SentBytes), 
    BytesReceived=sum(ReceivedBytes),
    Ports=make_set(DestinationPort) 
    by DestinationIP, bin(TimeGenerated, 1h)
| order by TimeGenerated desc

// Beaconing detection (regular intervals)
CommonSecurityLog
| where TimeGenerated > ago(24h)
| where SourceIP == "10.1.2.50"
| summarize Times=make_list(TimeGenerated) by SourceIP, DestinationIP
| mv-apply Times to typeof(datetime) on (
    extend NextTime = next(Times)
    | extend Interval = datetime_diff('second', NextTime, Times)
    | summarize AvgInterval=avg(Interval), StdDev=stdev(Interval), Count=count()
)
| where StdDev < 10 and Count > 20

// DNS query analysis
DnsEvents
| where TimeGenerated > ago(24h)
| where ClientIP == "10.1.2.50"
| summarize QueryCount=count(), 
    UniqueQueries=dcount(Name),
    Queries=make_set(Name, 100) 
    by bin(TimeGenerated, 1h)
| where QueryCount > 100

Exploitation Attempt Triage

// Check if target is vulnerable
// Cross-reference with vulnerability scanner data
SecurityAlert
| where TimeGenerated > ago(24h)
| where AlertName contains "CVE"
| extend CVE = extract("CVE-[0-9]+-[0-9]+", 0, AlertName)
| join kind=leftouter (
    // Vulnerability scan results
    SecurityBaseline_CL
    | where CVE_s != ""
) on $left.CVE == $right.CVE_s

// Check for post-exploitation activity
DeviceProcessEvents
| where TimeGenerated > ago(1h)
| where DeviceName == "TARGET-SERVER"
| where InitiatingProcessFileName in~ ("w3wp.exe", "httpd.exe", "nginx.exe")
| where FileName in~ ("cmd.exe", "powershell.exe", "whoami.exe", "net.exe")

Lateral Movement Detection

// SMB connections from non-file-server
CommonSecurityLog
| where TimeGenerated > ago(24h)
| where DestinationPort == 445
| where SourceIP !in ("10.1.1.10", "10.1.1.11") // Exclude file servers
| summarize Targets=make_set(DestinationIP), Count=count() by SourceIP
| where Count > 10

// RDP from workstation to workstation
CommonSecurityLog
| where TimeGenerated > ago(24h)
| where DestinationPort == 3389
| where SourceIP startswith "10.2." // Workstation subnet
| where DestinationIP startswith "10.2." // Also workstation subnet
| project TimeGenerated, SourceIP, DestinationIP, Action

// WMI/WinRM lateral movement
CommonSecurityLog
| where TimeGenerated > ago(24h)
| where DestinationPort in (135, 5985, 5986)
| where SourceIP !in ("10.1.1.5") // Exclude management server
| summarize Targets=dcount(DestinationIP) by SourceIP
| where Targets > 5

Common False Positives

  • CDN/Cloud IPs: Akamai, CloudFlare, AWS ranges flagged
  • Security scanners: Authorized vulnerability scans trigger IDS
  • Backup traffic: Large transfers to backup services
  • Software updates: Download traffic to legitimate sources
  • VPN/Proxy: Traffic to corporate VPN endpoints
  • Load balancer health checks: Regular connection patterns
