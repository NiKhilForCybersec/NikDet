Agent-Based Collection

Windows: Splunk Universal Forwarder

# Install Splunk UF silently
msiexec /i splunkforwarder-9.0.0-x64.msi AGREETOLICENSE=yes ^
  RECEIVING_INDEXER="splunk-hf.company.com:9997" ^
  SPLUNKPASSWORD=changeme /quiet

# Configure inputs (inputs.conf)
[WinEventLog://Security]
disabled = 0
index = windows
sourcetype = WinEventLog:Security

[WinEventLog://System]
disabled = 0
index = windows
sourcetype = WinEventLog:System

[WinEventLog://Microsoft-Windows-Sysmon/Operational]
disabled = 0
index = windows
sourcetype = XmlWinEventLog:Microsoft-Windows-Sysmon/Operational
renderXml = true

Windows: Microsoft Azure Monitor Agent (AMA)

# Install AMA via PowerShell
$ResourceGroup = "rg-security"

Linux: Elastic Agent

# Download and install
curl -L -O https://artifacts.elastic.co/downloads/beats/elastic-agent/elastic-agent-8.11.0-linux-x86_64.tar.gz
tar xzvf elastic-agent-8.11.0-linux-x86_64.tar.gz
cd elastic-agent-8.11.0-linux-x86_64

# Enroll with Fleet server
sudo ./elastic-agent install --url=https://fleet.company.com:8220 \
  --enrollment-token=YOUR_ENROLLMENT_TOKEN

# Verify status
sudo elastic-agent status

Syslog Collection

Syslog Server Configuration (rsyslog)

# /etc/rsyslog.conf - Enable TCP syslog listener

# Load modules
module(load="imtcp")
input(type="imtcp" port="514")

# For TLS (recommended for security)
module(load="imtcp" StreamDriver.Name="gtls" 
       StreamDriver.Mode="1" 
       StreamDriver.AuthMode="x509/name")

# Write logs to files by source
template(name="RemoteLogs" type="string"
         string="/var/log/remote/%HOSTNAME%/%PROGRAMNAME%.log")

if $fromhost-ip != '127.0.0.1' then {
    action(type="omfile" dynaFile="RemoteLogs")
}

# Forward to SIEM
*.* action(type="omfwd" target="siem.company.com" port="514" protocol="tcp")

Network Device Syslog Configuration

# Cisco ASA
logging enable
logging host inside 10.0.0.50 UDP/514
logging trap informational
logging facility 20

# Palo Alto
set deviceconfig system log-settings syslog syslog-server siem-server server 10.0.0.50
set deviceconfig system log-settings syslog syslog-server siem-server transport UDP
set deviceconfig system log-settings syslog syslog-server siem-server port 514
set deviceconfig system log-settings syslog syslog-server siem-server format BSD

# Fortinet FortiGate
config log syslogd setting
  set status enable
  set server 10.0.0.50
  set port 514
end

Cloud Log Collection

AWS CloudTrail to SIEM

# Option 1: S3 bucket polling (Splunk Add-on for AWS)
# Configure in Splunk: Settings → Data inputs → AWS S3

# Option 2: CloudWatch Logs subscription filter to Kinesis/Lambda
# Then forward to SIEM via HTTP Event Collector

# Option 3: EventBridge to SIEM webhook
aws events put-rule --name cloudtrail-to-siem \
  --event-pattern '{"source": ["aws.cloudtrail"]}'

aws events put-targets --rule cloudtrail-to-siem \
  --targets "Id"="1","Arn"="arn:aws:events:us-east-1:ACCOUNT:api-destination/siem"

Azure Activity Logs to Sentinel

# Enable Azure Activity Logs connector in Sentinel
# Data Connectors → Azure Activity → Connect

# Or via Azure CLI:
az monitor diagnostic-settings create \
  --name "send-to-sentinel" \
  --resource "/subscriptions/SUBSCRIPTION_ID" \
  --workspace "LOG_ANALYTICS_WORKSPACE_ID" \
  --logs '[{"category": "Administrative", "enabled": true}]'

Collection Best Practices

PracticeWhy It Matters
Use TLS for syslogPrevents eavesdropping and tampering
Implement bufferingPrevents log loss during SIEM outages
Monitor collection healthDetect gaps in log coverage immediately
Normalize timestampsEnsure all logs use UTC for correlation
Document source configsEasier troubleshooting and onboarding
Test failoverVerify logs still flow when primary collector fails

Troubleshooting Collection Issues

SymptomPossible CauseSolution
No logs arrivingFirewall blocking trafficCheck firewall rules for syslog/agent ports
Intermittent gapsNetwork congestion, UDP dropsSwitch to TCP/TLS, add buffering
Delayed logsCollection queue backlogScale collectors, increase throughput
Parsing errorsUnexpected log formatUpdate sourcetype/parser configuration
Agent not runningService crashed or disabledCheck service status, enable autostart
