Cloud Detection Engineering

Build effective detections for cloud-native threats and identity-based attacks.

Cloud Detection Principles

Identity First

Identity is the perimeter. Focus on authentication and authorization anomalies.

API-Centric

Everything is an API call. Monitor control plane activity closely.

Configuration

Misconfigurations are the top cloud risk. Detect changes to security settings.

Behavioral

Baseline normal behavior. Alert on deviations from established patterns.

Identity-Based Detections

1. Password Spray Detection

// Many failed logins across different accounts from same IP
SigninLogs
| where ResultType == "50126"  // Invalid credentials
| summarize 
    FailedAttempts = count(),
    UniqueUsers = dcount(UserPrincipalName),
    Users = make_set(UserPrincipalName, 10)
    by IPAddress, bin(TimeGenerated, 1h)
| where UniqueUsers > 10 and FailedAttempts > 20
| project TimeGenerated, IPAddress, UniqueUsers, FailedAttempts, Users

2. MFA Fatigue Detection

// Multiple MFA prompts followed by success
let MFAPrompts = SigninLogs
| where ResultType == "50074"  // MFA required
| summarize MFACount = count() by UserPrincipalName, bin(TimeGenerated, 15m);

let Successes = SigninLogs
| where ResultType == 0
| project UserPrincipalName, SuccessTime = TimeGenerated;

MFAPrompts
| join kind=inner Successes on UserPrincipalName
| where SuccessTime between (TimeGenerated .. (TimeGenerated + 15m))
| where MFACount > 5
| project UserPrincipalName, MFACount, TimeGenerated, SuccessTime

3. Token Theft / Session Hijacking

// Same token used from different devices/locations
SigninLogs
| where ResultType == 0
| extend DeviceId = tostring(DeviceDetail.deviceId)
| summarize 
    UniqueIPs = dcount(IPAddress),
    UniqueDevices = dcount(DeviceId),
    IPs = make_set(IPAddress),
    Devices = make_set(DeviceId)
    by UserPrincipalName, CorrelationId, bin(TimeGenerated, 1h)
| where UniqueIPs > 1 or UniqueDevices > 1
| project TimeGenerated, UserPrincipalName, UniqueIPs, UniqueDevices, IPs

4. Anomalous Application Consent

// High-risk OAuth permissions granted
AuditLogs
| where OperationName == "Consent to application"
| extend Permissions = tostring(TargetResources[0].modifiedProperties)
| extend AppName = tostring(TargetResources[0].displayName)
| extend User = tostring(InitiatedBy.user.userPrincipalName)
| where Permissions has_any (
    "Mail.Read", "Mail.ReadWrite", "Mail.Send",
    "Files.ReadWrite.All", "Directory.ReadWrite.All",
    "User.ReadWrite.All", "offline_access"
)
| project TimeGenerated, User, AppName, Permissions

Configuration Change Detections

Security Settings Modified

// Conditional Access Policy changes
AuditLogs
| where OperationName has_any (
    "Update conditional access policy",
    "Delete conditional access policy",
    "Add conditional access policy"
)
| extend Actor = tostring(InitiatedBy.user.userPrincipalName)
| extend PolicyName = tostring(TargetResources[0].displayName)
| project TimeGenerated, Actor, OperationName, PolicyName

// AWS: CloudTrail logging modified
AWSCloudTrail
| where EventName in ("StopLogging", "DeleteTrail", "UpdateTrail", 
    "PutEventSelectors")
| project TimeGenerated, UserIdentityArn, EventName, SourceIpAddress

Data Exfiltration Detections

Mass Download Detection

// SharePoint/OneDrive mass download
OfficeActivity
| where Operation in ("FileDownloaded", "FileSyncDownloadedFull")
| summarize 
    DownloadCount = count(),
    TotalSize = sum(OfficeObjectId),
    Files = make_set(SourceFileName, 20)
    by UserId, bin(TimeGenerated, 1h)
| where DownloadCount > 100
| project TimeGenerated, UserId, DownloadCount, Files

// AWS S3: Large data transfer
AWSCloudTrail
| where EventSource == "s3.amazonaws.com"
| where EventName == "GetObject"
| summarize 
    ObjectCount = count(),
    Buckets = make_set(tostring(parse_json(RequestParameters).bucketName))
    by UserIdentityArn, bin(TimeGenerated, 1h)
| where ObjectCount > 100

Sigma Rules for Cloud

title: Azure AD Global Administrator Role Assigned
id: 8e6c0c5d-8f6a-4f9a-9c6b-1a2b3c4d5e6f
status: production
description: Detects assignment of Global Administrator role
logsource:
    product: azure
    service: auditlogs
detection:
    selection:
        OperationName: 'Add member to role'
    filter_role:
        TargetResources[0].modifiedProperties[0].newValue|contains: 'Global Administrator'
    condition: selection and filter_role
level: high
tags:
    - attack.persistence
    - attack.t1098
