FortiGate VPN Configuration

Configure IPSec site-to-site VPN and SSL VPN for remote access.

IPSec Site-to-Site VPN

Phase 1 Configuration

config vpn ipsec phase1-interface
    edit "VPN-to-BranchOffice"
        set interface "wan1"
        set ike-version 2
        set peertype any
        set net-device disable
        set proposal aes256-sha256
        set dhgrp 14
        set remote-gw 198.51.100.10
        set psksecret "YourPreSharedKey123!"
        set comments "VPN to branch office"
    next
end

Phase 2 Configuration

config vpn ipsec phase2-interface
    edit "VPN-to-BranchOffice-P2"
        set phase1name "VPN-to-BranchOffice"
        set proposal aes256-sha256
        set pfs enable
        set dhgrp 14
        set auto-negotiate enable
        set src-subnet 10.0.0.0 255.255.0.0
        set dst-subnet 10.1.0.0 255.255.0.0
    next
end

Firewall Policies for VPN

config firewall policy
    edit 100
        set name "HQ-to-Branch"
        set srcintf "internal"
        set dstintf "VPN-to-BranchOffice"
        set srcaddr "all"
        set dstaddr "all"
        set action accept
        set schedule "always"
        set service "ALL"
    next
    edit 101
        set name "Branch-to-HQ"
        set srcintf "VPN-to-BranchOffice"
        set dstintf "internal"
        set srcaddr "all"
        set dstaddr "all"
        set action accept
        set schedule "always"
        set service "ALL"
    next
end

SSL VPN (Remote Access)

SSL VPN Settings

config vpn ssl settings
    set servercert "Fortinet_Factory"
    set tunnel-ip-pools "SSLVPN-Pool"
    set dns-server1 10.0.1.10
    set port 443
    set source-interface "wan1"
    set source-address "all"
    set default-portal "full-access"
end

# IP Pool for VPN clients
config firewall address
    edit "SSLVPN-Pool"
        set type iprange
        set start-ip 10.212.134.1
        set end-ip 10.212.134.200
    next
end

SSL VPN Portal

config vpn ssl web portal
    edit "full-access"
        set tunnel-mode enable
        set ip-pools "SSLVPN-Pool"
        set split-tunneling disable
        set web-mode enable
    next
end

SSL VPN Firewall Policy

config firewall policy
    edit 200
        set name "SSLVPN-to-Internal"
        set srcintf "ssl.root"
        set dstintf "internal"
        set srcaddr "all"
        set dstaddr "all"
        set action accept
        set schedule "always"
        set service "ALL"
        set groups "VPN-Users"
    next
end

VPN Troubleshooting

# Show IPSec VPN status
get vpn ipsec tunnel summary
diagnose vpn ike gateway list
diagnose vpn tunnel list

# Show SSL VPN status
get vpn ssl monitor
diagnose vpn ssl list

# Debug IPSec
diagnose debug application ike -1
diagnose debug enable
