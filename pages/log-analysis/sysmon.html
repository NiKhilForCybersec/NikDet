Sysmon Detection Queries

// KQL - Suspicious process creation
Event
| where Source == "Microsoft-Windows-Sysmon/Operational"
| where EventID == 1
| extend ProcessData = parse_xml(EventData)
| extend Image = tostring(ProcessData.DataItem.Image)
| extend CommandLine = tostring(ProcessData.DataItem.CommandLine)
| extend ParentImage = tostring(ProcessData.DataItem.ParentImage)
| where Image has_any ("powershell", "cmd", "wscript", "mshta")
| where CommandLine has_any ("-enc", "bypass", "hidden", "downloadstring")

// KQL - LSASS access (credential theft)
Event
| where Source == "Microsoft-Windows-Sysmon/Operational"
| where EventID == 10
| extend ProcessData = parse_xml(EventData)
| extend TargetImage = tostring(ProcessData.DataItem.TargetImage)
| extend SourceImage = tostring(ProcessData.DataItem.SourceImage)
| where TargetImage endswith "lsass.exe"
| where SourceImage !has_any ("csrss.exe", "wininit.exe", "services.exe")

// KQL - Process injection (CreateRemoteThread)
Event
| where Source == "Microsoft-Windows-Sysmon/Operational"
| where EventID == 8
| extend ProcessData = parse_xml(EventData)
| extend SourceImage = tostring(ProcessData.DataItem.SourceImage)
| extend TargetImage = tostring(ProcessData.DataItem.TargetImage)
| where SourceImage != TargetImage

// KQL - DNS queries to suspicious TLDs
Event
| where Source == "Microsoft-Windows-Sysmon/Operational"
| where EventID == 22
| extend QueryName = tostring(parse_xml(EventData).DataItem.QueryName)
| where QueryName has_any (".tk", ".ml", ".ga", ".cf", ".top", ".xyz")

// KQL - Registry Run key persistence
Event
| where Source == "Microsoft-Windows-Sysmon/Operational"
| where EventID in (12, 13)
| extend TargetObject = tostring(parse_xml(EventData).DataItem.TargetObject)
| where TargetObject has_any ("Run", "RunOnce")

// KQL - WMI persistence
Event
| where Source == "Microsoft-Windows-Sysmon/Operational"
| where EventID in (19, 20, 21)
| extend EventType = case(EventID == 19, "Filter", EventID == 20, "Consumer", "Binding")

// SPL - Sysmon process creation
source="WinEventLog:Microsoft-Windows-Sysmon/Operational" EventCode=1
| table _time, Computer, User, Image, CommandLine, ParentImage

// SPL - Network connections
source="WinEventLog:Microsoft-Windows-Sysmon/Operational" EventCode=3
| where Initiated="true"
| stats count by Image, DestinationIp, DestinationPort
