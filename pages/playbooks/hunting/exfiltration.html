Exfiltration Hunt Playbook

Proactive threat hunting for data exfiltration techniques. Covers network-based, cloud storage, DNS tunneling, email, USB, and encrypted channel exfiltration methods.

Hunt 1: Unusual Outbound Data Volume

// KQL - Devices with abnormal upload volume
let lookback = 30d;
let threshold_multiplier = 3;
let baseline = DeviceNetworkEvents
| where TimeGenerated between (ago(lookback) .. ago(7d))
| where RemoteIPType == "Public"
| summarize AvgDailySent = avg(SentBytes) by DeviceName;
DeviceNetworkEvents
| where TimeGenerated > ago(7d)
| where RemoteIPType == "Public"
| summarize TodaySent = sum(SentBytes) by DeviceName, 
    bin(TimeGenerated, 1d)
| join kind=inner baseline on DeviceName
| where TodaySent > (AvgDailySent * threshold_multiplier)
| extend AnomalyRatio = round(TodaySent / AvgDailySent, 2)
| extend SentGB = round(TodaySent / 1073741824.0, 2)
| order by AnomalyRatio desc

// KQL - Upload/Download ratio anomalies
DeviceNetworkEvents
| where TimeGenerated > ago(24h)
| where RemoteIPType == "Public"
| summarize 
    Sent = sum(SentBytes),
    Received = sum(ReceivedBytes)
    by DeviceName, InitiatingProcessAccountName
| where Received > 0
| extend UploadRatio = round(todouble(Sent) / todouble(Received), 2)
| where UploadRatio > 5 // 5x more upload than download
| where Sent > 104857600 // At least 100 MB uploaded
| extend SentMB = round(Sent / 1048576.0, 2)
| order by UploadRatio desc

// KQL - Large transfers to single destination
DeviceNetworkEvents
| where TimeGenerated > ago(7d)
| where RemoteIPType == "Public"
| summarize 
    BytesSent = sum(SentBytes),
    Connections = count(),
    Devices = dcount(DeviceName)
    by RemoteIP
| where BytesSent > 524288000 // 500 MB
| extend SentGB = round(BytesSent / 1073741824.0, 2)
| order by BytesSent desc

// SPL - Data transfer anomalies
index=firewall action=allowed direction=outbound
| stats sum(bytes_out) as total_out, sum(bytes_in) as total_in by src_ip
| eval ratio = total_out / total_in
| where ratio > 5 AND total_out > 104857600
| sort -total_out

Hunt 2: DNS Tunneling Detection

// KQL - Long subdomain queries (data in DNS)
DnsEvents
| where TimeGenerated > ago(24h)
| extend Subdomain = tostring(split(Name, ".")[0])
| extend SubdomainLength = strlen(Subdomain)
| where SubdomainLength > 40
| summarize 
    QueryCount = count(),
    AvgLength = round(avg(SubdomainLength), 0),
    MaxLength = max(SubdomainLength),
    UniqueSubdomains = dcount(Name)
    by ClientIP, tostring(split(Name, ".")[-2])
| where QueryCount > 50 or AvgLength > 30
| order by QueryCount desc

// KQL - High volume DNS to single domain
DnsEvents
| where TimeGenerated > ago(24h)
| extend BaseDomain = strcat(tostring(split(Name, ".")[-2]), ".", 
    tostring(split(Name, ".")[-1]))
| summarize 
    Queries = count(),
    UniqueSubdomains = dcount(Name),
    SampleQueries = make_set(Name, 5)
    by ClientIP, BaseDomain
| where Queries > 500 or UniqueSubdomains > 100
| order by Queries desc

// KQL - TXT record abuse
DnsEvents
| where QueryType == "TXT"
| summarize Queries = count() by ClientIP, Name
| where Queries > 20
| order by Queries desc

// KQL - Entropy-based detection
DnsEvents
| extend Subdomain = tostring(split(Name, ".")[0])
| where strlen(Subdomain) > 20
| extend HasNumbers = Subdomain matches regex "[0-9]"
| extend HasMixedCase = Subdomain matches regex "[A-Z]" and 
    Subdomain matches regex "[a-z]"
| where HasNumbers
| summarize Count = count() by ClientIP, 
    BaseDomain = strcat(split(Name, ".")[-2], ".", split(Name, ".")[-1])
| where Count > 50

// SPL - DNS tunneling indicators
index=dns
| eval subdomain_len = len(mvindex(split(query, "."), 0))
| where subdomain_len > 40
| stats count, avg(subdomain_len) as avg_len by src_ip, query_domain
| where count > 50

Hunt 3: Cloud Storage Exfiltration

// KQL - Uploads to personal cloud storage
let personalCloudDomains = dynamic([
    "dropbox.com", "drive.google.com", "onedrive.live.com",
    "box.com", "mega.nz", "wetransfer.com", "sendspace.com",
    "mediafire.com", "file.io", "anonfiles.com", "gofile.io"
]);
DeviceNetworkEvents
| where TimeGenerated > ago(7d)
| where RemoteUrl has_any (personalCloudDomains)
| where SentBytes > 5242880 // 5 MB
| summarize 
    TotalSent = sum(SentBytes),
    Uploads = count(),
    FirstUpload = min(TimeGenerated),
    LastUpload = max(TimeGenerated)
    by DeviceName, InitiatingProcessAccountName, RemoteUrl
| extend SentMB = round(TotalSent / 1048576.0, 2)
| order by TotalSent desc

// KQL - Rclone usage detection
DeviceProcessEvents
| where TimeGenerated > ago(7d)
| where FileName =~ "rclone.exe"
| project TimeGenerated, DeviceName, ProcessCommandLine, AccountName
| extend RemoteType = extract("([a-zA-Z0-9]+):", 1, ProcessCommandLine)

// KQL - Cloud sync tool activity
DeviceProcessEvents
| where FileName in~ (
    "rclone.exe", "megasync.exe", "dropbox.exe",
    "googledrivesync.exe", "onedrive.exe", "boxsync.exe"
)
| where ProcessCommandLine has_any ("copy", "sync", "upload", "push")
| project TimeGenerated, DeviceName, FileName, ProcessCommandLine

// KQL - Git push to external repositories
DeviceProcessEvents
| where FileName =~ "git.exe"
| where ProcessCommandLine has "push"
| project TimeGenerated, DeviceName, ProcessCommandLine, AccountName

Hunt 4: Email-Based Exfiltration

// KQL - Large attachments to external recipients
EmailEvents
| where TimeGenerated > ago(7d)
| where EmailDirection == "Outbound"
| where RecipientEmailAddress !endswith "@company.com"
| join kind=inner (
    EmailAttachmentInfo
    | where FileSize > 10485760 // 10 MB
) on NetworkMessageId
| project TimeGenerated, SenderFromAddress, RecipientEmailAddress,
    Subject, FileName, FileSizeMB = round(FileSize / 1048576.0, 2)
| order by FileSizeMB desc

// KQL - Emails to personal accounts
let personalDomains = dynamic([
    "gmail.com", "yahoo.com", "hotmail.com", "outlook.com",
    "icloud.com", "protonmail.com", "aol.com"
]);
EmailEvents
| where TimeGenerated > ago(30d)
| where EmailDirection == "Outbound"
| extend RecipientDomain = tostring(split(RecipientEmailAddress, "@")[1])
| where RecipientDomain in (personalDomains)
| summarize 
    EmailCount = count(),
    WithAttachments = countif(AttachmentCount > 0)
    by SenderFromAddress, RecipientDomain
| where WithAttachments > 5
| order by WithAttachments desc

// KQL - Sensitive file types sent externally
let sensitiveExtensions = dynamic([
    ".sql", ".csv", ".xlsx", ".mdb", ".bak", ".zip", 
    ".7z", ".rar", ".pst", ".db", ".pfx", ".key"
]);
EmailAttachmentInfo
| where TimeGenerated > ago(7d)
| where FileName has_any (sensitiveExtensions)
| join kind=inner (
    EmailEvents
    | where EmailDirection == "Outbound"
    | where RecipientEmailAddress !endswith "@company.com"
) on NetworkMessageId
| project TimeGenerated, SenderFromAddress, RecipientEmailAddress,
    FileName, FileSize

// KQL - Auto-forwarding rules
CloudAppEvents
| where ActionType in ("New-InboxRule", "Set-Mailbox")
| where RawEventData has_any ("ForwardTo", "RedirectTo", 
    "ForwardingSmtpAddress")
| project TimeGenerated, AccountDisplayName, ActionType, RawEventData

Hunt 5: USB and Physical Media

// KQL - USB device connections
DeviceEvents
| where TimeGenerated > ago(7d)
| where ActionType == "PnpDeviceConnected"
| where DeviceDescription has_any ("USB", "Removable", "Mass Storage")
| summarize 
    Connections = count(),
    FirstSeen = min(TimeGenerated),
    LastSeen = max(TimeGenerated)
    by DeviceName, DeviceDescription, DeviceId
| order by Connections desc

// KQL - Files written to removable drives
DeviceFileEvents
| where TimeGenerated > ago(7d)
| where ActionType in ("FileCreated", "FileModified")
| where FolderPath matches regex @"^[D-Z]:\\"
| where FolderPath !has "\\Windows\\" and FolderPath !has "\\Program Files"
| summarize 
    FileCount = count(),
    TotalSize = sum(FileSize),
    FileTypes = make_set(extract(@"\.([^.]+)$", 1, FileName), 10)
    by DeviceName, InitiatingProcessAccountName
| where FileCount > 20 or TotalSize > 52428800
| extend TotalMB = round(TotalSize / 1048576.0, 2)
| order by TotalSize desc

// KQL - Sensitive files copied to USB
DeviceFileEvents
| where ActionType in ("FileCreated", "FileCopied")
| where FolderPath matches regex @"^[D-Z]:\\"
| where FileName has_any (".sql", ".csv", ".xlsx", ".mdb", 
    ".bak", ".pst", ".key", ".pfx")
| project TimeGenerated, DeviceName, FileName, FolderPath,
    InitiatingProcessAccountName

Hunt 6: Data Staging Detection

// KQL - Large archive file creation
DeviceFileEvents
| where TimeGenerated > ago(7d)
| where ActionType == "FileCreated"
| where FileName endswith_any (".zip", ".7z", ".rar", ".tar", 
    ".gz", ".tar.gz", ".tgz")
| where FileSize > 104857600 // 100 MB
| project TimeGenerated, DeviceName, FileName, FolderPath,
    FileSizeMB = round(FileSize / 1048576.0, 2),
    InitiatingProcessFileName, InitiatingProcessAccountName
| order by FileSize desc

// KQL - Archive tool usage
DeviceProcessEvents
| where TimeGenerated > ago(7d)
| where FileName in~ ("7z.exe", "winrar.exe", "zip.exe", "tar.exe")
| where ProcessCommandLine has_any ("a ", "-a", "add", "create")
| project TimeGenerated, DeviceName, ProcessCommandLine, AccountName

// KQL - Files copied to staging locations
DeviceFileEvents
| where TimeGenerated > ago(7d)
| where ActionType in ("FileCreated", "FileCopied")
| where FolderPath has_any ("\\Temp\\", "\\AppData\\Local\\Temp",
    "\\Downloads\\", "\\Desktop\\staging")
| summarize 
    FileCount = count(),
    TotalSize = sum(FileSize)
    by DeviceName, InitiatingProcessAccountName, 
    StagingFolder = extract(@"(\\[^\\]+\\[^\\]+\\)", 1, FolderPath)
| where FileCount > 50 or TotalSize > 104857600
| extend TotalMB = round(TotalSize / 1048576.0, 2)
