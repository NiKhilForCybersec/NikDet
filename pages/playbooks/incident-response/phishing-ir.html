1.2 Scoping Queries

// KQL - Find all recipients of phishing email
let phishingSender = "attacker@phishing-domain.com";
let phishingSubject = "Urgent: Verify your account";
EmailEvents
| where TimeGenerated > ago(48h)
| where SenderFromAddress =~ phishingSender 
    or Subject has phishingSubject
| summarize 
    TotalRecipients = dcount(RecipientEmailAddress),
    Delivered = countif(DeliveryAction == "Delivered"),
    Blocked = countif(DeliveryAction != "Delivered"),
    RecipientList = make_set(RecipientEmailAddress, 100)

// KQL - Check for link clicks
let phishingUrl = "phishing-domain.com";
UrlClickEvents
| where TimeGenerated > ago(48h)
| where Url has phishingUrl
| project TimeGenerated, AccountUpn, Url, ActionType, 
    IsClickedThrough, NetworkMessageId
| summarize 
    Clickers = dcount(AccountUpn),
    ClickerList = make_set(AccountUpn, 50),
    TotalClicks = count()

// KQL - Check for attachment opens
let phishingHash = "SHA256_HASH";
DeviceFileEvents
| where SHA256 == phishingHash
| where ActionType in ("FileCreated", "FileModified")
| summarize 
    AffectedDevices = dcount(DeviceName),
    DeviceList = make_set(DeviceName, 50)

// KQL - Post-click activity for compromised users
let clickedUsers = dynamic(["user1@company.com", "user2@company.com"]);
let clickTime = datetime(2025-01-01T12:00:00Z);
SigninLogs
| where TimeGenerated > clickTime
| where UserPrincipalName in (clickedUsers)
| project TimeGenerated, UserPrincipalName, IPAddress, 
    Location, AppDisplayName, ResultType
| order by TimeGenerated asc

2.2 Email Removal Commands

// PowerShell - Search and purge phishing emails
# Create compliance search
New-ComplianceSearch -Name "Phishing_Remediation_$(Get-Date -Format 'yyyyMMdd')" `
    -ExchangeLocation All `
    -ContentMatchQuery '(From:attacker@phishing-domain.com) AND (Subject:"Urgent")'

# Start the search
Start-ComplianceSearch -Identity "Phishing_Remediation_20250101"

# Review results
Get-ComplianceSearch -Identity "Phishing_Remediation_20250101" | FL

# Purge (soft delete - recoverable)
New-ComplianceSearchAction -SearchName "Phishing_Remediation_20250101" `
    -Purge -PurgeType SoftDelete

# Purge (hard delete - permanent)
New-ComplianceSearchAction -SearchName "Phishing_Remediation_20250101" `
    -Purge -PurgeType HardDelete

// PowerShell - Block sender via transport rule
New-TransportRule -Name "Block Phishing Domain" `
    -FromAddressMatchesPatterns "@phishing-domain\.com$" `
    -RejectMessageReasonText "Blocked by security policy" `
    -RejectMessageEnhancedStatusCode "5.7.1"

Phase 3: Investigation (2-4 hours)

3.1 Compromise Assessment

// KQL - Suspicious sign-ins after phishing
let affectedUsers = dynamic(["user1@company.com", "user2@company.com"]);
let phishingTime = datetime(2025-01-01T12:00:00Z);
SigninLogs
| where TimeGenerated > phishingTime
| where UserPrincipalName in (affectedUsers)
| where ResultType == "0" // Successful
| project TimeGenerated, UserPrincipalName, IPAddress, Location,
    AppDisplayName, DeviceDetail, RiskLevelDuringSignIn
| order by TimeGenerated asc

// KQL - Mailbox rule changes
CloudAppEvents
| where TimeGenerated > phishingTime
| where AccountDisplayName in (affectedUsers)
| where ActionType in ("New-InboxRule", "Set-InboxRule", "Set-Mailbox")
| project TimeGenerated, AccountDisplayName, ActionType, RawEventData

// KQL - Email forwarding added
CloudAppEvents
| where ActionType == "Set-Mailbox"
| where RawEventData has "ForwardingSmtpAddress"
| project TimeGenerated, AccountDisplayName, RawEventData

// KQL - OAuth apps consented
AuditLogs
| where TimeGenerated > phishingTime
| where OperationName == "Consent to application"
| where InitiatedBy.user.userPrincipalName in (affectedUsers)
| project TimeGenerated, User = InitiatedBy.user.userPrincipalName,
    AppName = TargetResources[0].displayName

Escalation Criteria

Escalate to Major Incident If:
  • Executive accounts compromised
  • Finance team targeted with BEC
  • Multiple accounts showing post-compromise activity
  • Malware executed from attachment
  • Attacker accessed sensitive data
  • Attacker sent emails from compromised accounts
  • Evidence of persistence mechanisms
  • Targeted attack (spear phishing) confirmed

Interview-Ready Answer

"Walk me through your phishing incident response process"

My Answer:

"Phishing incident response scales based on scope - single user versus enterprise-wide campaign. Here's my structured approach from detection to lessons learned.

Phase 1 - Detection & Initial Response:
Whether it's a user report or automated detection, I first assess severity. Did anyone click? Did anyone enter credentials? That determines urgency. For credential compromise, I reset passwords and revoke sessions within minutes. For clicked-only, I have more time to analyze the payload.

Phase 2 - Scope the Campaign:
I never treat phishing as a single-user problem. I search email logs for the sender, subject, attachment hashes, and URLs to find all recipients. I categorize them: received only, opened, clicked, entered credentials. This tells me how big the response needs to be.

Phase 3 - Containment & Eradication:
Containment involves purging emails from all mailboxes - I hard delete so users can't recover from trash. I block the sender domain, URLs, and file hashes across email gateway, proxy, and endpoint. For compromised accounts, full remediation: password reset, session revocation, MFA re-enrollment, inbox rule cleanup.

Phase 4 - Recovery & Lessons Learned:
Once contained, I document the full timeline and IOCs for threat intel. I brief affected users on what happened and what to watch for. For significant campaigns, I work with the team to improve controls - maybe we need better email filtering, enhanced user training, or new detection rules based on the attack patterns."

💡 Pro Tips for Interview:
  • Show your response scales with scope - single user vs campaign
  • Emphasize that credential compromise = immediate containment
  • Know your containment tools - email purge, block sender, revoke sessions
  • Mention lessons learned and control improvements
