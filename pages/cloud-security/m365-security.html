M365 Detection Queries

// KQL - Inbox rules with forwarding or deletion
CloudAppEvents
| where ActionType == "New-InboxRule"
| where RawEventData has_any ("ForwardTo", "RedirectTo", "DeleteMessage", "MoveToFolder")
| extend RuleDetails = parse_json(RawEventData)
| project TimeGenerated, AccountDisplayName, ActionType, RuleDetails

// KQL - Email auto-forwarding to external
CloudAppEvents
| where ActionType == "Set-Mailbox"
| where RawEventData has "ForwardingSmtpAddress"
| extend ForwardAddress = extract(@"ForwardingSmtpAddress.*?([a-zA-Z0-9._%+-]+@[a-zA-Z0-9.-]+)", 1, tostring(RawEventData))
| where ForwardAddress !endswith "yourcompany.com"
| project TimeGenerated, AccountDisplayName, ForwardAddress

// KQL - OAuth application consent
AuditLogs
| where OperationName == "Consent to application"
| extend AppName = tostring(TargetResources[0].displayName)
| extend ConsentUser = tostring(InitiatedBy.user.userPrincipalName)
| extend Permissions = tostring(TargetResources[0].modifiedProperties)
| project TimeGenerated, ConsentUser, AppName, Permissions

// KQL - Risky OAuth apps (mail/files access)
AuditLogs
| where OperationName == "Consent to application"
| where TargetResources[0].modifiedProperties has_any ("Mail.Read", "Mail.ReadWrite", "Files.Read", "Files.ReadWrite")
| extend AppName = tostring(TargetResources[0].displayName)
| project TimeGenerated, InitiatedBy.user.userPrincipalName, AppName

// KQL - SharePoint external sharing
CloudAppEvents
| where ActionType in ("SharingSet", "AnonymousLinkCreated", "SecureLinkCreated")
| where RawEventData has "External"
| project TimeGenerated, AccountDisplayName, ActionType, ObjectName

// KQL - Mass file downloads
CloudAppEvents
| where ActionType == "FileDownloaded"
| summarize DownloadCount = count(), Files = make_set(ObjectName, 10) by AccountDisplayName, bin(TimeGenerated, 1h)
| where DownloadCount > 50

// KQL - Admin role assignments
AuditLogs
| where OperationName has "Add member to role"
| extend Role = tostring(TargetResources[0].modifiedProperties[1].newValue)
| extend Target = tostring(TargetResources[0].userPrincipalName)
| extend Actor = tostring(InitiatedBy.user.userPrincipalName)
| project TimeGenerated, Actor, Target, Role

// KQL - eDiscovery searches
CloudAppEvents
| where ActionType in ("SearchCreated", "SearchStarted", "SearchExported")
| project TimeGenerated, AccountDisplayName, ActionType, ObjectName

// KQL - Mailbox permission changes
CloudAppEvents
| where ActionType in ("Add-MailboxPermission", "Add-RecipientPermission")
| project TimeGenerated, AccountDisplayName, ActionType, RawEventData

Investigation Checklist

M365 Compromise Investigation
  • Check sign-in logs - Unusual locations, IPs, devices
  • Review inbox rules - Forwarding, deletion rules
  • Check OAuth apps - Unauthorized app consent
  • Review sent items - Phishing sent from account
  • Check mailbox permissions - Delegated access added
  • Review file activity - Mass downloads, external sharing
  • Check Teams/SharePoint - External sharing enabled
