FortiGate Authentication

Configure LDAP, RADIUS, FSSO, and user-based firewall policies.

LDAP Server Configuration

config user ldap
    edit "AD-LDAP"
        set server "10.0.1.10"
        set secondary-server "10.0.1.11"
        set cnid "sAMAccountName"
        set dn "DC=company,DC=local"
        set type regular
        set username "CN=svc-fortigate,OU=Service Accounts,DC=company,DC=local"
        set password "ServiceAccountPassword123!"
        set group-member-check user-attr
        set group-search-base "DC=company,DC=local"
        set secure ldaps
        set port 636
    next
end

Test LDAP Connection

diagnose test authserver ldap "AD-LDAP" jsmith Password123!

User Groups

config user group
    edit "VPN-Users"
        set member "AD-LDAP"
        config match
            edit 1
                set server-name "AD-LDAP"
                set group-name "CN=VPN-Users,OU=Groups,DC=company,DC=local"
            next
        end
    next
    edit "IT-Admins"
        set member "AD-LDAP"
        config match
            edit 1
                set server-name "AD-LDAP"
                set group-name "CN=IT-Admins,OU=Groups,DC=company,DC=local"
            next
        end
    next
end

FSSO (Fortinet Single Sign-On)

# Configure FSSO Collector Agent
config user fsso
    edit "FSSO-Agent"
        set server "10.0.1.15"
        set port 8000
        set password "FSSoPassword123!"
        set ldap-server "AD-LDAP"
    next
end

# Enable FSSO on interface
config system interface
    edit "internal"
        set device-identification enable
    next
end

User-Based Policy

config firewall policy
    edit 50
        set name "IT-Admin-FullAccess"
        set srcintf "internal"
        set dstintf "wan1"
        set srcaddr "all"
        set dstaddr "all"
        set action accept
        set schedule "always"
        set service "ALL"
        set groups "IT-Admins"
        set utm-status enable
        set av-profile "default"
        set webfilter-profile "default"
    next
end

Verification

# Show authenticated users
diagnose firewall auth list

# Show FSSO users
diagnose debug authd fsso list

# Test authentication
diagnose test authserver ldap "AD-LDAP" username password
