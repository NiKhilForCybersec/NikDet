Authentication Alert Triage

Comprehensive triage procedure for authentication-related security alerts including failed logins, brute force, password spray, MFA bypass, impossible travel, and credential stuffing attacks.

TierL1 / L2
Time5-15 min
ToolsSIEM, Azure AD, AD

Triage Decision Flow

1
Identify Attack Pattern
PatternSingle UserMultiple UsersLikely Attack
Many failuresYesNoBrute Force
Few failures per userNoYesPassword Spray
Same source IPVariesYesAutomated attack
Different passwordsYesNoCredential stuffing
Same passwordNoYesPassword spray
2
Check for Successful Login
  • All failures: Lower risk, monitor for escalation
  • Success after failures: Higher risk, verify legitimacy
  • Success with no failures: Check for stolen credentials
3
Assess Source Reputation
Source TypeRisk LevelAction
Corporate IP/VPNLowLikely legitimate user
Residential ISPMediumCould be WFH or attacker
Cloud/HostingHighLikely automated attack
Tor/AnonymousHighSuspicious by nature
Foreign countryMedium-HighCheck user's travel
4
Verify MFA Status
  • MFA required and passed: Stronger confidence in legitimacy
  • MFA bypassed: Legacy auth or conditional access gap
  • MFA failed multiple times: MFA fatigue attack
  • MFA not required: Policy gap, higher risk

Triage Queries

Azure AD / Entra ID Queries

// KQL - Failed login analysis for specific user
let targetUser = "user@company.com";
SigninLogs
| where TimeGenerated > ago(24h)
| where UserPrincipalName =~ targetUser
| where ResultType != "0"
| project TimeGenerated, ResultType, ResultDescription,
    IPAddress, Location, AppDisplayName, 
    ClientAppUsed, DeviceDetail, UserAgent
| order by TimeGenerated desc

// KQL - Brute force detection (single user, many failures)
SigninLogs
| where TimeGenerated > ago(1h)
| where ResultType != "0"
| summarize 
    FailureCount = count(),
    DistinctIPs = dcount(IPAddress),
    IPList = make_set(IPAddress, 5),
    ErrorCodes = make_set(ResultType, 5)
    by UserPrincipalName
| where FailureCount > 10
| order by FailureCount desc

// KQL - Password spray detection (many users, same errors)
SigninLogs
| where TimeGenerated > ago(1h)
| where ResultType in ("50126", "50053", "50055") // Bad password errors
| summarize 
    AttemptCount = count(),
    TargetedUsers = dcount(UserPrincipalName),
    UserList = make_set(UserPrincipalName, 10)
    by IPAddress
| where TargetedUsers > 5
| order by TargetedUsers desc

// KQL - Successful login after failures
let failedLogins = SigninLogs
| where TimeGenerated > ago(24h)
| where ResultType != "0"
| summarize FailureCount = count() by UserPrincipalName, IPAddress;
SigninLogs
| where TimeGenerated > ago(24h)
| where ResultType == "0"
| join kind=inner failedLogins on UserPrincipalName, IPAddress
| where FailureCount > 3
| project TimeGenerated, UserPrincipalName, IPAddress, 
    FailureCount, AppDisplayName, Location

// KQL - Impossible travel detection
SigninLogs
| where TimeGenerated > ago(7d)
| where ResultType == "0"
| project TimeGenerated, UserPrincipalName, IPAddress, 
    Location, City = tostring(LocationDetails.city),
    Latitude = todouble(LocationDetails.geoCoordinates.latitude),
    Longitude = todouble(LocationDetails.geoCoordinates.longitude)
| order by UserPrincipalName, TimeGenerated asc
| extend PrevTime = prev(TimeGenerated), 
    PrevLocation = prev(Location),
    PrevLat = prev(Latitude), PrevLong = prev(Longitude),
    PrevUser = prev(UserPrincipalName)
| where UserPrincipalName == PrevUser
| extend TimeDiffHours = datetime_diff('hour', TimeGenerated, PrevTime)
| extend DistanceKm = geo_distance_2points(Longitude, Latitude, PrevLong, PrevLat) / 1000
| extend SpeedKmH = DistanceKm / TimeDiffHours
| where SpeedKmH > 500 and TimeDiffHours < 24 // Faster than commercial flight
| project TimeGenerated, UserPrincipalName, Location, PrevLocation,
    TimeDiffHours, DistanceKm, SpeedKmH

// KQL - MFA fatigue attack detection
SigninLogs
| where TimeGenerated > ago(1h)
| where ResultType in ("50074", "50076") // MFA required/denied
| summarize 
    MFAAttempts = count(),
    TimeSpan = max(TimeGenerated) - min(TimeGenerated)
    by UserPrincipalName, IPAddress
| where MFAAttempts > 5
| extend AttemptsPerMinute = MFAAttempts / (TimeSpan / 1m)
| where AttemptsPerMinute > 1
| order by MFAAttempts desc

// KQL - Legacy authentication usage
SigninLogs
| where TimeGenerated > ago(24h)
| where ClientAppUsed in ("Exchange ActiveSync", "IMAP4", 
    "POP3", "SMTP", "Authenticated SMTP", "Other clients")
| project TimeGenerated, UserPrincipalName, ClientAppUsed,
    IPAddress, Location, ResultType, AppDisplayName

Windows AD Queries

// KQL - Windows failed logins (Event 4625)
SecurityEvent
| where TimeGenerated > ago(24h)
| where EventID == 4625
| extend FailureReason = case(
    SubStatus == "0xc000006a", "Bad password",
    SubStatus == "0xc0000064", "User doesn't exist",
    SubStatus == "0xc000006d", "Bad username or password",
    SubStatus == "0xc0000072", "Account disabled",
    SubStatus == "0xc0000234", "Account locked",
    "Other"
)
| summarize FailureCount = count() by TargetUserName, 
    IpAddress, FailureReason, Computer
| where FailureCount > 5
| order by FailureCount desc

// KQL - Account lockouts (Event 4740)
SecurityEvent
| where TimeGenerated > ago(24h)
| where EventID == 4740
| project TimeGenerated, TargetUserName, Computer, 
    SubjectUserName
| order by TimeGenerated desc

// KQL - Successful logins after failures
let failures = SecurityEvent
| where TimeGenerated > ago(24h)
| where EventID == 4625
| summarize FailCount = count() by TargetUserName;
SecurityEvent
| where TimeGenerated > ago(24h)
| where EventID == 4624
| where LogonType in (2, 3, 10) // Interactive, Network, RemoteInteractive
| join kind=inner failures on $left.TargetUserName == $right.TargetUserName
| where FailCount > 3
| project TimeGenerated, TargetUserName, Computer, 
    IpAddress, LogonType, FailCount

// SPL - Failed login summary
index=wineventlog EventCode=4625
| stats count as failures, dc(src_ip) as unique_ips, 
    values(src_ip) as ip_list by user
| where failures > 10
| sort -failures

Escalation Criteria

Escalate Immediately If:
  • Successful login after password spray attack
  • MFA fatigue attack with eventual MFA approval
  • Privileged account targeted (admin, service account)
  • Impossible travel with suspicious post-login activity
  • Legacy auth bypass leading to data access
  • Multiple users compromised in same attack
  • Token theft or session hijacking indicators
  • Post-login activity shows C2 or exfiltration
