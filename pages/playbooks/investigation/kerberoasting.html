Investigation Steps

Step 1: Detect Kerberoasting

KQL - Kerberoasting Detection (MDE)

// Detect Kerberoasting - high volume TGS requests for unique SPNs
DeviceEvents
| where TimeGenerated > ago(24h)
| where ActionType == "KerberosTgsRequest"
| extend ServiceName = tostring(AdditionalFields.ServiceName)
| where ServiceName !endswith "$"  // Exclude machine accounts
| summarize 
    UniqueServices = dcount(ServiceName),
    ServiceList = make_set(ServiceName, 50),
    RequestCount = count()
    by DeviceName, AccountName
| where UniqueServices > 10  // Threshold for suspicious activity
| order by UniqueServices desc

Windows Event Log Query - Event 4769

// Kerberoasting via Windows Security Events
SecurityEvent
| where TimeGenerated > ago(24h)
| where EventID == 4769  // TGS Request
| where ServiceName !endswith "$"  // Not machine account
| where TicketEncryptionType in ("0x17", "0x18")  // RC4 encryption (weak)
| summarize 
    RequestCount = count(),
    UniqueServices = dcount(ServiceName),
    Services = make_set(ServiceName, 50)
    by TargetUserName, IpAddress, Computer
| where RequestCount > 20 and UniqueServices > 10
| order by UniqueServices desc

SPL - Kerberoasting Detection (Splunk)

index=wineventlog sourcetype=WinEventLog:Security EventCode=4769
| where ServiceName!="*$"
| where TicketEncryptionType IN ("0x17", "0x18")
| stats dc(ServiceName) as UniqueServices count by TargetUserName, IpAddress
| where UniqueServices > 10
| sort -UniqueServices

Step 2: Detect AS-REP Roasting

KQL - AS-REP Roasting Detection

// Detect AS-REP Roasting - requests for accounts without preauth
SecurityEvent
| where TimeGenerated > ago(24h)
| where EventID == 4768  // TGT Request
| where PreAuthType == "0"  // No preauthentication
| summarize 
    RequestCount = count(),
    UniqueAccounts = dcount(TargetUserName),
    Accounts = make_set(TargetUserName, 50),
    SourceIPs = make_set(IpAddress)
    by Computer
| where RequestCount > 5
| order by RequestCount desc

Find Accounts Vulnerable to AS-REP Roasting

// PowerShell - Find vulnerable accounts
Get-ADUser -Filter {DoesNotRequirePreAuth -eq $true} -Properties DoesNotRequirePreAuth |
    Select-Object Name, SamAccountName, Enabled, DoesNotRequirePreAuth

// KQL - Audit these accounts
IdentityInfo
| where TimeGenerated > ago(1d)
| where AccountDomain !endswith ".local"  // Adjust for your domain
| where isnotempty(OnPremisesDistinguishedName)
// Cross-reference with AD query results

Step 3: Detect Pass-the-Hash

KQL - Pass-the-Hash Detection

// Detect PtH - NTLM auth from unexpected sources
SecurityEvent
| where TimeGenerated > ago(24h)
| where EventID == 4624  // Successful logon
| where LogonType == 3   // Network logon
| where AuthenticationPackageName == "NTLM"
| where TargetUserName !endswith "$"  // Not machine account
| summarize 
    LogonCount = count(),
    SourceHosts = make_set(WorkstationName),
    SourceIPs = make_set(IpAddress)
    by TargetUserName, TargetDomainName
| where array_length(SourceHosts) > 3  // User logging in from many hosts
| order by LogonCount desc

KQL - NTLM Downgrade Attack

// Detect NTLM when Kerberos should be used
SecurityEvent
| where TimeGenerated > ago(24h)
| where EventID == 4624
| where AuthenticationPackageName == "NTLM"
| where LmPackageName == "NTLM V1"  // Very weak
| summarize count() by TargetUserName, WorkstationName, IpAddress
| order by count_ desc

Step 4: Detect Golden/Silver Ticket

KQL - Golden Ticket Indicators

// Detect potential Golden Ticket usage
SecurityEvent
| where TimeGenerated > ago(7d)
| where EventID == 4769  // TGS Request
| where TargetUserName == "Administrator" or TargetUserName == "krbtgt"
| extend TicketOptions = extract("Ticket Options:\\s*([^\\r\\n]+)", 1, EventData)
| where TicketOptions contains "renewable"
| project TimeGenerated, TargetUserName, ServiceName, IpAddress, Computer, TicketOptions
| order by TimeGenerated desc

// Also look for Event 4768 with unusual properties
SecurityEvent
| where TimeGenerated > ago(7d)
| where EventID == 4768
| where TicketEncryptionType == "0x17"  // RC4 - often used with Golden Ticket
| where PreAuthType != "2"  // Not standard preauth
| summarize count() by TargetUserName, IpAddress

KQL - DCSync Detection

// Detect DCSync - replication from non-DC
SecurityEvent
| where TimeGenerated > ago(24h)
| where EventID == 4662  // Directory service access
| where OperationType == "Object Access"
| where ObjectType contains "Domain" or ObjectType contains "Replication"
| where AccessMask has_any ("0x100", "Replicating Directory Changes")
| where SubjectUserName !endswith "$"  // Not a DC machine account
| project TimeGenerated, SubjectUserName, ObjectType, ObjectName, AccessMask
| order by TimeGenerated desc

Step 5: Identify Attack Tools

KQL - Credential Attack Tool Detection

// Detect common credential attack tools
DeviceProcessEvents
| where TimeGenerated > ago(7d)
| where FileName in~ ("mimikatz.exe", "rubeus.exe", "kekeo.exe", 
    "SharpRoast.exe", "Invoke-Kerberoast.ps1", "GetUserSPNs.py")
    or ProcessCommandLine has_any (
        "sekurlsa::logonpasswords",
        "lsadump::dcsync",
        "kerberos::golden",
        "kerberos::ptt",
        "kerberos::list",
        "asreproast",
        "kerberoast",
        "Invoke-Mimikatz",
        "Invoke-Rubeus",
        "Get-DomainSPNTicket"
    )
| project TimeGenerated, DeviceName, FileName, ProcessCommandLine, AccountName, 
    InitiatingProcessFileName
| order by TimeGenerated desc

KQL - LSASS Access Detection

// Detect suspicious LSASS access (credential dumping)
DeviceProcessEvents
| where TimeGenerated > ago(24h)
| where FileName == "lsass.exe" 
    or ProcessCommandLine has "lsass"
| where InitiatingProcessFileName !in~ ("svchost.exe", "csrss.exe", 
    "wininit.exe", "MsMpEng.exe", "mrt.exe")
| project TimeGenerated, DeviceName, InitiatingProcessFileName, 
    InitiatingProcessCommandLine, FileName
| order by TimeGenerated desc

PowerShell - KRBTGT Reset

# CRITICAL: Reset KRBTGT password (twice with gap)
# This invalidates ALL Kerberos tickets including Golden Tickets

# First reset
Set-ADAccountPassword -Identity krbtgt -Reset -NewPassword (
    ConvertTo-SecureString -AsPlainText (
        [System.Web.Security.Membership]::GeneratePassword(32,8)
    ) -Force
)

Write-Host "First KRBTGT reset complete. Wait 10+ hours before second reset."
Write-Host "This allows tickets to naturally expire."

# After 10+ hours, run second reset
# Set-ADAccountPassword -Identity krbtgt -Reset -NewPassword (...)

PowerShell - Fix Vulnerable Accounts

# Remove "Do not require Kerberos preauthentication"
Get-ADUser -Filter {DoesNotRequirePreAuth -eq $true} | 
    Set-ADAccountControl -DoesNotRequirePreAuth $false

# Find and fix service accounts with weak SPNs
Get-ADUser -Filter {ServicePrincipalName -like "*"} -Properties ServicePrincipalName |
    ForEach-Object {
        Write-Host "Account: $($_.SamAccountName)"
        Write-Host "SPNs: $($_.ServicePrincipalName -join ', ')"
        # Consider migrating to gMSA
    }

# Migrate service account to gMSA
New-ADServiceAccount -Name "gMSA_ServiceName" -DNSHostName "server.domain.com" `
    -PrincipalsAllowedToRetrieveManagedPassword "ServerGroup"
