Token Theft Techniques

Adversary-in-the-Middle (AiTM) Phishing

Most Common Modern Attack

How it works:

  1. Attacker sends phishing email with link to proxy site
  2. Victim enters credentials on fake login page
  3. Proxy forwards credentials to real Microsoft/Google login
  4. Victim completes MFA challenge (attacker relays this too)
  5. Attacker captures the resulting session cookie/token
  6. Attacker replays token from their own device - MFA bypassed

Tools used: Evilginx2, Modlishka, Muraena

Investigation Steps

Step 1: Identify Token Misuse

KQL - Detect AiTM Session Hijacking

// Detect sign-ins where session was established without interactive auth
// Indicates potential token replay
SigninLogs
| where TimeGenerated > ago(7d)
| where ResultType == 0  // Successful
| where AuthenticationRequirement == "singleFactorAuthentication"
    or IsInteractive == false
| where TokenIssuerType == "AzureAD"
| summarize 
    SignInCount = count(),
    UniqueIPs = dcount(IPAddress),
    UniqueDevices = dcount(DeviceDetail.deviceId),
    IPs = make_set(IPAddress),
    Locations = make_set(Location)
    by UserPrincipalName, AppDisplayName
| where UniqueIPs > 3 or UniqueDevices > 3
| order by UniqueIPs desc

KQL - Impossible Travel with Token

// Find users authenticating from impossible locations
SigninLogs
| where TimeGenerated > ago(24h)
| where ResultType == 0
| project TimeGenerated, UserPrincipalName, IPAddress, Location, 
    Latitude = toreal(LocationDetails.geoCoordinates.latitude),
    Longitude = toreal(LocationDetails.geoCoordinates.longitude)
| order by UserPrincipalName, TimeGenerated asc
| serialize
| extend PrevTime = prev(TimeGenerated), PrevLat = prev(Latitude), 
    PrevLong = prev(Longitude), PrevUser = prev(UserPrincipalName)
| where UserPrincipalName == PrevUser
| extend TimeDiffMinutes = datetime_diff('minute', TimeGenerated, PrevTime)
| extend DistanceKm = geo_distance_2points(Longitude, Latitude, PrevLong, PrevLat) / 1000
| extend SpeedKmh = DistanceKm / (TimeDiffMinutes / 60.0)
| where SpeedKmh > 500 and TimeDiffMinutes < 60  // Impossible speed
| project TimeGenerated, UserPrincipalName, Location, SpeedKmh, DistanceKm, TimeDiffMinutes

KQL - Token Used Without Interactive Login

// Find tokens being used from devices that never completed interactive auth
let TokenUsage = SigninLogs
| where TimeGenerated > ago(7d)
| where IsInteractive == false
| summarize TokenUseCount = count() by UserPrincipalName, DeviceDetail.deviceId, IPAddress
| where TokenUseCount > 5;
let InteractiveLogins = SigninLogs
| where TimeGenerated > ago(30d)
| where IsInteractive == true
| where ResultType == 0
| summarize by UserPrincipalName, DeviceDetail.deviceId;
TokenUsage
| join kind=leftanti InteractiveLogins on UserPrincipalName, $left.DeviceDetail_deviceId == $right.DeviceDetail_deviceId
| project UserPrincipalName, DeviceDetail_deviceId, IPAddress, TokenUseCount

SPL - Session Anomaly Detection (Splunk)

index=azure sourcetype=azure:aad:signin
| stats dc(src_ip) as UniqueIPs dc(device_id) as UniqueDevices count by user
| where UniqueIPs > 5 OR UniqueDevices > 5
| sort -UniqueIPs

Step 2: Identify Token Source

KQL - Check for AiTM Phishing

// Look for sign-ins immediately after clicking email links
let SuspiciousUsers = SigninLogs
| where TimeGenerated > ago(24h)
| where RiskState == "confirmedCompromised" or RiskLevelDuringSignIn in ("high", "medium")
| distinct UserPrincipalName;
// Check email activity before compromise
OfficeActivity
| where TimeGenerated > ago(24h)
| where UserId in (SuspiciousUsers)
| where Operation == "ClickedUrl" or Operation == "UrlClicked"
| project TimeGenerated, UserId, Operation, SourceFileName, Url = UrlClickedItem
| order by TimeGenerated desc

KQL - Token Theft Tool Detection

// Detect tools used to extract tokens
DeviceProcessEvents
| where TimeGenerated > ago(7d)
| where ProcessCommandLine has_any (
    "sekurlsa::cloudap",      // Mimikatz PRT
    "Get-AADIntAccessToken",   // AADInternals
    "roadrecon",               // ROADTools
    "TokenTactics",            // TokenTactics
    "Invoke-RefreshToken",     // Token refresh
    "browserpassview",         // NirSoft
    "ChromePass",              // Password dumpers
    "WebBrowserPassView"
)
    or FileName in~ ("mimikatz.exe", "roadrecon.exe", "aadint.ps1")
| project TimeGenerated, DeviceName, FileName, ProcessCommandLine, AccountName

Step 3: Assess Token Scope and Usage

KQL - What Did Attacker Access with Token?

// Track all activity for compromised user
let CompromisedUser = "user@domain.com";
let CompromiseTime = datetime(2024-01-15T10:00:00Z);
union
(
    SigninLogs
    | where TimeGenerated > CompromiseTime
    | where UserPrincipalName == CompromisedUser
    | project TimeGenerated, Activity = "SignIn", 
        Details = strcat(AppDisplayName, " from ", IPAddress, " (", Location, ")")
),
(
    AuditLogs
    | where TimeGenerated > CompromiseTime
    | where InitiatedBy.user.userPrincipalName == CompromisedUser
    | project TimeGenerated, Activity = OperationName, 
        Details = tostring(TargetResources)
),
(
    OfficeActivity
    | where TimeGenerated > CompromiseTime
    | where UserId == CompromisedUser
    | project TimeGenerated, Activity = Operation, 
        Details = strcat(Workload, ": ", OfficeObjectId)
)
| order by TimeGenerated asc

KQL - Check for Malicious OAuth App

// Find suspicious OAuth app consents
AuditLogs
| where TimeGenerated > ago(30d)
| where OperationName == "Consent to application"
| extend AppId = tostring(TargetResources[0].id)
| extend AppName = tostring(TargetResources[0].displayName)
| extend ConsentUser = tostring(InitiatedBy.user.userPrincipalName)
| extend Permissions = tostring(TargetResources[0].modifiedProperties)
| project TimeGenerated, ConsentUser, AppName, AppId, Permissions
| where Permissions has_any ("Mail.Read", "Mail.Send", "Files.ReadWrite", 
    "Directory.Read", "User.Read.All")
| order by TimeGenerated desc

PowerShell - Revoke All Tokens

# Connect to Microsoft Graph
Connect-MgGraph -Scopes "User.ReadWrite.All"

# Revoke all refresh tokens for user
$userId = "user@domain.com"
Revoke-MgUserSignInSession -UserId $userId

# Also revoke using Azure AD PowerShell
Connect-AzureAD
Revoke-AzureADUserAllRefreshToken -ObjectId $userId

# Force password change
Update-MgUser -UserId $userId -PasswordProfile @{
    ForceChangePasswordNextSignIn = $true
}

# Disable user temporarily if needed
Update-MgUser -UserId $userId -AccountEnabled:$false

PowerShell - Remove Malicious OAuth App

# Find OAuth grants for user
Get-MgUserOAuth2PermissionGrant -UserId "user@domain.com"

# Remove specific OAuth grant
Remove-MgOAuth2PermissionGrant -OAuth2PermissionGrantId "GRANT_ID"

# Disable the malicious app enterprise-wide
$appId = "MALICIOUS_APP_ID"
$sp = Get-MgServicePrincipal -Filter "appId eq '$appId'"
Update-MgServicePrincipal -ServicePrincipalId $sp.Id -AccountEnabled:$false
