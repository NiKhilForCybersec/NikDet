PB-INV-010

Privilege Escalation Investigation

Investigate unauthorized elevation of privileges from standard user to administrator or SYSTEM level access.

SeverityHIGH
TierL2/L3
Time30-90 min
T1068 - Exploitation for Privilege Escalation T1548 - Abuse Elevation Control Mechanism T1134 - Access Token Manipulation

Part 1: Detect Credential Theft

// KQL: LSASS access (credential dumping)
DeviceProcessEvents
| where TimeGenerated > ago(7d)
| where FileName =~ "lsass.exe"
| where ActionType == "ProcessAccessed"
| where InitiatingProcessFileName !in~ ("csrss.exe", "services.exe", "svchost.exe", "wininit.exe")
| project TimeGenerated, DeviceName, InitiatingProcessFileName, 
    InitiatingProcessCommandLine, AccountName

// KQL: Mimikatz and credential dumping tools
DeviceProcessEvents
| where TimeGenerated > ago(7d)
| where ProcessCommandLine has_any ("sekurlsa", "logonpasswords", "lsadump", 
    "kerberos::list", "crypto::capi", "privilege::debug", "mimikatz")
    or FileName in~ ("mimikatz.exe", "procdump.exe", "lazagne.exe")
| project TimeGenerated, DeviceName, AccountName, FileName, ProcessCommandLine

// KQL: Comsvcs.dll LSASS dump (LOLBin)
DeviceProcessEvents
| where TimeGenerated > ago(7d)
| where ProcessCommandLine has "comsvcs.dll" and ProcessCommandLine has "MiniDump"
| project TimeGenerated, DeviceName, AccountName, ProcessCommandLine

Part 2: Detect UAC Bypass

// KQL: Common UAC bypass techniques
DeviceProcessEvents
| where TimeGenerated > ago(7d)
| where InitiatingProcessFileName in~ ("fodhelper.exe", "eventvwr.exe", 
    "sdclt.exe", "slui.exe", "computerdefaults.exe")
| where InitiatingProcessIntegrityLevel == "High"  // Auto-elevated
| where FileName !in~ ("mmc.exe", "control.exe")  // Expected children
| project TimeGenerated, DeviceName, AccountName, 
    InitiatingProcessFileName, FileName, ProcessCommandLine

// KQL: Environment variable hijacking for UAC bypass
DeviceRegistryEvents
| where TimeGenerated > ago(7d)
| where RegistryKey has @"Environment\windir" 
    or RegistryKey has @"Environment\systemroot"
| project TimeGenerated, DeviceName, AccountName, RegistryKey, RegistryValueData

Part 3: Detect Service/Scheduled Task Abuse

// KQL: Service creation (Event 7045)
SecurityEvent
| where TimeGenerated > ago(7d)
| where EventID == 7045  // Service installed
| where ServiceAccount in ("LocalSystem", "NT AUTHORITY\\SYSTEM")
| project TimeGenerated, Computer, ServiceName, ServiceFileName, ServiceAccount

// KQL: Scheduled task creation with SYSTEM privileges
DeviceProcessEvents
| where TimeGenerated > ago(7d)
| where FileName =~ "schtasks.exe"
| where ProcessCommandLine has_any ("/ru system", "/ru \"system\"", 
    "highest", "SYSTEM")
| project TimeGenerated, DeviceName, AccountName, ProcessCommandLine

// KQL: Service binary path modification
DeviceRegistryEvents
| where TimeGenerated > ago(7d)
| where RegistryKey has @"SYSTEM\CurrentControlSet\Services\"
| where RegistryValueName == "ImagePath"
| project TimeGenerated, DeviceName, AccountName, RegistryKey, RegistryValueData

Part 4: Detect Group Membership Changes

// KQL: User added to privileged groups
SecurityEvent
| where TimeGenerated > ago(7d)
| where EventID in (4728, 4732, 4756)  // Member added to security groups
| where TargetUserName has_any ("Admin", "Domain Admins", "Enterprise Admins",
    "Schema Admins", "Backup Operators", "Account Operators")
| project TimeGenerated, Computer, SubjectUserName, MemberName, TargetUserName

// KQL: Azure AD role assignment
AuditLogs
| where TimeGenerated > ago(7d)
| where OperationName has "Add member to role"
| where TargetResources[0].displayName has_any ("Global Administrator", 
    "Privileged Role Administrator", "Security Administrator")
| project TimeGenerated, InitiatedBy.user.userPrincipalName, 
    TargetResources[0].displayName, TargetResources[0].userPrincipalName

Part 5: Token Manipulation

// KQL: SeDebugPrivilege use (required for token theft)
SecurityEvent
| where TimeGenerated > ago(7d)
| where EventID == 4673  // Privileged service called
| where PrivilegeList has "SeDebugPrivilege"
| where ProcessName !has "lsass.exe"  // Expected
| project TimeGenerated, Computer, SubjectUserName, ProcessName, PrivilegeList

// KQL: Token impersonation indicators
DeviceProcessEvents
| where TimeGenerated > ago(7d)
| where ProcessCommandLine has_any ("token::", "getsystem", "incognito", 
    "impersonate", "steal_token")
| project TimeGenerated, DeviceName, AccountName, ProcessCommandLine
