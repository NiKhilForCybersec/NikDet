Advanced Detection Patterns

Authentication Attacks

// Password spray with successful login
let threshold = 10;
let timeframe = 5m;
SigninLogs
| where TimeGenerated > ago(1h)
| where ResultType == "50126" // Invalid username or password
| summarize FailedAttempts = count(), TargetAccounts = make_set(UserPrincipalName) by IPAddress, bin(TimeGenerated, timeframe)
| where FailedAttempts > threshold
| join kind=inner (
    SigninLogs
    | where ResultType == 0
    | project SuccessTime = TimeGenerated, UserPrincipalName, IPAddress
) on IPAddress
| where SuccessTime between (TimeGenerated .. (TimeGenerated + timeframe))
| project TimeGenerated, IPAddress, FailedAttempts, TargetAccounts, CompromisedAccount = UserPrincipalName

// Impossible travel with risk context
let threshold_km = 500;
let threshold_minutes = 60;
SigninLogs
| where TimeGenerated > ago(24h)
| where ResultType == 0
| project TimeGenerated, UserPrincipalName, IPAddress, Location, Latitude = todouble(LocationDetails.geoCoordinates.latitude), Longitude = todouble(LocationDetails.geoCoordinates.longitude)
| sort by UserPrincipalName, TimeGenerated
| extend PrevLat = prev(Latitude), PrevLon = prev(Longitude), PrevTime = prev(TimeGenerated), PrevUser = prev(UserPrincipalName)
| where UserPrincipalName == PrevUser
| extend Distance_km = geo_distance_2points(Longitude, Latitude, PrevLon, PrevLat) / 1000
| extend TimeDiff_min = datetime_diff('minute', TimeGenerated, PrevTime)
| where Distance_km > threshold_km and TimeDiff_min < threshold_minutes
| project TimeGenerated, UserPrincipalName, FromLocation = strcat(PrevLat, ",", PrevLon), ToLocation = strcat(Latitude, ",", Longitude), Distance_km, TimeDiff_min

Endpoint Threats

// Living off the land - suspicious process chains
DeviceProcessEvents
| where TimeGenerated > ago(24h)
| where InitiatingProcessFileName in~ ("winword.exe", "excel.exe", "powerpnt.exe", "outlook.exe")
| where FileName in~ ("powershell.exe", "cmd.exe", "wscript.exe", "cscript.exe", "mshta.exe", "regsvr32.exe", "rundll32.exe", "certutil.exe")
| project TimeGenerated, DeviceName, AccountName, InitiatingProcessFileName, FileName, ProcessCommandLine
| extend DecodedCommand = base64_decode_tostring(extract(@"-[Ee](?:nc|ncodedcommand)?\s+([A-Za-z0-9+/=]+)", 1, ProcessCommandLine))

// Credential dumping via LSASS
DeviceProcessEvents
| where TimeGenerated > ago(24h)
| where FileName =~ "lsass.exe" or ProcessCommandLine has_any ("lsass", "mimikatz", "sekurlsa", "procdump", "comsvcs.dll")
| where InitiatingProcessFileName !in~ ("csrss.exe", "services.exe", "svchost.exe", "wininit.exe")
| project TimeGenerated, DeviceName, AccountName, InitiatingProcessFileName, FileName, ProcessCommandLine

// Ransomware indicators
DeviceFileEvents
| where TimeGenerated > ago(1h)
| where ActionType == "FileCreated"
| where FileName matches regex @"\.(encrypted|locked|crypted|enc|cry|[a-z]{5,8})$"
| summarize EncryptedFiles = count(), Folders = dcount(FolderPath) by DeviceName, bin(TimeGenerated, 5m)
| where EncryptedFiles > 100 or Folders > 10
| project TimeGenerated, DeviceName, EncryptedFiles, Folders

Network & C2 Detection

// Beaconing pattern detection
DeviceNetworkEvents
| where TimeGenerated > ago(24h)
| where RemoteIPType == "Public"
| summarize ConnectionCount = count(), Times = make_list(TimeGenerated) by DeviceName, RemoteIP, RemotePort
| where ConnectionCount > 50
| extend Intervals = array_sort_asc(Times)
| mv-apply Intervals to typeof(datetime) on (
    extend NextTime = next(Intervals)
    | extend Interval = datetime_diff('second', NextTime, Intervals)
    | summarize AvgInterval = avg(Interval), StdevInterval = stdev(Interval)
)
| where StdevInterval < 5 and AvgInterval between (30 .. 3600)
| project DeviceName, RemoteIP, RemotePort, ConnectionCount, AvgInterval, StdevInterval

// DNS tunneling detection
DnsEvents
| where TimeGenerated > ago(24h)
| extend SubdomainLength = strlen(tostring(split(Name, ".")[0]))
| extend DomainParts = array_length(split(Name, "."))
| where SubdomainLength > 30 or DomainParts > 6
| summarize QueryCount = count(), AvgSubdomainLen = avg(SubdomainLength) by ClientIP, tostring(split(Name, ".")[-2])
| where QueryCount > 100
