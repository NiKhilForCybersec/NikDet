Identity Security KQL Queries

// Admin role assignments
AuditLogs
| where OperationName has "Add member to role"
| extend Role = tostring(TargetResources[0].modifiedProperties[1].newValue)
| extend Target = tostring(TargetResources[0].userPrincipalName)
| where Role has_any ("Global Administrator", "Privileged Role", "Security Admin")
| project TimeGenerated, InitiatedBy.user.userPrincipalName, Target, Role

// Conditional Access policy changes
AuditLogs
| where OperationName has_any ("Update conditional access policy", "Delete conditional access policy")
| project TimeGenerated, InitiatedBy.user.userPrincipalName, OperationName, TargetResources

// MFA registration changes
AuditLogs
| where OperationName has_any ("User registered security info", "User deleted security info")
| project TimeGenerated, InitiatedBy.user.userPrincipalName, OperationName

// PIM role activations
AuditLogs
| where OperationName == "Add member to role in PIM completed (permanent)"
   or OperationName == "Add eligible member to role in PIM completed"
| extend Role = tostring(TargetResources[0].displayName)
| project TimeGenerated, InitiatedBy.user.userPrincipalName, Role
