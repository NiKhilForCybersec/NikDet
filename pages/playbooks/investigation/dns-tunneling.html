Investigation Steps

Step 1: Identify Anomalous DNS Activity

KQL - High Volume DNS to Single Domain

// Find hosts with unusually high DNS queries to single domain
DnsEvents
| where TimeGenerated > ago(24h)
| extend Domain = tostring(split(Name, ".")[-2]) + "." + tostring(split(Name, ".")[-1])
| summarize 
    QueryCount = count(),
    UniqueSubdomains = dcount(Name),
    Hosts = make_set(ClientIP, 10)
    by Domain
| where QueryCount > 500 and UniqueSubdomains > 100
| order by QueryCount desc

KQL - Long Subdomain Detection

// Detect DNS queries with unusually long subdomains (tunneling indicator)
DnsEvents
| where TimeGenerated > ago(24h)
| extend SubdomainLength = strlen(tostring(split(Name, ".")[0]))
| where SubdomainLength > 40
| summarize 
    QueryCount = count(),
    AvgLength = avg(SubdomainLength),
    SampleQueries = make_set(Name, 5)
    by ClientIP
| where QueryCount > 10
| order by AvgLength desc

KQL - High Entropy DNS Queries

// Detect high-entropy subdomains (encoded data)
let EntropyCalc = (s:string) {
    let chars = split(s, "");
    let freq = toscalar(chars | summarize count() by tostring(chars) | summarize make_bag(pack(tostring(chars), count_)));
    // Simplified entropy indicator based on unique char ratio
    todouble(array_length(chars)) / bag_keys(freq) 
};
DnsEvents
| where TimeGenerated > ago(24h)
| extend Subdomain = tostring(split(Name, ".")[0])
| where strlen(Subdomain) > 20
| extend UniqueChars = array_length(make_set(split(Subdomain, "")))
| extend EntropyRatio = todouble(UniqueChars) / strlen(Subdomain)
| where EntropyRatio > 0.6  // High character diversity
| summarize Count = count(), Samples = make_set(Name, 10) by ClientIP
| where Count > 5

KQL - TXT Record Query Anomaly

// Detect unusual TXT record queries (often used for tunneling)
DnsEvents
| where TimeGenerated > ago(24h)
| where QueryType == "TXT"
| summarize 
    TXTQueries = count(),
    UniqueDomains = dcount(Name),
    Domains = make_set(Name, 20)
    by ClientIP
| where TXTQueries > 100
| order by TXTQueries desc

SPL - DNS Tunneling Detection (Splunk)

index=dns sourcetype=dns
| eval subdomain_length=len(mvindex(split(query, "."), 0))
| where subdomain_length > 40
| stats count as QueryCount dc(query) as UniqueQueries by src_ip, query_type
| where QueryCount > 50
| sort -QueryCount

Step 2: Analyze Specific Domain

KQL - Deep Dive on Suspicious Domain

// Detailed analysis of suspicious domain
let SuspiciousDomain = "suspicious-domain.com";
DnsEvents
| where TimeGenerated > ago(7d)
| where Name endswith SuspiciousDomain
| summarize 
    TotalQueries = count(),
    UniqueSubdomains = dcount(Name),
    QueryTypes = make_set(QueryType),
    SourceHosts = make_set(ClientIP),
    FirstSeen = min(TimeGenerated),
    LastSeen = max(TimeGenerated),
    SampleQueries = make_set(Name, 20)
| extend QueryRate = TotalQueries / datetime_diff('hour', LastSeen, FirstSeen)

KQL - Extract Potential Encoded Data

// Extract and analyze subdomain content
let SuspiciousDomain = "evil.com";
DnsEvents
| where TimeGenerated > ago(24h)
| where Name endswith SuspiciousDomain
| extend Subdomain = replace_string(Name, strcat(".", SuspiciousDomain), "")
| extend SubdomainLength = strlen(Subdomain)
| extend PossibleBase64 = Subdomain matches regex "^[A-Za-z0-9+/=]+$"
| extend PossibleHex = Subdomain matches regex "^[0-9a-fA-F]+$"
| project TimeGenerated, ClientIP, Subdomain, SubdomainLength, PossibleBase64, PossibleHex
| order by TimeGenerated asc

Step 3: Check for DGA (Domain Generation Algorithm)

KQL - DGA Detection via NXDOMAIN

// Detect DGA by high NXDOMAIN rate
DnsEvents
| where TimeGenerated > ago(24h)
| summarize 
    TotalQueries = count(),
    NXDomainCount = countif(ResponseCode == "NXDOMAIN"),
    UniqueDomains = dcount(Name)
    by ClientIP
| extend NXDomainRatio = round(100.0 * NXDomainCount / TotalQueries, 2)
| where NXDomainRatio > 30 and NXDomainCount > 100
| order by NXDomainRatio desc

KQL - Random-Looking Domain Detection

// Find queries to random-looking domains (potential DGA)
DnsEvents
| where TimeGenerated > ago(24h)
| where ResponseCode == "NXDOMAIN"
| extend Domain = tostring(split(Name, ".")[-2])
| where strlen(Domain) > 10
// Check for low vowel ratio (DGA indicator)
| extend VowelCount = countof(tolower(Domain), "[aeiou]")
| extend VowelRatio = todouble(VowelCount) / strlen(Domain)
| where VowelRatio < 0.2  // Very few vowels
| summarize Count = count(), Samples = make_set(Name, 20) by ClientIP
| where Count > 20

Step 4: Endpoint Investigation

KQL - Find Process Making DNS Queries

// Correlate DNS with process on endpoint
let SuspiciousClients = DnsEvents
| where TimeGenerated > ago(24h)
| where Name endswith "suspicious-domain.com"
| distinct ClientIP;
DeviceNetworkEvents
| where TimeGenerated > ago(24h)
| where RemotePort == 53
| where LocalIP in (SuspiciousClients)
| summarize 
    QueryCount = count(),
    Destinations = make_set(RemoteIP)
    by DeviceName, InitiatingProcessFileName, InitiatingProcessCommandLine
| order by QueryCount desc

KQL - DNS Tunneling Tool Detection

// Detect known DNS tunneling tools
DeviceProcessEvents
| where TimeGenerated > ago(7d)
| where FileName in~ ("iodine.exe", "iodine", "dnscat2", "dnscat", 
    "dns2tcp", "dnscat2.rb")
    or ProcessCommandLine has_any ("iodine", "dnscat", "dns2tcp", 
        "dnstunnel", "dnsexfil")
| project TimeGenerated, DeviceName, FileName, ProcessCommandLine, 
    AccountName, InitiatingProcessFileName

DNS Sinkhole Configuration

# Windows DNS Server - Add sinkhole zone
Add-DnsServerPrimaryZone -Name "malicious-domain.com" -ZoneFile "malicious-domain.com.dns"
Add-DnsServerResourceRecordA -ZoneName "malicious-domain.com" -Name "@" -IPv4Address "127.0.0.1"
Add-DnsServerResourceRecordA -ZoneName "malicious-domain.com" -Name "*" -IPv4Address "127.0.0.1"

# Or using hosts file for immediate block
echo "127.0.0.1 malicious-domain.com" >> C:\Windows\System32\drivers\etc\hosts

Firewall Block for DNS to External

# Block direct DNS (force use of internal resolver)
# Allow DNS only to internal DNS servers

# Linux iptables
iptables -A OUTPUT -p udp --dport 53 -d 10.0.0.53 -j ACCEPT  # Internal DNS
iptables -A OUTPUT -p udp --dport 53 -j DROP  # Block all other DNS

# Windows Firewall
New-NetFirewallRule -DisplayName "Block External DNS" `
    -Direction Outbound -Protocol UDP -RemotePort 53 `
    -RemoteAddress "!10.0.0.53" -Action Block
