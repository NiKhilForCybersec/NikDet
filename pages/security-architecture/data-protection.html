DLP Monitoring Queries

// KQL - DLP policy matches by user
DLPEvent
| where TimeGenerated > ago(24h)
| summarize Matches = count(), Policies = make_set(PolicyName) by UserPrincipalName
| where Matches > 5
| order by Matches desc

// KQL - DLP blocks and overrides
DLPEvent
| where TimeGenerated > ago(7d)
| where ActionTaken in ("Block", "Override")
| summarize 
    Blocks = countif(ActionTaken == "Block"),
    Overrides = countif(ActionTaken == "Override")
    by PolicyName

// KQL - Sensitive data to external
DLPEvent
| where TimeGenerated > ago(24h)
| where DestinationType == "External"
| where SensitiveInfoTypes has_any ("Credit Card", "SSN", "PII")
| project TimeGenerated, UserPrincipalName, FileName, SensitiveInfoTypes, Destination

// KQL - Endpoint DLP - USB activity
DeviceEvents
| where ActionType == "RemovableStorageFileEvent"
| where AdditionalFields has "Sensitive"
| summarize FileCount = count(), TotalSize = sum(tolong(AdditionalFields.FileSize)) by AccountName, DeviceName

// SPL - DLP violations summary
index=dlp action=block OR action=warn
| stats count by user, policy, action, destination_type
| where count > 3
