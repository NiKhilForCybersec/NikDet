Discovery Queries

// KQL - Discover devices from network logs
CommonSecurityLog
| where TimeGenerated > ago(7d)
| summarize FirstSeen=min(TimeGenerated), 
    LastSeen=max(TimeGenerated),
    Connections=count() by SourceIP
| where SourceIP startswith "10." or SourceIP startswith "192.168."
| order by Connections desc

// KQL - Compare EDR inventory to AD computers
let EDRDevices = DeviceInfo 
    | where TimeGenerated > ago(1d)
    | summarize by DeviceName=tolower(DeviceName);
let ADComputers = IdentityInfo 
    | where Type == "Computer" 
    | summarize by AccountName=tolower(AccountName);
ADComputers
| join kind=leftanti EDRDevices on $left.AccountName == $right.DeviceName
| project MissingFromEDR = AccountName

// KQL - Find devices not sending logs recently
let ActiveDevices = Heartbeat 
    | where TimeGenerated > ago(24h) 
    | distinct Computer;
let AllDevices = Heartbeat 
    | where TimeGenerated > ago(30d) 
    | distinct Computer;
AllDevices 
| join kind=leftanti ActiveDevices on Computer
| project SilentDevice = Computer

// SPL - Asset discovery from firewall logs
index=firewall earliest=-7d
| stats earliest(_time) as first_seen, latest(_time) as last_seen, 
    count as connections by src_ip
| where cidrmatch("10.0.0.0/8", src_ip) OR cidrmatch("192.168.0.0/16", src_ip)
| sort - connections

Coverage Gap Detection

// Find servers without EDR agent
let AllServers = _GetWatchlist('ServerInventory') | project Hostname;
let EDRCovered = DeviceInfo 
    | where TimeGenerated > ago(1d)
    | where DeviceType == "Server"
    | distinct DeviceName;
AllServers
| join kind=leftanti EDRCovered on $left.Hostname == $right.DeviceName

// Find endpoints not logging to SIEM
let ExpectedEndpoints = SigninLogs 
    | where TimeGenerated > ago(7d)
    | distinct DeviceDetail.deviceId;
let LoggingEndpoints = DeviceEvents
    | where TimeGenerated > ago(1d)
    | distinct DeviceId;
ExpectedEndpoints
| join kind=leftanti LoggingEndpoints on $left.DeviceDetail_deviceId == $right.DeviceId
