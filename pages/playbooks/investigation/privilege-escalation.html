INV-006

Privilege Escalation Investigation

Comprehensive investigation procedure for suspected privilege escalation attacks. Covers UAC bypass, token manipulation, exploitation, credential theft, group changes, service abuse, and Kerberos attacks.

SeverityHigh
MITRE TacticTA0004
Time1-2 hours
T1548 T1134 T1068 T1078 T1053 T1543 T1558
Attack Progression Indicator

Privilege escalation typically follows initial access and indicates an attacker is attempting to gain higher-level permissions. This is a critical attack stage - successful escalation enables lateral movement, persistence, data access, and domain compromise.

2.2 UAC Bypass Detection Queries

// KQL - Registry-based UAC bypass detection
DeviceRegistryEvents
| where TimeGenerated > ago(24h)
| where RegistryKey has_any (
    "ms-settings\\shell\\open\\command",
    "mscfile\\shell\\open\\command",
    "Classes\\Folder\\shell\\open\\command",
    "Classes\\exefile\\shell\\open\\command",
    "Software\\Classes\\AppX"
)
| project TimeGenerated, DeviceName, RegistryKey, 
    RegistryValueName, RegistryValueData,
    InitiatingProcessFileName, InitiatingProcessAccountName
| order by TimeGenerated desc

// KQL - Auto-elevate binary spawning shell
let autoElevateBinaries = dynamic([
    "fodhelper.exe", "eventvwr.exe", "computerdefaults.exe",
    "sdclt.exe", "slui.exe", "wsreset.exe", "cmstp.exe",
    "mmc.exe", "pkgmgr.exe", "spinstall.exe", "sysprep.exe"
]);
DeviceProcessEvents
| where TimeGenerated > ago(24h)
| where InitiatingProcessFileName in~ (autoElevateBinaries)
| where FileName in~ ("cmd.exe", "powershell.exe", "pwsh.exe",
    "wscript.exe", "cscript.exe", "mshta.exe")
| project TimeGenerated, DeviceName, 
    ParentProcess = InitiatingProcessFileName,
    ChildProcess = FileName,
    CommandLine = ProcessCommandLine,
    User = AccountName

// KQL - CMSTP UAC bypass
DeviceProcessEvents
| where FileName =~ "cmstp.exe"
| where ProcessCommandLine has_any ("/ni", "/au", "/s", ".inf")
| project TimeGenerated, DeviceName, ProcessCommandLine, 
    AccountName, InitiatingProcessFileName

// KQL - Environment variable UAC bypass
DeviceRegistryEvents
| where RegistryKey has "Environment"
| where RegistryValueName in ("windir", "systemroot", "comspec")
| project TimeGenerated, DeviceName, RegistryValueName, 
    RegistryValueData, InitiatingProcessFileName

// SPL - UAC bypass via registry
index=sysmon EventCode=13
| search TargetObject="*\\shell\\open\\command*"
| table _time, Computer, User, TargetObject, Details, Image

3.2 Token Manipulation Detection Queries

// KQL - Sensitive privilege use
SecurityEvent
| where TimeGenerated > ago(24h)
| where EventID == 4673
| extend PrivList = tostring(PrivilegeList)
| where PrivList has_any (
    "SeDebugPrivilege", "SeTcbPrivilege", 
    "SeAssignPrimaryTokenPrivilege", "SeImpersonatePrivilege",
    "SeLoadDriverPrivilege", "SeBackupPrivilege",
    "SeRestorePrivilege", "SeTakeOwnershipPrivilege"
)
| project TimeGenerated, Computer, SubjectUserName, 
    ProcessName, PrivilegeList
| summarize Count = count(), 
    Processes = make_set(ProcessName, 5) 
    by Computer, SubjectUserName, PrivilegeList

// KQL - Token manipulation tool signatures
DeviceProcessEvents
| where TimeGenerated > ago(24h)
| where ProcessCommandLine has_any (
    "token::elevate", "token::duplicate", "token::revert",
    "Invoke-TokenManipulation", "Get-System", "getsystem",
    "incognito", "impersonate_token", "steal_token",
    "ImpersonateLoggedOnUser", "DuplicateTokenEx"
)
| project TimeGenerated, DeviceName, FileName, 
    ProcessCommandLine, AccountName

// KQL - Potato exploit detection
DeviceProcessEvents
| where TimeGenerated > ago(24h)
| where FileName has_any (
    "juicypotato", "sweetpotato", "godpotato", "roguepotato",
    "hotpotato", "rottenpotato", "printspoofer", "efspotato"
) or ProcessCommandLine has_any (
    "JuicyPotato", "SweetPotato", "GodPotato", "PrintSpoofer"
)
| project TimeGenerated, DeviceName, FileName, 
    ProcessCommandLine, AccountName

// KQL - Named pipe impersonation
DeviceEvents
| where ActionType == "NamedPipeEvent"
| where AdditionalFields has "Created"
| where PipeName !startswith "\\Device\\NamedPipe\\MSSE"
    and PipeName !startswith "\\Device\\NamedPipe\\wkssvc"
| project TimeGenerated, DeviceName, PipeName, 
    InitiatingProcessFileName, InitiatingProcessAccountName

// KQL - Process running as SYSTEM spawned by user
DeviceProcessEvents
| where TimeGenerated > ago(24h)
| where InitiatingProcessAccountName !in ("SYSTEM", 
    "LOCAL SERVICE", "NETWORK SERVICE")
| where AccountName == "SYSTEM"
| project TimeGenerated, DeviceName, FileName, 
    ProcessCommandLine, InitiatingProcessFileName,
    InitiatingProcessAccountName, AccountName

Step 4: Group Membership Changes (15 min)

// KQL - User added to privileged groups
let privilegedGroups = dynamic([
    "Domain Admins", "Enterprise Admins", "Schema Admins",
    "Administrators", "Account Operators", "Backup Operators",
    "Server Operators", "Print Operators", "DnsAdmins",
    "Group Policy Creator Owners"
]);
SecurityEvent
| where TimeGenerated > ago(7d)
| where EventID in (4728, 4732, 4756) // Member added to group
| where TargetUserName has_any (privilegedGroups)
| project TimeGenerated, Computer, 
    AddedBy = SubjectUserName,
    AddedUser = MemberName,
    Group = TargetUserName,
    EventID

// KQL - User rights assignment
SecurityEvent
| where TimeGenerated > ago(7d)
| where EventID == 4704 // User right assigned
| project TimeGenerated, Computer, SubjectUserName, 
    TargetUserName, PrivilegeList

// KQL - AdminSDHolder protected groups
SecurityEvent
| where EventID in (4728, 4732, 4756)
| where TargetUserName has_any (
    "Domain Admins", "Enterprise Admins", "Administrators",
    "Schema Admins", "Account Operators", "Backup Operators"
)
| project TimeGenerated, AddedBy = SubjectUserName,
    AddedUser = MemberName, Group = TargetUserName

// KQL - net commands adding to groups
DeviceProcessEvents
| where FileName in~ ("net.exe", "net1.exe")
| where ProcessCommandLine has "localgroup" 
    and ProcessCommandLine has "/add"
| project TimeGenerated, DeviceName, ProcessCommandLine, AccountName

Step 5: Service and Scheduled Task Abuse (20 min)

// KQL - Service binary path modification
DeviceRegistryEvents
| where TimeGenerated > ago(24h)
| where RegistryKey has "\\Services\\" 
    and RegistryValueName == "ImagePath"
| where RegistryValueData has_any (
    "cmd", "powershell", "\\Temp\\", "\\Users\\", 
    "\\AppData\\", "\\Downloads\\", "nc.exe", "nc64.exe"
)
| project TimeGenerated, DeviceName, 
    ServiceKey = RegistryKey, NewPath = RegistryValueData,
    ModifiedBy = InitiatingProcessFileName

// KQL - sc.exe service configuration changes
DeviceProcessEvents
| where FileName =~ "sc.exe"
| where ProcessCommandLine has "config" 
    and ProcessCommandLine has_any ("binpath", "obj=", "start=")
| project TimeGenerated, DeviceName, ProcessCommandLine, AccountName

// KQL - New service with suspicious path
SecurityEvent
| where EventID == 7045 // Service installed
| where ServiceFileName has_any (
    "cmd", "powershell", "\\Temp\\", "\\Users\\", 
    "\\AppData\\", "-enc", "-e ", "IEX"
)
| project TimeGenerated, Computer, ServiceName, 
    ServiceFileName, ServiceStartType, SubjectUserName

// KQL - Scheduled task running as SYSTEM by non-admin
SecurityEvent
| where EventID == 4698 // Task created
| where TaskContent has "SYSTEM" or TaskContent has "S-1-5-18"
| where SubjectUserName !in ("SYSTEM", "Administrator")
| project TimeGenerated, Computer, SubjectUserName, TaskName,
    TaskContent

// KQL - schtasks.exe creating privileged task
DeviceProcessEvents
| where FileName =~ "schtasks.exe"
| where ProcessCommandLine has "/create"
| where ProcessCommandLine has_any ("/ru SYSTEM", 
    "/ru \"NT AUTHORITY\\SYSTEM\"", "/rl highest")
| project TimeGenerated, DeviceName, ProcessCommandLine, AccountName

Step 6: Exploitation Detection (15 min)

// Common privesc exploits reference:
// PrintNightmare - CVE-2021-34527 (Print Spooler)
// ZeroLogon - CVE-2020-1472 (Netlogon)
// PetitPotam - CVE-2021-36942 (LSARPC)
// HiveNightmare - CVE-2021-36934 (SAM/SYSTEM hives)
// EternalBlue - CVE-2017-0144 (SMBv1)

// KQL - PrintNightmare detection
DeviceProcessEvents
| where InitiatingProcessFileName =~ "spoolsv.exe"
| where FileName in~ ("cmd.exe", "powershell.exe", 
    "rundll32.exe", "regsvr32.exe")
| project TimeGenerated, DeviceName, FileName, 
    ProcessCommandLine, AccountName

// KQL - Suspicious DLL load by spooler
DeviceImageLoadEvents
| where InitiatingProcessFileName =~ "spoolsv.exe"
| where FolderPath startswith "\\\\" // UNC path
| project TimeGenerated, DeviceName, FileName, FolderPath

// KQL - Known exploit tool detection
DeviceProcessEvents
| where FileName has_any (
    "printnightmare", "zerologon", "petitpotam",
    "CVE-", "exploit", "poc", "mimikatz", "rubeus"
)
| project TimeGenerated, DeviceName, FileName, 
    ProcessCommandLine, AccountName

// KQL - Kernel driver load (potential exploit)
DeviceEvents
| where ActionType == "DriverLoad"
| where FolderPath has_any ("\\Temp\\", "\\Users\\", 
    "\\AppData\\", "\\Downloads\\")
| project TimeGenerated, DeviceName, FileName, 
    FolderPath, SHA256

Escalation Criteria

Escalate Immediately If:
  • Domain Admin or Enterprise Admin privileges gained
  • Domain Controller compromised
  • Multiple systems show privilege escalation activity
  • Known APT techniques observed
  • Service accounts with domain-wide access compromised
  • Evidence of Kerberos ticket manipulation (Golden/Silver ticket)
  • AdminSDHolder or KRBTGT account modified
