Endpoint Security Queries

// KQL - Devices without EDR sensor
DeviceInfo
| where Timestamp > ago(7d)
| summarize arg_max(Timestamp, *) by DeviceId
| where OnboardingStatus != "Onboarded"
| project DeviceName, OSPlatform, OnboardingStatus

// KQL - Unencrypted devices
DeviceTvmSecureConfigurationAssessment
| where ConfigurationId == "scid-2000" // BitLocker
| where IsCompliant == 0
| project DeviceName, ConfigurationId, IsCompliant

// KQL - Devices missing critical patches
DeviceTvmSoftwareVulnerabilities
| where VulnerabilitySeverityLevel == "Critical"
| summarize CriticalVulns = count() by DeviceName
| where CriticalVulns > 0
| order by CriticalVulns desc

// KQL - ASR rule triggers
DeviceEvents
| where ActionType startswith "Asr"
| summarize Count = count() by ActionType, DeviceName
| order by Count desc
