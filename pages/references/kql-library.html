KQL Query Library

Comprehensive collection of KQL queries for Microsoft Sentinel, Defender for Endpoint, and Azure AD. Copy-paste ready for SOC operations.

Authentication & Identity

// Failed logins - last 24 hours
SigninLogs
| where TimeGenerated > ago(24h)
| where ResultType != "0"
| summarize FailCount = count() by UserPrincipalName, IPAddress, ResultDescription
| where FailCount > 5
| order by FailCount desc

// Brute force detection
SigninLogs
| where TimeGenerated > ago(1h)
| where ResultType != "0"
| summarize Failures = count(), IPs = dcount(IPAddress) by UserPrincipalName
| where Failures > 10

// Password spray detection
SigninLogs
| where TimeGenerated > ago(1h)
| where ResultType in ("50126", "50053")
| summarize Users = dcount(UserPrincipalName), Attempts = count() by IPAddress
| where Users > 5

// Successful login after failures
let failures = SigninLogs | where ResultType != "0" | summarize FailCount = count() by UserPrincipalName, IPAddress;
SigninLogs
| where ResultType == "0"
| join kind=inner failures on UserPrincipalName, IPAddress
| where FailCount > 3

// Impossible travel
SigninLogs
| where ResultType == "0"
| project TimeGenerated, UserPrincipalName, IPAddress, Location
| order by UserPrincipalName, TimeGenerated
| extend PrevTime = prev(TimeGenerated), PrevLocation = prev(Location), PrevUser = prev(UserPrincipalName)
| where UserPrincipalName == PrevUser
| extend TimeDiff = datetime_diff('hour', TimeGenerated, PrevTime)
| where TimeDiff < 2 and Location != PrevLocation

// MFA fatigue attack
SigninLogs
| where ResultType in ("50074", "50076")
| summarize MFAAttempts = count() by UserPrincipalName, bin(TimeGenerated, 1h)
| where MFAAttempts > 5

// Legacy authentication
SigninLogs
| where ClientAppUsed in ("Exchange ActiveSync", "IMAP4", "POP3", "SMTP", "Other clients")
| summarize count() by UserPrincipalName, ClientAppUsed, ResultType

// Risky sign-ins
SigninLogs
| where RiskLevelDuringSignIn in ("high", "medium")
| project TimeGenerated, UserPrincipalName, IPAddress, RiskLevelDuringSignIn, RiskDetail

// New country login
let userCountries = SigninLogs | where TimeGenerated > ago(30d) | summarize Countries = make_set(Location) by UserPrincipalName;
SigninLogs
| where TimeGenerated > ago(24h)
| join kind=leftouter userCountries on UserPrincipalName
| where Location !in (Countries)

// Service principal sign-ins
AADServicePrincipalSignInLogs
| where TimeGenerated > ago(24h)
| summarize count() by ServicePrincipalName, IPAddress, ResultType

Privileged Account Activity

// Admin role assignments
AuditLogs
| where OperationName has "Add member to role"
| extend TargetUser = tostring(TargetResources[0].userPrincipalName)
| extend Role = tostring(TargetResources[0].modifiedProperties[1].newValue)
| project TimeGenerated, InitiatedBy.user.userPrincipalName, TargetUser, Role

// Global Admin activity
SigninLogs
| where TimeGenerated > ago(24h)
| join kind=inner (
    IdentityInfo | where AssignedRoles has "Global Administrator"
) on $left.UserPrincipalName == $right.AccountUPN
| project TimeGenerated, UserPrincipalName, IPAddress, AppDisplayName

// Privileged group changes
AuditLogs
| where OperationName in ("Add member to group", "Remove member from group")
| where TargetResources[0].displayName has_any ("Admin", "Privileged", "Executive")
| project TimeGenerated, Operation = OperationName, Group = TargetResources[0].displayName, Actor = InitiatedBy.user.userPrincipalName

// New user with admin role
let newUsers = AuditLogs | where OperationName == "Add user" | extend UPN = tostring(TargetResources[0].userPrincipalName) | project UPN, Created = TimeGenerated;
AuditLogs
| where OperationName has "Add member to role"
| extend UPN = tostring(TargetResources[0].userPrincipalName)
| join kind=inner newUsers on UPN
| where TimeGenerated - Created < 1h

// Password reset by admin
AuditLogs
| where OperationName == "Reset password (by admin)"
| extend Target = tostring(TargetResources[0].userPrincipalName)
| extend Admin = tostring(InitiatedBy.user.userPrincipalName)
| project TimeGenerated, Admin, Target

Endpoint Security (Defender)

// Process creation with command line
DeviceProcessEvents
| where TimeGenerated > ago(24h)
| where FileName in~ ("powershell.exe", "cmd.exe", "wscript.exe", "cscript.exe")
| project TimeGenerated, DeviceName, FileName, ProcessCommandLine, AccountName

// Encoded PowerShell
DeviceProcessEvents
| where FileName =~ "powershell.exe"
| where ProcessCommandLine has_any ("-enc", "-encoded", "-e ", "frombase64")
| project TimeGenerated, DeviceName, ProcessCommandLine

// LOLBAS execution
DeviceProcessEvents
| where FileName in~ ("certutil.exe", "mshta.exe", "regsvr32.exe", "rundll32.exe", "msiexec.exe", "bitsadmin.exe")
| where ProcessCommandLine has_any ("http", "ftp", "/i:", "-decode", "javascript")
| project TimeGenerated, DeviceName, FileName, ProcessCommandLine

// Suspicious parent-child process
DeviceProcessEvents
| where InitiatingProcessFileName in~ ("outlook.exe", "winword.exe", "excel.exe", "powerpnt.exe")
| where FileName in~ ("powershell.exe", "cmd.exe", "wscript.exe", "mshta.exe")
| project TimeGenerated, DeviceName, Parent = InitiatingProcessFileName, Child = FileName, CommandLine = ProcessCommandLine

// Credential dumping indicators
DeviceProcessEvents
| where FileName in~ ("mimikatz.exe", "procdump.exe", "dumpert.exe")
   or ProcessCommandLine has_any ("sekurlsa", "lsass", "comsvcs.dll", "MiniDump")
| project TimeGenerated, DeviceName, FileName, ProcessCommandLine

// Scheduled task creation
DeviceProcessEvents
| where FileName =~ "schtasks.exe"
| where ProcessCommandLine has "/create"
| project TimeGenerated, DeviceName, ProcessCommandLine, AccountName

// Service installation
DeviceProcessEvents
| where FileName =~ "sc.exe"
| where ProcessCommandLine has_any ("create", "config")
| project TimeGenerated, DeviceName, ProcessCommandLine

// Registry persistence
DeviceRegistryEvents
| where RegistryKey has_any ("Run", "RunOnce", "Winlogon")
| where ActionType == "RegistryValueSet"
| project TimeGenerated, DeviceName, RegistryKey, RegistryValueName, RegistryValueData

// Malware detections
DeviceEvents
| where ActionType == "AntivirusDetection"
| project TimeGenerated, DeviceName, FileName, ThreatName = AdditionalFields

// USB device connections
DeviceEvents
| where ActionType == "PnpDeviceConnected"
| where DeviceDescription has_any ("USB", "Removable")
| project TimeGenerated, DeviceName, DeviceDescription, DeviceId

Network & C2 Detection

// Outbound connections to rare destinations
DeviceNetworkEvents
| where TimeGenerated > ago(24h)
| where RemoteIPType == "Public"
| summarize ConnectionCount = count(), Devices = dcount(DeviceName) by RemoteIP
| where Devices == 1 and ConnectionCount > 100

// Beaconing detection (regular intervals)
DeviceNetworkEvents
| where TimeGenerated > ago(24h)
| where RemoteIPType == "Public"
| summarize Timestamps = make_list(TimeGenerated) by DeviceName, RemoteIP
| extend Intervals = array_sort_asc(Timestamps)
| where array_length(Intervals) > 10

// High data transfer
DeviceNetworkEvents
| where TimeGenerated > ago(24h)
| where RemoteIPType == "Public"
| summarize BytesSent = sum(SentBytes) by DeviceName, RemoteIP
| where BytesSent > 104857600  // 100 MB
| extend SentMB = round(BytesSent / 1048576.0, 2)

// DNS to known bad domains
DeviceNetworkEvents
| where TimeGenerated > ago(24h)
| where RemoteUrl has_any ("pastebin", "ngrok", "duckdns", "no-ip")
| project TimeGenerated, DeviceName, RemoteUrl

// Lateral movement ports
DeviceNetworkEvents
| where RemotePort in (445, 135, 139, 3389, 22, 5985, 5986)
| where LocalIP startswith "10." and RemoteIP startswith "10."
| summarize Connections = count() by DeviceName, RemoteIP, RemotePort
| where Connections > 5

// TOR exit node connections
let torExits = externaldata(ip:string) [@"https://check.torproject.org/exit-addresses"] with (format="txt");
DeviceNetworkEvents
| where RemoteIP in (torExits)
| project TimeGenerated, DeviceName, RemoteIP

Email Security

// Phishing emails delivered
EmailEvents
| where TimeGenerated > ago(24h)
| where ThreatTypes has "Phish"
| where DeliveryAction == "Delivered"
| project TimeGenerated, SenderFromAddress, RecipientEmailAddress, Subject

// Malware attachments
EmailAttachmentInfo
| where TimeGenerated > ago(24h)
| where ThreatTypes has "Malware"
| join kind=inner EmailEvents on NetworkMessageId
| project TimeGenerated, Sender = SenderFromAddress, Recipient = RecipientEmailAddress, FileName, ThreatNames

// URL clicks
UrlClickEvents
| where TimeGenerated > ago(24h)
| where IsClickedThrough == true
| project TimeGenerated, AccountUpn, Url, ActionType

// Inbox rules created
CloudAppEvents
| where ActionType == "New-InboxRule"
| where RawEventData has_any ("ForwardTo", "RedirectTo", "DeleteMessage")
| project TimeGenerated, AccountDisplayName, RawEventData

// Email forwarding to external
CloudAppEvents
| where ActionType == "Set-Mailbox"
| where RawEventData has "ForwardingSmtpAddress"
| project TimeGenerated, AccountDisplayName, RawEventData

// OAuth app consent
AuditLogs
| where OperationName == "Consent to application"
| extend AppName = tostring(TargetResources[0].displayName)
| extend User = tostring(InitiatedBy.user.userPrincipalName)
| project TimeGenerated, User, AppName

Cloud Security (Azure)

// Azure activity - resource creation
AzureActivity
| where OperationNameValue has "MICROSOFT.COMPUTE/VIRTUALMACHINES/WRITE"
| where ActivityStatusValue == "Success"
| project TimeGenerated, Caller, ResourceGroup, Resource

// Key Vault access
AzureDiagnostics
| where ResourceType == "VAULTS"
| where OperationName in ("SecretGet", "SecretList", "KeyGet")
| project TimeGenerated, CallerIPAddress, identity_claim_upn_s, OperationName

// Storage account changes
AzureActivity
| where OperationNameValue has "MICROSOFT.STORAGE"
| where ActivityStatusValue == "Success"
| project TimeGenerated, Caller, OperationNameValue, ResourceGroup

// Network security group changes
AzureActivity
| where OperationNameValue has "MICROSOFT.NETWORK/NETWORKSECURITYGROUPS"
| project TimeGenerated, Caller, OperationNameValue, Properties

// Subscription-level changes
AzureActivity
| where CategoryValue == "Administrative"
| where OperationNameValue has_any ("roleAssignments", "policyAssignments")
| project TimeGenerated, Caller, OperationNameValue

Threat Hunting

// Rare process names
DeviceProcessEvents
| where TimeGenerated > ago(7d)
| summarize ExecutionCount = count(), Devices = dcount(DeviceName) by FileName
| where ExecutionCount < 5 and Devices == 1

// Process from unusual path
DeviceProcessEvents
| where FolderPath !startswith "C:\\Windows" and FolderPath !startswith "C:\\Program Files"
| where FolderPath has_any ("\\Temp\\", "\\AppData\\", "\\Downloads\\", "\\Public\\")
| project TimeGenerated, DeviceName, FileName, FolderPath

// Unsigned binaries
DeviceProcessEvents
| where TimeGenerated > ago(24h)
| where isempty(ProcessVersionInfoCompanyName)
| summarize count() by FileName, FolderPath
| where count_ < 5

// First-time execution
let baseline = DeviceProcessEvents | where TimeGenerated between (ago(30d) .. ago(1d)) | distinct FileName, DeviceName;
DeviceProcessEvents
| where TimeGenerated > ago(1d)
| join kind=leftanti baseline on FileName, DeviceName
| summarize FirstSeen = min(TimeGenerated) by DeviceName, FileName, FolderPath

// WMI persistence
DeviceEvents
| where ActionType == "WmiBindingEvent"
| project TimeGenerated, DeviceName, AdditionalFields

// DLL side-loading candidates
DeviceImageLoadEvents
| where FolderPath !startswith "C:\\Windows\\System32"
| where FileName in~ ("version.dll", "cryptsp.dll", "dbghelp.dll")
| project TimeGenerated, DeviceName, FileName, FolderPath, InitiatingProcessFileName
