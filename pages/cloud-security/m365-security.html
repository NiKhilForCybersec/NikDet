Business Email Compromise (BEC) Detection

Inbox Rule Creation

// Suspicious inbox rules (common BEC tactic)
OfficeActivity
| where Operation in ("New-InboxRule", "Set-InboxRule")
| extend RuleParameters = tostring(Parameters)
| where RuleParameters has_any ("DeleteMessage", "MoveToDeletedItems", 
    "ForwardTo", "RedirectTo", "SoftDelete")
| project TimeGenerated, UserId, Operation, RuleParameters, ClientIP

Mail Forwarding to External

// External forwarding enabled
OfficeActivity
| where Operation == "Set-Mailbox"
| extend ForwardingAddress = tostring(parse_json(Parameters)[?Name == "ForwardingSmtpAddress"].Value)
| where isnotempty(ForwardingAddress)
| where ForwardingAddress !endswith "@yourdomain.com"
| project TimeGenerated, UserId, ForwardingAddress, ClientIP

Delegate Access Granted

// Mailbox permissions added
OfficeActivity
| where Operation in ("Add-MailboxPermission", "Add-RecipientPermission")
| project TimeGenerated, UserId, Operation, 
    parse_json(Parameters), ClientIP

Data Exfiltration Detection

Mass File Download

// Large number of file downloads
OfficeActivity
| where RecordType == "SharePointFileOperation"
| where Operation in ("FileDownloaded", "FileSyncDownloadedFull")
| summarize DownloadCount = count(), Files = make_set(SourceFileName) 
    by UserId, bin(TimeGenerated, 1h)
| where DownloadCount > 50
| project TimeGenerated, UserId, DownloadCount

External Sharing

// Files shared with external users
OfficeActivity
| where RecordType == "SharePointSharingOperation"
| where Operation in ("SharingInvitationCreated", "AddedToSecureLink", 
    "AnonymousLinkCreated")
| extend TargetUser = tostring(parse_json(TargetUserOrGroupName))
| where TargetUser !endswith "@yourdomain.com" or Operation == "AnonymousLinkCreated"
| project TimeGenerated, UserId, Operation, TargetUser, SourceFileName

Phishing Detection

Defender for Office 365 Alerts

// Phishing emails detected
EmailEvents
| where ThreatTypes has "Phish"
| project TimeGenerated, RecipientEmailAddress, SenderFromAddress,
    Subject, ThreatTypes, DeliveryAction, DetectionMethods

// User clicked phishing link
UrlClickEvents
| where IsClickedThrough == true
| where ThreatTypes has "Phish"
| project TimeGenerated, AccountUpn, Url, NetworkMessageId

Post-Delivery Actions

// Emails ZAP'd (Zero-hour Auto Purge)
EmailPostDeliveryEvents
| where Action == "ZAP"
| project TimeGenerated, RecipientEmailAddress, NetworkMessageId,
    ThreatTypes, ActionType

Investigation Queries

// All activity for compromised user
OfficeActivity
| where UserId == "compromised@company.com"
| where TimeGenerated > ago(7d)
| project TimeGenerated, Operation, ClientIP, RecordType
| sort by TimeGenerated desc

// Find emails from malicious sender
EmailEvents
| where SenderFromAddress == "attacker@phishing.com"
| project TimeGenerated, RecipientEmailAddress, Subject,
    DeliveryAction, ThreatTypes

// Find users who clicked malicious URL
UrlClickEvents
| where Url has "malicious.com"
| project TimeGenerated, AccountUpn, Url, IsClickedThrough
