Data Breach Incident Response

Advanced response procedures for confirmed or suspected data exfiltration incidents requiring regulatory notification assessment.

Phase 1: Initial Assessment (0-4 Hours)

Immediate Containment Decisions

ScenarioActionAuthorization Required
Active exfiltration in progressImmediate network isolation, block destination IPs/domainsSOC authority
Compromised account with data accessDisable account, revoke all sessions, reset credentialsSOC authority
Compromised system with sensitive dataNetwork isolation via EDR, preserve for forensicsSOC authority
Third-party/vendor compromiseRevoke API keys, disable integrations, contact vendorManagement approval
Cloud storage exposureRevoke public access, rotate keys, audit access logsSOC authority

Data Impact Assessment Queries

// KQL: Identify all files accessed by compromised account
let compromised_user = "user@company.com";
let incident_start = datetime(2025-01-01);
OfficeActivity
| where TimeGenerated > incident_start
| where UserId =~ compromised_user
| where Operation in ("FileAccessed", "FileDownloaded", "FileSyncDownloadedFull", "FileModified")
| extend FileName = tostring(parse_json(tostring(SourceFileExtension)))
| project TimeGenerated, Operation, SourceFileName, SourceRelativeUrl, ClientIP, UserAgent
| summarize AccessCount=count(), Operations=make_set(Operation), FirstAccess=min(TimeGenerated), LastAccess=max(TimeGenerated) by SourceFileName, SourceRelativeUrl
| order by AccessCount desc

// KQL: Quantify data volume exfiltrated
let compromised_user = "user@company.com";
let compromised_ip = "1.2.3.4";
let incident_start = datetime(2025-01-01);
union 
(OfficeActivity | where UserId =~ compromised_user | where TimeGenerated > incident_start | where Operation == "FileDownloaded" | extend bytes = toint(SourceFileSize)),
(DeviceFileEvents | where InitiatingProcessAccountName =~ compromised_user | where TimeGenerated > incident_start | where ActionType == "FileCreated" | extend bytes = FileSize)
| summarize TotalBytes=sum(bytes), FileCount=count() by bin(TimeGenerated, 1h)
| extend TotalMB = TotalBytes / 1048576
| order by TimeGenerated asc

// KQL: Identify sensitive file access patterns
let sensitive_paths = dynamic(["HR", "Finance", "Legal", "Confidential", "PII", "PHI", "Secret"]);
OfficeActivity
| where TimeGenerated > ago(30d)
| where Operation in ("FileAccessed", "FileDownloaded")
| where SourceRelativeUrl has_any (sensitive_paths) or SourceFileName has_any (sensitive_paths)
| summarize AccessCount=count(), UniqueFiles=dcount(SourceFileName) by UserId, bin(TimeGenerated, 1d)
| where AccessCount > 100 or UniqueFiles > 50

// SPL: Track all data movement from compromised system
index=proxy OR index=firewall src_ip="10.1.2.3" earliest=-7d
| stats sum(bytes_out) as total_bytes dc(dest_ip) as unique_destinations values(dest_ip) as destinations by src_ip, user
| eval total_GB = round(total_bytes/1024/1024/1024, 2)
| where total_GB > 1
| sort -total_GB

Evidence Collection Checklist

Network Evidence:

  • ☐ Firewall logs showing outbound data transfers
  • ☐ Proxy logs with full URLs and response sizes
  • ☐ DNS query logs for exfiltration domains
  • ☐ NetFlow/packet captures if available
  • ☐ VPN connection logs

Endpoint Evidence:

  • ☐ File access/modification timestamps (MACB)
  • ☐ USB device connection history
  • ☐ Browser download history
  • ☐ Email client sent items and attachments
  • ☐ Cloud sync client logs (OneDrive, Dropbox)
  • ☐ Print spooler logs
  • ☐ Clipboard history (if captured)

Cloud Evidence:

  • ☐ Cloud storage access logs
  • ☐ Sharing permission changes
  • ☐ External sharing links created
  • ☐ API access logs
  • ☐ OAuth application consents

Phase 4: Forensic Analysis

Timeline Reconstruction

// KQL: Build comprehensive breach timeline
let compromised_accounts = dynamic(["user1@company.com", "user2@company.com"]);
let compromised_ips = dynamic(["1.2.3.4", "5.6.7.8"]);
let incident_start = datetime(2025-01-01);
union
(SigninLogs | where UserPrincipalName in~ (compromised_accounts) | project TimeGenerated, EventType="Authentication", Details=strcat("Sign-in from ", IPAddress, " - ", ResultDescription), Source="AzureAD"),
(AuditLogs | where InitiatedBy.user.userPrincipalName in~ (compromised_accounts) | project TimeGenerated, EventType="AuditEvent", Details=OperationName, Source="AzureAD"),
(OfficeActivity | where UserId in~ (compromised_accounts) | project TimeGenerated, EventType="FileActivity", Details=strcat(Operation, ": ", SourceFileName), Source="M365"),
(DeviceProcessEvents | where InitiatingProcessAccountUpn in~ (compromised_accounts) | project TimeGenerated, EventType="ProcessExec", Details=strcat(FileName, " ", ProcessCommandLine), Source="MDE"),
(DeviceNetworkEvents | where InitiatingProcessAccountUpn in~ (compromised_accounts) | project TimeGenerated, EventType="NetworkConn", Details=strcat(RemoteIP, ":", RemotePort), Source="MDE"),
(DeviceFileEvents | where InitiatingProcessAccountUpn in~ (compromised_accounts) | project TimeGenerated, EventType="FileEvent", Details=strcat(ActionType, ": ", FileName), Source="MDE"),
(EmailEvents | where SenderFromAddress in~ (compromised_accounts) | project TimeGenerated, EventType="EmailSent", Details=strcat("To: ", RecipientEmailAddress, " Subj: ", Subject), Source="MDO")
| where TimeGenerated > incident_start
| sort by TimeGenerated asc
| project TimeGenerated, EventType, Details, Source

// KQL: Identify all data repositories accessed
let compromised_user = "user@company.com";
union
(OfficeActivity | where UserId =~ compromised_user | where Operation has_any ("File", "Folder", "List", "Site") | summarize Count=count() by Site=SourceRelativeUrl),
(AzureActivity | where Caller =~ compromised_user | summarize Count=count() by Resource=ResourceGroup),
(AuditLogs | where InitiatedBy.user.userPrincipalName =~ compromised_user | where TargetResources has "application" | summarize Count=count() by App=tostring(TargetResources[0].displayName))
| order by Count desc

Post-Incident Documentation

  • Incident timeline: Complete chronology from initial access to containment
  • Data impact assessment: Final determination of data types and volumes
  • Root cause analysis: How initial access was gained
  • Notification log: All notifications made, to whom, when, content
  • Evidence inventory: All preserved evidence with chain of custody
  • Lessons learned: What worked, what didn't, improvements needed
  • Cost accounting: Response costs, notification costs, remediation costs
