Investigation Steps

Step 1: Identify Scope and Type

First 10 Minutes

Questions to Answer:

  • Is this brute force (one account) or password spray (many accounts)?
  • How many accounts affected?
  • Any successful authentications?
  • Single source IP or distributed?
  • Targeting cloud (Entra ID) or on-prem (AD)?

KQL - Identify Attack Pattern (Microsoft Sentinel)

// Identify brute force vs password spray pattern
SigninLogs
| where TimeGenerated > ago(24h)
| where ResultType != 0  // Failed logins
| summarize 
    FailedAttempts = count(),
    DistinctUsers = dcount(UserPrincipalName),
    DistinctIPs = dcount(IPAddress),
    Users = make_set(UserPrincipalName, 100),
    Passwords = make_set(ResultDescription, 10)
    by IPAddress, bin(TimeGenerated, 1h)
| extend AttackType = case(
    DistinctUsers == 1 and FailedAttempts > 10, "Brute Force",
    DistinctUsers > 10 and FailedAttempts/DistinctUsers < 3, "Password Spray",
    DistinctUsers > 5, "Credential Stuffing",
    "Unknown"
)
| where FailedAttempts > 5
| order by FailedAttempts desc

KQL - Find Successful Login After Failed Attempts

// CRITICAL: Find if attack succeeded
let FailedLogins = SigninLogs
| where TimeGenerated > ago(24h)
| where ResultType != 0
| summarize FailedCount = count(), FirstFail = min(TimeGenerated) by UserPrincipalName, IPAddress;
let SuccessfulLogins = SigninLogs
| where TimeGenerated > ago(24h)
| where ResultType == 0
| project UserPrincipalName, IPAddress, SuccessTime = TimeGenerated, AppDisplayName, DeviceDetail;
FailedLogins
| join kind=inner SuccessfulLogins on UserPrincipalName
| where SuccessTime > FirstFail
| where SuccessTime < FirstFail + 1h  // Success within 1 hour of failures
| project UserPrincipalName, IPAddress, FailedCount, FirstFail, SuccessTime, AppDisplayName, DeviceDetail
| order by FailedCount desc

SPL - Brute Force Detection (Splunk)

index=windows sourcetype=WinEventLog:Security EventCode=4625
| stats count as FailedAttempts dc(TargetUserName) as UniqueUsers values(TargetUserName) as Users by src_ip
| where FailedAttempts > 10
| eval AttackType=if(UniqueUsers=1, "Brute Force", if(UniqueUsers>10, "Password Spray", "Mixed"))
| sort -FailedAttempts

SPL - Successful Login After Failures (Splunk)

index=windows sourcetype=WinEventLog:Security (EventCode=4625 OR EventCode=4624)
| stats count(eval(EventCode=4625)) as Failures count(eval(EventCode=4624)) as Successes 
    min(_time) as FirstEvent max(_time) as LastEvent by TargetUserName src_ip
| where Failures > 5 AND Successes > 0
| eval TimeDelta=LastEvent-FirstEvent
| where TimeDelta < 3600
| table TargetUserName src_ip Failures Successes TimeDelta

KQL - Source IP Intelligence

// Analyze source IPs
let SuspiciousIPs = SigninLogs
| where TimeGenerated > ago(24h)
| where ResultType != 0
| summarize FailCount = count() by IPAddress
| where FailCount > 10
| project IPAddress;
SigninLogs
| where TimeGenerated > ago(24h)
| where IPAddress in (SuspiciousIPs)
| summarize 
    TotalAttempts = count(),
    SuccessCount = countif(ResultType == 0),
    FailCount = countif(ResultType != 0),
    UniqueUsers = dcount(UserPrincipalName),
    Countries = make_set(Location),
    UserAgents = make_set(UserAgent, 5),
    Apps = make_set(AppDisplayName, 10)
    by IPAddress
| extend SuccessRate = round(100.0 * SuccessCount / TotalAttempts, 2)
| order by FailCount desc

Step 3: Account Impact Assessment

Determine Account Compromise

For each potentially compromised account, check:

  • Was there a successful login from the attack source?
  • Any logins from new locations/devices after the attack?
  • MFA changes or security info updates?
  • New OAuth app consents?
  • Email forwarding rules created?
  • Password changed by user or attacker?

KQL - Post-Compromise Activity Check

// Check for post-compromise activity on potentially compromised accounts
let CompromisedUsers = SigninLogs
| where TimeGenerated > ago(24h)
| where ResultType == 0  // Successful logins
| where IPAddress in ("ATTACKER_IP_HERE")  // Replace with suspicious IP
| distinct UserPrincipalName;
// Check for suspicious post-login activity
union
(
    AuditLogs
    | where TimeGenerated > ago(24h)
    | where InitiatedBy.user.userPrincipalName in (CompromisedUsers)
    | where OperationName in (
        "Update user", 
        "Add member to group",
        "Add app role assignment to service principal",
        "Consent to application",
        "Add OAuth2PermissionGrant"
    )
    | project TimeGenerated, Activity = OperationName, User = InitiatedBy.user.userPrincipalName, Details = TargetResources
),
(
    OfficeActivity
    | where TimeGenerated > ago(24h)
    | where UserId in (CompromisedUsers)
    | where Operation in ("Set-Mailbox", "New-InboxRule", "Set-InboxRule", "Add-MailboxPermission")
    | project TimeGenerated, Activity = Operation, User = UserId, Details = Parameters
)
| order by TimeGenerated desc

Step 4: Timeline Construction

KQL - Complete Attack Timeline

// Build complete timeline for investigation
let InvestigationUser = "user@domain.com";  // Replace with target user
let TimeWindow = ago(7d);
union
(
    SigninLogs
    | where TimeGenerated > TimeWindow
    | where UserPrincipalName == InvestigationUser
    | project TimeGenerated, 
        EventType = "SignIn",
        Result = iff(ResultType == 0, "Success", "Failed"),
        IPAddress,
        Location,
        App = AppDisplayName,
        Device = DeviceDetail.displayName,
        Details = ResultDescription
),
(
    AuditLogs
    | where TimeGenerated > TimeWindow
    | where TargetResources[0].userPrincipalName == InvestigationUser 
        or InitiatedBy.user.userPrincipalName == InvestigationUser
    | project TimeGenerated,
        EventType = "AuditLog",
        Result = Result,
        IPAddress = InitiatedBy.user.ipAddress,
        Location = "",
        App = "",
        Device = "",
        Details = OperationName
),
(
    AADUserRiskEvents
    | where TimeGenerated > TimeWindow
    | where UserPrincipalName == InvestigationUser
    | project TimeGenerated,
        EventType = "RiskEvent",
        Result = RiskLevel,
        IPAddress = IpAddress,
        Location = Location,
        App = "",
        Device = "",
        Details = RiskEventType
)
| order by TimeGenerated asc

Containment Actions

Immediate Actions Based on Findings

FindingImmediate ActionCommand/Procedure
Successful login from attack IP Reset password + revoke sessions Revoke-AzureADUserAllRefreshToken -ObjectId <userid>
MFA bypass detected Re-enroll MFA, revoke app passwords Azure Portal → User → Authentication methods → Require re-register
External attack source Block IP/range at firewall Add to Conditional Access named location block list
Ongoing spray attack Enable smart lockout, temporary geo-block Azure Portal → Entra ID → Security → Authentication methods
No success, external source Monitor, consider preemptive block Add to watchlist, create detection rule

PowerShell - Emergency Response Commands

# Connect to Azure AD
Connect-AzureAD

# Revoke all sessions for compromised user
Revoke-AzureADUserAllRefreshToken -ObjectId "user@domain.com"

# Force password reset on next login
Set-AzureADUser -ObjectId "user@domain.com" -PasswordPolicies "DisablePasswordExpiration"
$password = ConvertTo-SecureString "TempP@ssw0rd123!" -AsPlainText -Force
Set-AzureADUserPassword -ObjectId "user@domain.com" -Password $password -ForceChangePasswordNextLogin $true

# Disable user if critical compromise
Set-AzureADUser -ObjectId "user@domain.com" -AccountEnabled $false

# Block sign-in for user
Set-AzureADUser -ObjectId "user@domain.com" -AccountEnabled $false

# List MFA methods for user (to identify rogue devices)
Get-MgUserAuthenticationMethod -UserId "user@domain.com"

# Remove specific authentication method if compromised
Remove-MgUserAuthenticationPhoneMethod -UserId "user@domain.com" -PhoneAuthenticationMethodId <id>

Conditional Access - Block Attack Sources

# Create Named Location for blocking
$params = @{
    "@odata.type" = "#microsoft.graph.ipNamedLocation"
    displayName = "Blocked Attack Sources"
    isTrusted = $false
    ipRanges = @(
        @{
            "@odata.type" = "#microsoft.graph.iPv4CidrRange"
            cidrAddress = "ATTACKER_IP/32"
        }
    )
}
New-MgIdentityConditionalAccessNamedLocation -BodyParameter $params

  • Source analysis: IPs, geolocation, reputation

  • User notifications: Who was notified, password resets
