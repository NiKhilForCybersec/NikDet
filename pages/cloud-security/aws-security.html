Detection Queries

Root Account Usage

// Root account should never be used for daily operations
AWSCloudTrail
| where UserIdentityType == "Root"
| project TimeGenerated, EventName, SourceIpAddress, 
    UserAgent, AWSRegion, ErrorCode

IAM User Created

// New IAM user - potential persistence
AWSCloudTrail
| where EventName == "CreateUser"
| extend NewUser = tostring(parse_json(RequestParameters).userName)
| project TimeGenerated, UserIdentityArn, NewUser, SourceIpAddress

Access Key Created

// New access key - persistence or credential theft
AWSCloudTrail
| where EventName == "CreateAccessKey"
| extend TargetUser = tostring(parse_json(RequestParameters).userName)
| project TimeGenerated, UserIdentityArn, TargetUser, SourceIpAddress

CloudTrail Disabled

// Defense evasion - disable logging
AWSCloudTrail
| where EventName in ("StopLogging", "DeleteTrail", "UpdateTrail")
| project TimeGenerated, UserIdentityArn, EventName, 
    SourceIpAddress, ErrorCode

S3 Bucket Made Public

// Public bucket exposure
AWSCloudTrail
| where EventName in ("PutBucketAcl", "PutBucketPolicy")
| where tostring(RequestParameters) has "AllUsers" 
    or tostring(RequestParameters) has "AuthenticatedUsers"
| extend BucketName = tostring(parse_json(RequestParameters).bucketName)
| project TimeGenerated, UserIdentityArn, BucketName, EventName

Security Group Opened

// Inbound rule added - check for 0.0.0.0/0
AWSCloudTrail
| where EventName == "AuthorizeSecurityGroupIngress"
| extend SGId = tostring(parse_json(RequestParameters).groupId)
| where tostring(RequestParameters) has "0.0.0.0/0"
| project TimeGenerated, UserIdentityArn, SGId, SourceIpAddress

Best Practices

  • Enable CloudTrail in all regions with S3 logging
  • Enable GuardDuty for automated threat detection
  • Use Security Hub to aggregate findings
  • Implement SCPs to prevent dangerous actions
  • Monitor root account - should rarely be used
  • Enable MFA Delete on critical S3 buckets
