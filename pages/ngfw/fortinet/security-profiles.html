Antivirus Profile

Create Antivirus Profile

# Create AV profile for client protection
config antivirus profile
    edit "AV-Client-Protect"
        set comment "Standard client protection"
        
        # HTTP scanning
        config http
            set av-scan block
            set outbreak-prevention block
        end
        
        # FTP scanning
        config ftp
            set av-scan block
            set outbreak-prevention block
        end
        
        # SMTP scanning
        config smtp
            set av-scan block
            set outbreak-prevention block
        end
        
        # POP3 scanning
        config pop3
            set av-scan block
            set outbreak-prevention block
        end
        
        # IMAP scanning
        config imap
            set av-scan block
            set outbreak-prevention block
        end
        
        # Other protocols
        config mapi
            set av-scan block
            set outbreak-prevention block
        end
        
        config nntp
            set av-scan block
            set outbreak-prevention block
        end
        
        # Submit unknown files to FortiSandbox
        set analytics-wl disable
        set analytics-bl enable
        set analytics-db enable
        set mobile-malware-db enable
        set analytics-max-upload 10240  # 10MB max file size
        set analytics-accept-filetype *  # All file types
        
    next
end

AV Profile for Servers (More Aggressive)

config antivirus profile
    edit "AV-Server-Strict"
        set comment "Strict AV for inbound traffic"
        
        config http
            set av-scan block
            set quarantine enable  # Quarantine detected files
            set outbreak-prevention block
        end
        
        # Enable extended scan (slower but deeper)
        set extended-utm-log enable
        set scan-mode full  # Full file scan, not quick
        
    next
end

Web Filter Profile

Create Corporate Web Filter

config webfilter profile
    edit "WebFilter-Corporate"
        set comment "Corporate web filtering policy"
        
        # FortiGuard Category Filtering
        config ftgd-wf
            # Block security risk categories
            set category 1 2 3 4 5 6  # Malware, phishing, spam URLs, etc.
            
            # Configure each category
            config filters
                edit 1
                    set category 26  # Malware
                    set action block
                next
                edit 2
                    set category 61  # Phishing
                    set action block
                next
                edit 3
                    set category 62  # Spam URLs
                    set action block
                next
                edit 4
                    set category 59  # Adult Content
                    set action block
                next
                edit 5
                    set category 24  # Gambling
                    set action block
                next
                edit 6
                    set category 75  # Streaming Media
                    set action warning  # Warn but allow
                next
                edit 7
                    set category 76  # Peer-to-peer
                    set action block
                next
                edit 8
                    set category 79  # Social Networking
                    set action monitor  # Allow but log
                next
            end
        end
        
        # Safe Search enforcement
        set safe-search url  # Force safe search on Google, Bing, etc.
        set youtube-restrict moderate
        
        # Block pages with expired SSL certificates
        set https-replacemsg enable
        
        # Logging
        set log-all-url enable
        set extended-log enable
        
    next
end

Web Filter URL Override (Whitelist/Blacklist)

# Create URL filter to whitelist/blacklist specific sites
config webfilter urlfilter
    edit 1
        set name "Corporate-Overrides"
        config entries
            edit 1
                set url "*.company.com"
                set type wildcard
                set action allow
            next
            edit 2
                set url "competitor.com"
                set type simple
                set action block
            next
            edit 3
                set url "*.gambling-site.com"
                set type wildcard
                set action block
            next
        end
    next
end

# Add to web filter profile
config webfilter profile
    edit "WebFilter-Corporate"
        config override
            set ovrd-user-group "Web-Override-Users"  # Who can request bypass
            set ovrd-dur-mode constant
            set ovrd-dur 86400  # 24 hour override
        end
        set web-filter-url-log enable
    next
end

Application Control

Create Application Control Profile

config application list
    edit "AppControl-Corporate"
        set comment "Corporate application policy"
        
        # Default action for unknown apps
        set unknown-application-action pass
        set unknown-application-log enable
        
        # Block high-risk apps
        config entries
            edit 1
                set category 2  # P2P
                set action block
            next
            edit 2
                set category 7  # Proxy
                set action block
            next
            edit 3
                set category 17  # Game
                set action block
            next
        end
        
        # Control specific applications
        config entries
            edit 10
                set application 15836  # BitTorrent
                set action block
            next
            edit 11
                set application 16395  # Tor
                set action block
            next
            edit 12
                set application 16058  # TeamViewer
                set action block  # Or set to 'pass' if needed
            next
            edit 13
                set application 15895  # Facebook
                set action pass  # Allow Facebook
            next
            edit 14
                set application 31077  # Facebook.Video
                set action block  # But block Facebook videos
            next
        end
        
    next
end

IPS (Intrusion Prevention System)

Create IPS Sensor

config ips sensor
    edit "IPS-Client-Protect"
        set comment "IPS for client networks"
        
        # Protect against common client-side attacks
        config entries
            edit 1
                set rule 27671  # Example: CVE signature
                set status enable
                set action block
                set log-packet enable
            next
        end
        
        # Use filters to protect broad categories
        config filter
            edit 1
                set location client
                set severity high critical
                set action block
                set log enable
                set log-packet enable
            next
            edit 2
                set location client
                set severity medium
                set action default  # Use signature's default action
                set log enable
            next
        end
        
    next
end

# Create IPS Sensor for Servers (Strict)
config ips sensor
    edit "IPS-Server-Protect"
        set comment "IPS for server networks - strict"
        
        config filter
            edit 1
                set location server
                set severity high critical medium
                set action block
                set log enable
                set log-packet enable  # Capture packets for forensics
            next
            edit 2
                set application IIS Apache Nginx PHP
                set severity high critical
                set action block
            next
        end
        
        config override
            edit 1
                set rule-id 12345  # Override specific rule
                set status enable
                set action block
            next
        end
        
    next
end

SSL/TLS Inspection

Why SSL Inspection is Critical

~90% of web traffic is encrypted. Without SSL inspection:

  • Web Filter sees only domain, not URL path
  • AV cannot scan files in HTTPS
  • IPS cannot detect attacks in encrypted traffic
  • App Control has limited visibility

SSL Inspection Modes

ModeDescriptionUse Case
Certificate InspectionInspect certificate only, not contentBasic protection, compliance concerns
Deep InspectionDecrypt, inspect, re-encrypt all trafficFull visibility and protection

Create SSL Inspection Profile

# Certificate Inspection (Light - Recommended Start)
config firewall ssl-ssh-profile
    edit "SSL-Certificate-Inspection"
        set comment "Certificate inspection only"
        
        config ssl
            set inspect-all disable  # Don't deep inspect
            set client-certificate request  # Request client cert
        end
        
        config https
            set ports 443
            set status certificate-inspection
            set client-cert-request bypass
        end
        
        # Exempt categories from inspection
        config ssl-exempt
            edit 1
                set type category
                set fortiguard-category 11  # Financial Services
            next
            edit 2
                set type category
                set fortiguard-category 26  # Health Care
            next
        end
        
    next
end

# Deep Inspection (Full Visibility)
config firewall ssl-ssh-profile
    edit "SSL-Deep-Inspection"
        set comment "Full deep inspection"
        
        config ssl
            set inspect-all deep-inspection
        end
        
        config https
            set ports 443 8443
            set status deep-inspection
            
            # Use FortiGate's CA certificate
            set client-cert-request bypass
        end
        
        # VERY IMPORTANT: Exempt certificate-pinned sites
        config ssl-exempt
            edit 1
                set type wildcard-fqdn
                set wildcard-fqdn "*.apple.com"  # iOS updates
            next
            edit 2
                set type wildcard-fqdn
                set wildcard-fqdn "*.windows.com"  # Windows Update
            next
            edit 3
                set type wildcard-fqdn
                set wildcard-fqdn "*.microsoft.com"
            next
            edit 4
                set type wildcard-fqdn
                set wildcard-fqdn "*.icloud.com"
            next
            edit 5
                set type category
                set fortiguard-category 11  # Financial
            next
            edit 6
                set type category
                set fortiguard-category 26  # Health
            next
        end
        
    next
end

Deep Inspection Requirements

  • CA Certificate: Must install FortiGate CA on all clients
  • GPO Deployment: Use Active Directory GPO to push certificate
  • Mobile Devices: MDM to push certificate to phones/tablets
  • Privacy: Some jurisdictions have legal requirements around inspection

Export and Deploy FortiGate CA

# Export FortiGate CA certificate
config vpn certificate local
    edit "Fortinet_CA_SSL"
        # Certificate is built-in, export from GUI
    next
end

# GUI: Security Profiles → SSL/SSH Inspection → 
#      Download Certificate (Fortinet_CA_SSL)

Deploy via GPO:

  1. Open Group Policy Management
  2. Edit GPO → Computer Configuration → Policies → Windows Settings
  3. Security Settings → Public Key Policies
  4. Trusted Root Certification Authorities
  5. Import the FortiGate CA certificate

FortiSandbox Integration

Configure FortiSandbox Connection

# Cloud FortiSandbox (FortiGuard)
config system fortisandbox
    set status enable
    set forticloud enable
end

# On-premise FortiSandbox
config system fortisandbox
    set status enable
    set server "10.1.1.200"  # FortiSandbox IP
end

# Verify connection
diagnose test application forticldd 1

Enable in AV Profile

config antivirus profile
    edit "AV-With-Sandbox"
        set analytics-db enable
        set analytics-max-upload 10240
        set analytics-wl enable
        set analytics-bl enable
        set fortisandbox-mode inline-block  # Wait for verdict
        # or: set fortisandbox-mode analytics  # Log only
    next
end

Apply Profiles to Policies

# Apply all security profiles to a policy
config firewall policy
    edit 10
        set name "Users-to-Internet"
        set srcintf "Trust"
        set dstintf "Untrust"
        set srcaddr "all"
        set dstaddr "all"
        set action accept
        set schedule "always"
        set service "ALL"
        set nat enable
        
        # Enable UTM
        set utm-status enable
        
        # Apply profiles
        set av-profile "AV-Client-Protect"
        set webfilter-profile "WebFilter-Corporate"
        set application-list "AppControl-Corporate"
        set ips-sensor "IPS-Client-Protect"
        set ssl-ssh-profile "SSL-Deep-Inspection"
        
        # Logging
        set logtraffic all
        set logtraffic-start enable
    next
end
