Data Exfiltration Investigation

Detect and investigate unauthorized data transfers leaving your organization.

Part 1: Large Data Transfer Detection

// KQL: Large outbound data transfers
DeviceNetworkEvents
| where TimeGenerated > ago(24h)
| where RemoteIPType == "Public"
| where SentBytes > 0
| summarize 
    TotalBytesSent = sum(SentBytes),
    ConnectionCount = count(),
    UniqueDestinations = dcount(RemoteIP)
  by DeviceName, InitiatingProcessFileName
| where TotalBytesSent > 104857600  // > 100 MB
| extend TotalMB = round(TotalBytesSent / 1048576.0, 2)
| order by TotalBytesSent desc

// KQL: Data transfer to cloud storage domains
let CloudStorageDomains = dynamic(["dropbox.com", "drive.google.com", "onedrive.live.com",
    "mega.nz", "mediafire.com", "wetransfer.com", "sendspace.com", "box.com"]);
DeviceNetworkEvents
| where TimeGenerated > ago(24h)
| where RemoteUrl has_any (CloudStorageDomains)
| summarize TotalBytes = sum(SentBytes) by DeviceName, RemoteUrl, InitiatingProcessAccountName
| where TotalBytes > 10485760  // > 10 MB
| order by TotalBytes desc

Part 2: Archive Creation Before Exfil

// KQL: Archive tool usage (staging data for exfil)
DeviceProcessEvents
| where TimeGenerated > ago(7d)
| where FileName in~ ("7z.exe", "7za.exe", "rar.exe", "winrar.exe", "zip.exe", "tar.exe")
    or ProcessCommandLine has_any (".7z", ".rar", ".zip", ".tar.gz", "-p")
| project TimeGenerated, DeviceName, AccountName, FileName, ProcessCommandLine
| order by TimeGenerated asc

// KQL: Large archive files created
DeviceFileEvents
| where TimeGenerated > ago(7d)
| where ActionType == "FileCreated"
| where FileName endswith ".zip" or FileName endswith ".7z" or FileName endswith ".rar"
| where FileSize > 52428800  // > 50 MB
| project TimeGenerated, DeviceName, InitiatingProcessAccountName, FileName, FileSize, FolderPath

Part 3: Email Exfiltration

// KQL: Large email attachments to external recipients
EmailEvents
| where TimeGenerated > ago(7d)
| where EmailDirection == "Outbound"
| where RecipientEmailAddress !endswith "@yourdomain.com"
| where AttachmentCount > 0
| join kind=inner (
    EmailAttachmentInfo
    | where FileSize > 5242880  // > 5 MB attachments
) on NetworkMessageId
| project TimeGenerated, SenderFromAddress, RecipientEmailAddress, Subject, FileName, FileSize
| order by FileSize desc

// KQL: Emails to personal email domains
let PersonalDomains = dynamic(["gmail.com", "yahoo.com", "hotmail.com", "outlook.com", 
    "protonmail.com", "aol.com", "icloud.com"]);
EmailEvents
| where TimeGenerated > ago(7d)
| where EmailDirection == "Outbound"
| extend RecipientDomain = tostring(split(RecipientEmailAddress, "@")[1])
| where RecipientDomain in (PersonalDomains)
| where AttachmentCount > 0
| summarize EmailCount = count(), TotalAttachments = sum(AttachmentCount) by SenderFromAddress
| order by TotalAttachments desc

Part 4: USB/Removable Media

// KQL: USB device connections
DeviceEvents
| where TimeGenerated > ago(7d)
| where ActionType == "PnpDeviceConnected"
| where DeviceDescription has_any ("USB", "Removable", "Mass Storage")
| project TimeGenerated, DeviceName, DeviceId, DeviceDescription

// KQL: Files copied to removable drives
DeviceFileEvents
| where TimeGenerated > ago(7d)
| where ActionType == "FileCreated"
| where FolderPath matches regex @"^[D-Z]:\\"  // Non-C: drives
| where FolderPath !startswith "D:\\Windows"  // Exclude system
| summarize FilesCopied = count(), TotalSize = sum(FileSize) by DeviceName, InitiatingProcessAccountName
| where FilesCopied > 10
| order by TotalSize desc

Part 5: Known Exfiltration Tools

// KQL: Rclone and similar sync tools
DeviceProcessEvents
| where TimeGenerated > ago(7d)
| where FileName in~ ("rclone.exe", "rsync.exe", "robocopy.exe", "xcopy.exe")
    or ProcessCommandLine has_any ("rclone", "mega-cmd", "gdrive")
| where ProcessCommandLine has_any ("copy", "sync", "move")
| project TimeGenerated, DeviceName, AccountName, ProcessCommandLine

// KQL: FTP/SFTP connections
DeviceNetworkEvents
| where TimeGenerated > ago(7d)
| where RemotePort in (21, 22, 990)  // FTP, SFTP, FTPS
| where RemoteIPType == "Public"
| summarize TotalBytes = sum(SentBytes), Connections = count() by DeviceName, RemoteIP
| order by TotalBytes desc

Part 6: Determine What Was Taken

Critical for breach notification and impact assessment:

  • File access logs: What files were accessed before transfer?
  • DLP logs: What data classifications were triggered?
  • Database audit: Were database exports run?
  • Archive contents: Analyze any recovered archives
