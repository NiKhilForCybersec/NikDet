Common Log Analysis Queries

// Timeline for entity (KQL)
union SecurityEvent, SigninLogs, OfficeActivity, DeviceProcessEvents
| where TimeGenerated > ago(24h)
| where Account contains "username" or UserPrincipalName contains "username"
| project TimeGenerated, Type, Activity=coalesce(Activity, Operation, ActionType), 
    Details=coalesce(ProcessCommandLine, TargetResources, "")
| order by TimeGenerated asc

// Rare process detection
DeviceProcessEvents
| where TimeGenerated > ago(7d)
| summarize Count=count(), Hosts=dcount(DeviceName) by FileName, SHA256
| where Count < 5 and Hosts == 1
| order by Count asc

// Beaconing detection (regular intervals)
CommonSecurityLog
| where TimeGenerated > ago(24h)
| summarize Times=make_list(TimeGenerated) by SourceIP, DestinationIP
| mv-apply Times to typeof(datetime) on (
    extend NextTime = next(Times)
    | extend Interval = datetime_diff('second', NextTime, Times)
    | summarize Avg=avg(Interval), Stdev=stdev(Interval), Count=count()
)
| where Stdev < 5 and Count > 50
