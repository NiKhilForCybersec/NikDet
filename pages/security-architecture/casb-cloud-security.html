Cloud Security Queries

// KQL - OAuth apps with high-risk permissions
AuditLogs
| where OperationName == "Consent to application"
| extend Permissions = tostring(TargetResources[0].modifiedProperties)
| where Permissions has_any ("Mail.ReadWrite", "Files.ReadWrite.All", "Directory.ReadWrite")
| project TimeGenerated, User = InitiatedBy.user.userPrincipalName, App = TargetResources[0].displayName, Permissions

// KQL - Unusual cloud app usage
CloudAppEvents
| where TimeGenerated > ago(24h)
| summarize EventCount = count() by Application, AccountDisplayName
| join kind=leftouter (
    CloudAppEvents | where TimeGenerated between(ago(30d) .. ago(1d)) | summarize BaselineCount = count() by Application, AccountDisplayName
) on Application, AccountDisplayName
| where EventCount > BaselineCount * 3

// KQL - External sharing in SharePoint/OneDrive
CloudAppEvents
| where ActionType in ("SharingSet", "AnonymousLinkCreated")
| where RawEventData has "External" or RawEventData has "Anyone"
| project TimeGenerated, AccountDisplayName, ActionType, ObjectName

// KQL - Azure resource changes
AzureActivity
| where OperationNameValue has_any ("delete", "write")
| where ActivityStatusValue == "Succeeded"
| project TimeGenerated, Caller, OperationNameValue, ResourceGroup, Resource
