C2 IP/Domain Blocking

Palo Alto Firewall

// Block IP via CLI
configure
set address-group SOC-Blocked-IPs static 185.123.45.67
set security rulebase security rules Block-C2 source any destination SOC-Blocked-IPs action deny log-end yes
commit

// Block Domain (Anti-Spyware Profile)
set profiles spyware SOC-Block-Domains botnet-domains sinkhole ipv4-address 127.0.0.1
set profiles spyware SOC-Block-Domains botnet-domains dns-security-categories enabled

// Add to External Dynamic List (EDL)
set external-list SOC-IOCs type ip
set external-list SOC-IOCs source url https://yourserver.com/iocs.txt
set external-list SOC-IOCs refresh-rate hour 1

Cisco ASA/FTD

// Block IP via ACL
access-list SOC-BLOCK extended deny ip any host 185.123.45.67 log
access-list SOC-BLOCK extended deny ip host 185.123.45.67 any log
access-group SOC-BLOCK in interface outside

// Add to object group for easier management
object-group network SOC-Blocked-IPs
 network-object host 185.123.45.67
 network-object host 192.0.2.100

access-list SOC-BLOCK extended deny ip any object-group SOC-Blocked-IPs

Fortinet FortiGate

// Block IP via address object
config firewall address
    edit "SOC-Block-185.123.45.67"
        set type ipmask
        set subnet 185.123.45.67 255.255.255.255
    next
end

config firewall policy
    edit 1
        set name "SOC-Block-C2"
        set srcintf "any"
        set dstintf "any"
        set srcaddr "all"
        set dstaddr "SOC-Block-185.123.45.67"
        set action deny
        set logtraffic all
    next
end

DNS Sinkholing

Microsoft DNS

// Create sinkhole zone (PowerShell)
Add-DnsServerPrimaryZone -Name "malicious-domain.com" -ZoneFile "malicious-domain.com.dns"
Add-DnsServerResourceRecordA -Name "@" -ZoneName "malicious-domain.com" -IPv4Address "127.0.0.1"
Add-DnsServerResourceRecordA -Name "*" -ZoneName "malicious-domain.com" -IPv4Address "127.0.0.1"

// Using Response Policy Zone (RPZ) if supported
Add-DnsServerResponseRateLimitingExceptionlist -Name "SOC-Sinkhole" -Fqdn "malicious-domain.com"

Infoblox

// Add to RPZ via API
POST /wapi/v2.10/record:rpz:cname
{
    "name": "malicious-domain.com.rpz.local",
    "canonical": "sinkhole.company.com",
    "rp_zone": "rpz.local"
}

// Block via Threat Intelligence Feed
Grid > DNS > Response Policy Zones > Add Custom Feed

Web Proxy Blocking

Zscaler

// Block URL via Admin Portal
Administration > URL Categories > Add URL > Custom Category: SOC-Blocked

// Block via API
POST /api/v1/webApplicationRules
{
    "name": "SOC-Block-Malicious",
    "state": "ENABLED",
    "rank": 1,
    "urlCategories": ["SOC-Blocked"],
    "action": "BLOCK"
}

Cisco Umbrella

// Block domain via Dashboard
Policies > Destination Lists > Add > Block List

// Block via API
POST /1.0/destinationlists/{destinationListId}/destinations
{
    "destination": "malicious-domain.com",
    "comment": "SOC-2025-001 C2 domain"
}

Emergency Lateral Movement Block

// Block SMB between workstation VLANs (Palo Alto example)
set rulebase security rules Emergency-Block-SMB source Workstation-Zone destination Workstation-Zone 
    application ms-ds-smb action deny log-end yes

// Windows Firewall GPO - Block inbound SMB
New-NetFirewallRule -DisplayName "Emergency-Block-SMB" -Direction Inbound -Protocol TCP -LocalPort 445 -Action Block

// Block RDP except from jump server
set rulebase security rules Allow-RDP-Jump source Jump-Server-Zone destination any application ms-rdp action allow
set rulebase security rules Block-RDP-Other source any destination any application ms-rdp action deny

Verification Queries

// Verify IP block is working (Sentinel)
CommonSecurityLog
| where TimeGenerated > ago(1h)
| where DestinationIP == "185.123.45.67" or SourceIP == "185.123.45.67"
| summarize Count=count() by DeviceAction, DeviceVendor
| where DeviceAction !in ("Deny", "Block", "Drop")

// Check for DNS queries to blocked domain
DnsEvents
| where TimeGenerated > ago(1h)
| where Name contains "malicious-domain.com"
| summarize Count=count() by ClientIP, Name

// Verify traffic stopped from isolated host
CommonSecurityLog
| where TimeGenerated > ago(1h)
| where SourceIP == "10.1.2.50"
| where DeviceAction in ("Allow", "Permit")
| summarize Count=count() by DestinationIP, DestinationPort
