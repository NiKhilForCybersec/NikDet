Secrets Management

Secure storage, rotation, and distribution of secrets for infrastructure and applications.

Never Store Secrets In
  • Git repositories (even private ones)
  • Environment variables in CI/CD configs
  • Terraform state files (without encryption)
  • Container images
  • Application code

Vault CLI Commands

# Login
vault login -method=oidc
vault login -method=aws

# KV Secrets Engine (v2)
vault kv put secret/myapp/config username=admin password=supersecret
vault kv get secret/myapp/config
vault kv get -field=password secret/myapp/config
vault kv delete secret/myapp/config

# List secrets
vault kv list secret/myapp/

# Enable secrets engine
vault secrets enable -path=myapp kv-v2

# Dynamic Database Credentials
vault read database/creds/myapp-role
# Returns: username=v-token-myapp-xxxxx, password=A1b2C3...
# Credentials auto-expire after TTL

# PKI - Generate certificate
vault write pki/issue/myapp-role common_name="app.example.com" ttl="24h"

Terraform + Vault

# Read secrets from Vault
provider "vault" {
  address = "https://vault.company.com"
}

data "vault_kv_secret_v2" "db_creds" {
  mount = "secret"
  name  = "database/prod"
}

resource "aws_db_instance" "main" {
  identifier = "prod-database"
  username   = data.vault_kv_secret_v2.db_creds.data["username"]
  password   = data.vault_kv_secret_v2.db_creds.data["password"]
}

# Write secrets to Vault
resource "vault_kv_secret_v2" "api_key" {
  mount = "secret"
  name  = "api-keys/service-a"
  
  data_json = jsonencode({
    api_key = random_password.api_key.result
  })
}

AWS Secrets Manager

# Terraform: Create secret with auto-rotation
resource "aws_secretsmanager_secret" "db_password" {
  name        = "prod/database/password"
  description = "Production database password"
  
  tags = {
    Environment = "prod"
    Application = "myapp"
  }
}

resource "aws_secretsmanager_secret_version" "db_password" {
  secret_id     = aws_secretsmanager_secret.db_password.id
  secret_string = random_password.db.result
}

# Enable auto-rotation
resource "aws_secretsmanager_secret_rotation" "db_password" {
  secret_id           = aws_secretsmanager_secret.db_password.id
  rotation_lambda_arn = aws_lambda_function.rotation.arn
  
  rotation_rules {
    automatically_after_days = 30
  }
}

# Read secret in Terraform
data "aws_secretsmanager_secret_version" "db_password" {
  secret_id = "prod/database/password"
}

# Use in resource
resource "aws_db_instance" "main" {
  password = data.aws_secretsmanager_secret_version.db_password.secret_string
}

Kubernetes External Secrets

# Install External Secrets Operator
helm repo add external-secrets https://charts.external-secrets.io
helm install external-secrets external-secrets/external-secrets

# Configure AWS Secrets Manager as SecretStore
apiVersion: external-secrets.io/v1beta1
kind: ClusterSecretStore
metadata:
  name: aws-secrets-manager
spec:
  provider:
    aws:
      service: SecretsManager
      region: us-east-1
      auth:
        jwt:
          serviceAccountRef:
            name: external-secrets
            namespace: external-secrets

# Sync secret from AWS to Kubernetes
apiVersion: external-secrets.io/v1beta1
kind: ExternalSecret
metadata:
  name: db-credentials
  namespace: production
spec:
  refreshInterval: 1h
  secretStoreRef:
    name: aws-secrets-manager
    kind: ClusterSecretStore
  target:
    name: db-credentials  # K8s Secret name
    creationPolicy: Owner
  data:
    - secretKey: username
      remoteRef:
        key: prod/database/credentials
        property: username
    - secretKey: password
      remoteRef:
        key: prod/database/credentials
        property: password

Best Practices

  • Use dynamic secrets: Generate credentials on-demand with TTL
  • Principle of least privilege: Applications get only the secrets they need
  • Audit all access: Log every secret read/write operation
  • Encrypt at rest and in transit: TLS everywhere, encrypted storage
  • Separate environments: Different secrets for dev/staging/prod
  • No shared secrets: Each application gets unique credentials
  • Automate rotation: Never manually rotate secrets
