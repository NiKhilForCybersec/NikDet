FortiGate Firewall Objects

Create and manage address objects, service objects, and groups for use in firewall policies.

Address Objects

IP/Subnet Address

config firewall address
    edit "Server-Web01"
        set type ipmask
        set subnet 10.0.10.50 255.255.255.255
        set comment "Production web server"
    next
    edit "Network-Servers"
        set type ipmask
        set subnet 10.0.10.0 255.255.255.0
        set comment "Server VLAN"
    next
end

FQDN Address

config firewall address
    edit "Office365-Endpoints"
        set type fqdn
        set fqdn "outlook.office365.com"
    next
end

IP Range

config firewall address
    edit "DHCP-Range"
        set type iprange
        set start-ip 10.0.1.100
        set end-ip 10.0.1.200
    next
end

Geography-Based Address

config firewall address
    edit "Country-Russia"
        set type geography
        set country "RU"
    next
    edit "Country-China"
        set type geography
        set country "CN"
    next
end

Address Groups

config firewall addrgrp
    edit "Group-WebServers"
        set member "Server-Web01" "Server-Web02" "Server-Web03"
        set comment "All production web servers"
    next
    edit "Group-BlockedCountries"
        set member "Country-Russia" "Country-China" "Country-NorthKorea"
        set comment "Countries to block"
    next
end

Service Objects

Custom TCP/UDP Service

config firewall service custom
    edit "Custom-App-8080"
        set tcp-portrange 8080
        set comment "Custom application port"
    next
    edit "Custom-App-Range"
        set tcp-portrange 8000-8100
        set comment "Application port range"
    next
    edit "Custom-UDP-Service"
        set udp-portrange 5000-5010
    next
end

Service Groups

config firewall service group
    edit "Web-Services"
        set member "HTTP" "HTTPS" "Custom-App-8080"
    next
    edit "Email-Services"
        set member "SMTP" "SMTPS" "POP3" "POP3S" "IMAP" "IMAPS"
    next
end

Schedule Objects

config firewall schedule recurring
    edit "Business-Hours"
        set day monday tuesday wednesday thursday friday
        set start 08:00
        set end 18:00
    next
end

config firewall schedule onetime
    edit "Maintenance-Window"
        set start 23:00 2025/01/15
        set end 05:00 2025/01/16
    next
end

Verification

# Show all address objects
show firewall address

# Show address groups
show firewall addrgrp

# Show services
show firewall service custom

# Diagnose FQDN resolution
diagnose firewall fqdn list
