Create CA policy → Grant → Require authentication strength → Select custom strength

PowerShell Configuration

# Connect to Microsoft Graph
Connect-MgGraph -Scopes "Policy.ReadWrite.AuthenticationMethod"

# Enable Authenticator with number matching (CRITICAL - prevents MFA fatigue)
$params = @{
    "@odata.type" = "#microsoft.graph.microsoftAuthenticatorAuthenticationMethodConfiguration"
    state = "enabled"
    featureSettings = @{
        numberMatchingRequiredState = @{ state = "enabled" }
        displayAppInformationRequiredState = @{ state = "enabled" }
        displayLocationInformationRequiredState = @{ state = "enabled" }
    }
}
Update-MgPolicyAuthenticationMethodPolicyAuthenticationMethodConfiguration `
    -AuthenticationMethodConfigurationId "MicrosoftAuthenticator" -BodyParameter $params

# Disable SMS (recommended)
$smsParams = @{
    "@odata.type" = "#microsoft.graph.smsAuthenticationMethodConfiguration"
    state = "disabled"
}
Update-MgPolicyAuthenticationMethodPolicyAuthenticationMethodConfiguration `
    -AuthenticationMethodConfigurationId "Sms" -BodyParameter $smsParams

# Enable FIDO2 security keys
$fido2Params = @{
    "@odata.type" = "#microsoft.graph.fido2AuthenticationMethodConfiguration"
    state = "enabled"
    isAttestationEnforced = $true
    isSelfServiceRegistrationAllowed = $true
}
Update-MgPolicyAuthenticationMethodPolicyAuthenticationMethodConfiguration `
    -AuthenticationMethodConfigurationId "Fido2" -BodyParameter $fido2Params

# Get user's registered methods
Get-MgUserAuthenticationMethod -UserId "user@contoso.com"

# Reset user's MFA (delete Authenticator registration)
$methods = Get-MgUserAuthenticationMicrosoftAuthenticatorMethod -UserId "user@contoso.com"
$methods | ForEach-Object {
    Remove-MgUserAuthenticationMicrosoftAuthenticatorMethod `
        -UserId "user@contoso.com" -MicrosoftAuthenticatorAuthenticationMethodId $_.Id
}

User Self-Registration

  1. Navigate to mysignins.microsoft.com/security-info
  2. Click "Add sign-in method" → "Security key"
  3. Choose USB or NFC device type
  4. Insert key, follow browser prompts
  5. Name the key (e.g., "YubiKey - Work Laptop")

Monitoring MFA Events (KQL)

// MFA failures by user and method
SigninLogs
| where TimeGenerated > ago(7d)
| where ResultType != 0 and AuthenticationRequirement == "multiFactorAuthentication"
| mv-expand AuthenticationDetails
| extend AuthMethod = tostring(AuthenticationDetails.authenticationMethod)
| where tostring(AuthenticationDetails.succeeded) == "false"
| summarize Failures = count() by UserPrincipalName, AuthMethod, ResultDescription
| order by Failures desc

// Users still using SMS (identify for migration)
SigninLogs
| where TimeGenerated > ago(30d) and ResultType == 0
| mv-expand AuthenticationDetails
| where tostring(AuthenticationDetails.authenticationMethod) == "Text message"
| summarize SMSLogins = count() by UserPrincipalName
| order by SMSLogins desc

// MFA fatigue attack detection
SigninLogs
| where TimeGenerated > ago(1h)
| where ResultType == 50074  // MFA required
| summarize MFAPrompts = count() by UserPrincipalName, IPAddress, bin(TimeGenerated, 5m)
| where MFAPrompts > 5
