Investigation Steps

Step 1: Assess Exposure

First 30 Minutes - Critical Questions
  • Is the affected software/package in our environment?
  • What version(s) are affected vs. what we have?
  • How many systems/applications are impacted?
  • Was the malicious version deployed in our environment?
  • What access does this software have?

KQL - Software Inventory Check

// Check for specific software installation
DeviceTvmSoftwareInventory
| where SoftwareName contains "AFFECTED_SOFTWARE_NAME"
| summarize 
    DeviceCount = dcount(DeviceName),
    Versions = make_set(SoftwareVersion),
    Devices = make_set(DeviceName, 100)
| extend Exposure = strcat(DeviceCount, " devices with versions: ", Versions)

KQL - File Hash Search

// Search for known malicious file hashes
let MaliciousHashes = dynamic([
    "SHA256_HASH_1",
    "SHA256_HASH_2"
]);
DeviceFileEvents
| where TimeGenerated > ago(90d)
| where SHA256 in (MaliciousHashes) or SHA1 in (MaliciousHashes)
| project TimeGenerated, DeviceName, FileName, FolderPath, SHA256, 
    ActionType, InitiatingProcessFileName
| order by TimeGenerated asc

KQL - Process From Compromised Software

// Check for processes from suspected compromised software
DeviceProcessEvents
| where TimeGenerated > ago(30d)
| where InitiatingProcessFileName =~ "COMPROMISED_SOFTWARE.exe"
    or ProcessVersionInfoCompanyName contains "AFFECTED_VENDOR"
| summarize 
    FirstSeen = min(TimeGenerated),
    LastSeen = max(TimeGenerated),
    Devices = make_set(DeviceName),
    ChildProcesses = make_set(FileName)
    by InitiatingProcessFileName, InitiatingProcessVersionInfoProductVersion

Step 2: Identify Indicators of Compromise

KQL - Behavioral Detection (SolarWinds-style)

// Detect anomalous behavior from trusted software
DeviceProcessEvents
| where TimeGenerated > ago(7d)
// Software that shouldn't spawn certain processes
| where InitiatingProcessFileName in~ ("BusinessApp.exe", "UpdateService.exe")
| where FileName in~ ("cmd.exe", "powershell.exe", "rundll32.exe", 
    "regsvr32.exe", "mshta.exe", "certutil.exe")
| project TimeGenerated, DeviceName, InitiatingProcessFileName, 
    FileName, ProcessCommandLine
| order by TimeGenerated desc

KQL - Unusual Network from Trusted Process

// Trusted software making unusual network connections
DeviceNetworkEvents
| where TimeGenerated > ago(7d)
| where InitiatingProcessFileName =~ "TrustedApp.exe"
| where RemoteUrl !contains "legitimatedomain.com"
| summarize 
    ConnectionCount = count(),
    UniqueDestinations = dcount(RemoteUrl),
    Destinations = make_set(RemoteUrl, 50)
    by DeviceName, InitiatingProcessFileName
| where UniqueDestinations > 3  // Adjust based on baseline

SPL - Package Manager Installs (Splunk)

# Check for malicious package installation
index=linux sourcetype=syslog 
| search "npm install" OR "pip install" OR "gem install"
| rex field=_raw "(?i)(npm|pip|gem)\s+install\s+(?\S+)"
| stats count by host, package
| lookup malicious_packages.csv package OUTPUT is_malicious
| where is_malicious="true"

Step 3: Timeline Analysis

KQL - Full Timeline for Affected System

// Build comprehensive timeline
let AffectedDevice = "HOSTNAME";
let TimeWindow = ago(30d);
union
(
    DeviceProcessEvents
    | where TimeGenerated > TimeWindow
    | where DeviceName == AffectedDevice
    | where InitiatingProcessFileName =~ "COMPROMISED_APP.exe"
    | project TimeGenerated, EventType = "Process", 
        Details = strcat(FileName, " - ", ProcessCommandLine)
),
(
    DeviceNetworkEvents
    | where TimeGenerated > TimeWindow
    | where DeviceName == AffectedDevice
    | where InitiatingProcessFileName =~ "COMPROMISED_APP.exe"
    | project TimeGenerated, EventType = "Network",
        Details = strcat(RemoteUrl, ":", RemotePort)
),
(
    DeviceFileEvents
    | where TimeGenerated > TimeWindow
    | where DeviceName == AffectedDevice
    | where InitiatingProcessFileName =~ "COMPROMISED_APP.exe"
    | project TimeGenerated, EventType = "File",
        Details = strcat(ActionType, ": ", FolderPath, "\\", FileName)
),
(
    DeviceRegistryEvents
    | where TimeGenerated > TimeWindow
    | where DeviceName == AffectedDevice
    | where InitiatingProcessFileName =~ "COMPROMISED_APP.exe"
    | project TimeGenerated, EventType = "Registry",
        Details = strcat(ActionType, ": ", RegistryKey)
)
| order by TimeGenerated asc

Step 4: Lateral Movement Check

KQL - Check for Spread from Compromised System

// Did attacker move laterally after supply chain compromise?
let CompromisedDevices = DeviceProcessEvents
| where InitiatingProcessFileName =~ "COMPROMISED_APP.exe"
| distinct DeviceName;
// Check for lateral movement indicators
DeviceNetworkEvents
| where TimeGenerated > ago(7d)
| where DeviceName in (CompromisedDevices)
| where RemotePort in (445, 135, 139, 3389, 5985, 5986, 22)
| where RemoteIPType == "Private"
| summarize ConnectionCount = count(), 
    TargetHosts = make_set(RemoteIP)
    by DeviceName, RemotePort
| where ConnectionCount > 5

Containment Actions

ScenarioImmediate ActionImplementation
Confirmed compromised software deployed Isolate affected systems, block software Network isolation, application block
Active C2 connections detected Block C2 domains/IPs, isolate systems Firewall rules, DNS sinkhole
Vulnerable version in environment Emergency patching, compensating controls Update software, restrict access
Third-party vendor breached Revoke vendor access, reset credentials Disable accounts, rotate API keys
Malicious package in dependencies Remove package, rebuild affected apps npm uninstall, pip uninstall, rebuild

Application Blocking (WDAC)

# Create WDAC policy to block compromised software
$Rules = @()
$Rules += New-CIPolicyRule -Level Publisher -Fallback Hash `
    -DriverFilePath "C:\Path\To\COMPROMISED.exe" -Deny

# Or block by hash
$Rules += New-CIPolicyRule -Level Hash `
    -Hash "SHA256_OF_MALICIOUS_FILE" -Deny

New-CIPolicy -FilePath ".\BlockCompromised.xml" -Rules $Rules
ConvertFrom-CIPolicy ".\BlockCompromised.xml" ".\BlockCompromised.bin"
