Blocked File Extensions

High-risk (always block): .exe, .scr, .bat, .cmd, .ps1, .vbs, .js, .jse, .wsf, .wsh, .msi, .msp, .com, .pif, .application, .gadget, .msc, .jar, .hta

Medium-risk (consider blocking): .docm, .xlsm, .pptm (macros), .iso, .img, .vhd (disk images), .lnk, .url (shortcuts)

Email Security KQL Queries

// Phishing emails that were delivered
EmailEvents
| where ThreatTypes has "Phish"
| where DeliveryAction == "Delivered"
| summarize Count = count() by SenderFromAddress, Subject
| order by Count desc

// Users who clicked malicious links
UrlClickEvents
| where ThreatTypes has_any ("Phish", "Malware")
| where IsClickedThrough == true
| summarize ClickCount = count() by AccountUpn
| order by ClickCount desc

// Emails with suspicious attachments
EmailAttachmentInfo
| where FileType in ("exe", "scr", "js", "vbs", "ps1")
| join EmailEvents on NetworkMessageId
| project TimeGenerated, SenderFromAddress, RecipientEmailAddress, FileName, FileType

// Authentication failures (spoofing attempts)
EmailEvents
| where AuthenticationDetails has "fail"
| extend SPF = extractjson("$.SPF", AuthenticationDetails)
| extend DKIM = extractjson("$.DKIM", AuthenticationDetails)
| extend DMARC = extractjson("$.DMARC", AuthenticationDetails)
| where SPF == "fail" or DKIM == "fail" or DMARC == "fail"
| summarize Count = count() by SenderFromDomain
