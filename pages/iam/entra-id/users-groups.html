Dynamic Group Rules

# All users in Marketing department
(user.department -eq "Marketing")

# All users in US
(user.country -eq "US")

# All users with Manager attribute set
(user.manager -ne null)

# All devices running Windows
(device.deviceOSType -eq "Windows")

# Combination with AND/OR
(user.department -eq "IT") and (user.jobTitle -contains "Engineer")

PowerShell Management

# Connect to Microsoft Graph
Connect-MgGraph -Scopes "User.Read.All","Group.ReadWrite.All"

# Get all users
Get-MgUser -All | Select DisplayName, UserPrincipalName, AccountEnabled

# Get all groups
Get-MgGroup -All | Select DisplayName, GroupTypes, SecurityEnabled

# Get group members
Get-MgGroupMember -GroupId "group-id" | Select Id

# Create user
New-MgUser -DisplayName "John Smith" -UserPrincipalName "jsmith@company.com" `
  -PasswordProfile @{Password="TempP@ss123!"; ForceChangePasswordNextSignIn=$true} `
  -AccountEnabled $true -MailNickname "jsmith"

# Add user to group
New-MgGroupMember -GroupId "group-id" -DirectoryObjectId "user-id"
